\begin{abstract}
For fixed-depth networks with widths proportional to the input dimension, we prove that the leading term of the average Bayes predictive KL is $\Pall(d)/(2n)$. The activation class includes centered normalized tanh, sigmoid and ReLU. Function distance controls parameter matching error, and mutual information bounds yield the Fisher spectral property. Conditional I--MMSE, Gaussian integration by parts and linear Gaussian reference distributions give matching upper and lower leading terms. The dimension limit precedes the sample-ratio limit; neither a risk limit at fixed sample ratio nor a locally Gaussian posterior is assumed.
\end{abstract}

\section{Model, activation class, and theorem}
Fix $L\ge1$, $k_0=d$, $k_\ell/d\to\gamma_\ell\in(0,\infty)$, and $\gamma_0=1$.
The entries of the layer weights $W^{(\ell)}\in\R^{k_\ell\times k_{\ell-1}}$ are mutually independent standard Gaussians, and the input $x\sim N(0,I_d)$ is independent of the weights. All layers share a fixed activation $\sigma_\ell\equiv\sigma$ satisfying
\[
\E\sigma(G)=0,\qquad\E\sigma(G)^2=1,\qquad G\sim N(0,1).
\]
Write $W=(W^{(\ell)})_{\ell=1}^L$ and $\theta=\theta(W)=(v,W)$, fix $v_i=1$, and define
\begin{equation}\begin{aligned}
h^{(0)}(x)&=x,\qquad
h^{(\ell)}(x)=\sigma_\ell\!\left(\frac{W^{(\ell)}h^{(\ell-1)}(x)}{\sqrt{k_{\ell-1}}}\right),\\
F_\theta^{(L)}(x)&=\frac{v^{\mathsf T}h^{(L)}(x)}{\sqrt{k_L}}.
\end{aligned}\end{equation}
The true parameter $\theta^0=(v,W^0)$ is drawn from the prior $P_0$ above. The training set $\mathcal D_n=\{(x_\mu,y_\mu)\}_{\mu=1}^n$ is generated by
\[
y_\mu=F_{\theta^0}^{(L)}(x_\mu)+\sqrt\Delta z_\mu,\qquad
z_\mu\sim N(0,1),\quad\Delta>0
\]
The training and test inputs are independent and identically distributed; the noise variables are mutually independent and independent of the parameter and all inputs.
Write $\Pout(y\mid\lambda)=N(y;\lambda,\Delta)$. The posterior under the true prior and channel is
\[
P(d\theta\mid\mathcal D_n)\propto P_0(d\theta)
\prod_{\mu=1}^n\Pout\!\left(y_\mu\mid F_\theta^{(L)}(x_\mu)\right).
\]
Only the hidden weights are learned, giving the parameter count
\[
\Pall(d)=\sum_{\ell=1}^Lk_{\ell-1}k_\ell,\qquad
c=\lim_d\frac{\Pall(d)}{d^2}
=\sum_{\ell=1}^L\gamma_{\ell-1}\gamma_\ell.
\]
Define the posterior predictive density and mean predictive KL divergence by
\begin{equation}\begin{aligned}
p_{\mathcal D_n}(y\mid x)
&=\E_{\theta\mid\mathcal D_n}\Pout\!\left(y\mid F_\theta^{(L)}(x)\right),\\
\epsilon_{\log,d,n}
&=\E_{\theta^0,\mathcal D_n,x}\KL\!\left(
\Pout\!\left(\cdot\mid F_{\theta^0}^{(L)}(x)\right)
\,\middle\|\,p_{\mathcal D_n}(\cdot\mid x)\right).
\end{aligned}\end{equation}
Write $\epsilon_{\log,d}(\alpha):=\epsilon_{\log,d,\lfloor\alpha d^2\rfloor}$.
Below, abbreviate $p=\Pall(d)$, $F_W=F=F_{\theta(W)}^{(L)}$, and $\theta'=\theta(V)$, and abbreviate $W^0$ by $W$ in prior averages. Differentiation, Gaussian integration, and OU couplings act only on $W\in\R^p$; homogeneous polynomial networks are defined separately.

Constants may depend on the depth, activation, width ratios, and noise, but not on $d,n$.

\subsection{Activation class}
\newthought*{Let $\mathcal A_{\rm bal}$ be the union of the following two classes.}
Both branches satisfy the centering and unit-variance conditions above.
\emph{Smooth odd branch:} $\sigma$ is bounded, odd, and belongs to $C^\infty$, with bounded derivatives of every fixed order.
\emph{Piecewise linear branch:} Let $c_0=\sqrt{2/\pi}$ and
\begin{equation}\label{eq:pl}
\sigma(z)=az+b(|z|-c_0),\qquad
b\ne0,\qquad a^2+b^2(1-2/\pi)=1.
\end{equation}
Both branches are globally Lipschitz.

Let $\operatorname{He}_j$ denote the probabilists' Hermite polynomial, and set $\psi_j=\operatorname{He}_j/\sqrt{j!}$ and $a_j=\E[\sigma(G)\psi_j(G)]$.
Accordingly, $\mu_{\ell,j}=\mu_j=\sqrt{j!}a_j$.
\begin{lemma}[Automatic existence of a higher-order odd coefficient]\label{lem:automatic-hermite}
If $\sigma$ is a bounded continuous odd function and $\E\sigma(G)^2=1$, then there is an odd integer $m\ge5$ such that $a_m\ne0$.
\end{lemma}
\begin{proof}
If the conclusion fails, oddness and completeness of the Hermite system give
$\sigma=a_1\psi_1+a_3\psi_3$ Gaussian almost everywhere; continuity makes the equality hold everywhere.
A bounded odd polynomial must be zero, contradicting unit variance.
\end{proof}
For the smooth branch, fix one such $m$; for the piecewise linear branch, take $m=6$, with $a_6\ne0$ verified in the next subsection.
The integer $m$ depends only on the activation, and constants may depend on $m$.

\begin{theorem}\label{thm:main}
In the model above, for every fixed depth $L\ge1$, any fixed $\sigma\in\mathcal A_{\rm bal}$ satisfies
\begin{equation}\label{eq:main}
\boxed{\lim_{\alpha\to\infty}\limsup_{d\to\infty}
\left|\alpha \epsilon_{\log,d}(\alpha)-\frac c2\right|=0.}
\end{equation}
This includes centered and normalized tanh and logistic sigmoid with any fixed positive slope, as well as centered and normalized ReLU.
\end{theorem}
The limit $d\to\infty$ precedes $\alpha\to\infty$; existence of the risk limit at fixed $\alpha$ is not required.

\subsection{tanh, sigmoid, and ReLU}
For any fixed $a>0$, let
\[
s_a^2=\E\tanh^2(aG)>0,\qquad \sigma_a(z)=\tanh(az)/s_a.
\]
This is a bounded smooth odd function with unit variance. Since $\tanh'=1-\tanh^2$,
each derivative is a polynomial in $\tanh(az)$ and is therefore bounded, so the function belongs to the smooth branch.
For fixed $b>0$, the logistic sigmoid $\ell_b(z)=(1+e^{-bz})^{-1}$ satisfies
$\ell_b(z)-1/2=\tanh(bz/2)/2$, with mean $1/2$ and variance $s_{b/2}^2/4$;
centering and normalizing it gives exactly $\sigma_{b/2}$.

For ReLU, direct integration gives
\[
\mu_R=\E G_+=\frac1{\sqrt{2\pi}},\qquad
v_R=\operatorname{Var}(G_+)=\frac12-\frac1{2\pi},\qquad
\sigma_R(z)=\frac{z_+-\mu_R}{\sqrt{v_R}}.
\]
Since $z_+=(z+|z|)/2$, this corresponds to
$a=b=1/(2\sqrt{v_R})$ in~\eqref{eq:pl}.
The entire piecewise linear branch is centered; orthogonality of odd and even functions gives its variance
$a^2+b^2(1-2/\pi)=1$.
Using $\operatorname{He}_6(z)=z^6-15z^4+45z^2-15$
and $\E|G|^{2j+1}=2^j j!\sqrt{2/\pi}$ gives
\[
\E[|G|\operatorname{He}_6(G)]=3\sqrt{2/\pi},\qquad
a_6=\frac{3b\sqrt{2/\pi}}{\sqrt{6!}}\ne0.
\]
In the piecewise linear branch, all odd coefficients $a_j$ with $j\ge3$ vanish; in the smooth odd branch, all even coefficients vanish.

\section{Prior moments and local estimates}
Let $q_\ell=\|h^{(\ell)}\|^2/k_\ell$ and $a(W,X)=\nabla_WF_{\theta(W)}^{(L)}(X)$.
Appendix Lemmas~\ref{an:moments}--\ref{an:ou} show that, for every fixed $r>0$,
\begin{equation}\label{eq:moments}
 \E|q_\ell-1|^r\le C_r d^{-r/2},\qquad
 \E|F_{\theta(W)}^{(L)}(X)|^r+\E\|a(W,X)\|^r\le C_r.
\end{equation}
There is a fixed $0<\delta\le1$ such that a standard Gaussian independent of $(W,X)$, denoted by $Z$, satisfies
\begin{equation}\label{eq:gradient-stability}
 W_\tau=e^{-\tau}W+\sqrt{1-e^{-2\tau}}Z,\qquad
 \E\|a(W_\tau,X)-a(W,X)\|^2\le C\tau^\delta\quad(0<\tau\le1).
\end{equation}
For $0<t\le1/2$, write $W(t)=\sqrt{1-t}W+\sqrt tZ$. The first-order expansion satisfies
\begin{equation}\label{eq:ou-rem}
 \begin{aligned}
 F_{\theta(W(t))}^{(L)}-F_{\theta(W)}^{(L)}&=\sqrt t\,a(W,\cdot)^TZ+E_t,\\
 \E\|E_t\|_{L_X^2}^2&\le Ct^{1+\delta}.
 \end{aligned}
\end{equation}
The constants and exponent are independent of the perturbation parameter; the dimension threshold may depend on this parameter, but is uniform on compact intervals bounded away from zero. All local bounds below are used above this threshold.

\section{Global function comparison and parameter matching}
We compare the actual outputs through homogeneous networks, then match the parameters. Set
\[
N=m^L,\qquad S=1+m+\cdots+m^{L-1},\qquad B=a_m^S\ne0.
\]
Replace each layer's activation by $z^m/\sqrt{m!}$, keeping the weights and readout. Denote its coefficient of the highest input degree by $C_N(W)\in\Sym^N\R^d$, and define
\[
H_m(W)=I_N(C_N(W)),\qquad T_W=BH_m(W).
\]
Let $\Pi_N$ be the orthogonal projection from $L^2(P_X)$ onto the Hermite subspace of order $N$. The map $I_N$ sends symmetric tensors to polynomials in this subspace:
for $v\ne0$, set $I_N(v^{\otimes N})(X)=\|v\|^N\operatorname{He}_N(\langle v,X\rangle/\|v\|)$, and extend linearly.
Its normalization satisfies
\begin{equation}\label{fp:actual-isometry}
 \langle I_N(C),I_N(D)\rangle_{L_X^2}=N!\langle C,D\rangle.
\end{equation}
All tensors carry the Hilbert norm inherited by the symmetric tensor space.

\begin{lemma}[Uniform estimate for function inner products]\label{lem:cross}
Let $\Pi_d$ be the original Gaussian parameter prior, and let $\Gamma(\Pi_d,\Pi_d)$ denote the set of joint distributions with both marginals equal to $\Pi_d$. Then
\begin{equation}\label{eq:cross}
e_d:=\sup_{\pi\in\Gamma(\Pi_d,\Pi_d)}
\E_\pi|\langle T_W,F_{\theta'}^{(L)}-T_V\rangle_{L_X^2}|\longrightarrow0.
\end{equation}
The coupling only needs the correct prior marginals; $W,V$ need not be independent.
\end{lemma}
\begin{proof}
Appendix~\ref{app:closure} gives a polynomial approximation of the actual output. Appendix~\ref{app:cuts}
compares terms and controls the coupling error, yielding~\eqref{fp:global-function-pairing}, which is the present lemma.
\end{proof}

For any coupling with the correct marginals, let
$D=\E\norm{T_W-T_V}^2$ and $B_F=\E\norm{F_\theta^{(L)}-F_{\theta'}^{(L)}}^2$.
Applying \eqref{eq:cross} to the original coupling, the transposed coupling, and the two diagonal terms gives
\[
|\E\langle T_W-T_V,F_\theta^{(L)}-F_{\theta'}^{(L)}\rangle-D|\le4e_d.
\]
Hence
\[
D\le\sqrt{DB_F}+4e_d\le(D+B_F)/2+4e_d,
\quad D\le B_F+8e_d.
\]
The homogeneous global matching theorem gives a finite family of measurable maps $\mathcal G_d$ such that
\begin{equation}\label{eq:matching}\begin{aligned}
\log|\mathcal G_d|&=O(d\log d)=o(p),\\
\frac1p\E\min_{g\in\mathcal G_d}\norm{W-gV}^2
&\le C\E\norm{H_m(W)-H_m(V)}^2+o_d(1)\\
&\le C\E\norm{F_\theta^{(L)}-F_{\theta'}^{(L)}}^2+o_d(1).
\end{aligned}\end{equation}
The error is uniform over couplings; the proof is in Appendix~\ref{app:matching}. Row sign changes serve only as finite labels for controlling conditional entropy and need not preserve the actual output.

\section{Mutual information bounds and posterior contraction}
Write $I_{d,n}=I(\theta^0;\mathcal D_n)=I(W;\mathcal D_n)$. The Bayesian definitions and conditional independence of the samples give
\begin{equation}\label{eq:increment}
\epsilon_{\log,d,n}=I_{d,n+1}-I_{d,n},\qquad
\epsilon_{\log,d,n+1}\le \epsilon_{\log,d,n}.
\end{equation}
The second relation follows because predictive conditional entropy decreases while noise entropy stays constant.

\subsection{Coarse risk upper bound}
Fix a sufficiently large $u$, and let $n=\lfloor ud^2\rfloor$ and $t=(1+n/p)^{-1}$.
Since $t\to(1+u/c)^{-1}>0$, for sufficiently large dimension we can use \eqref{eq:sample-information-upper}, derived in the next section solely from prior moments and the OU remainder. Moreover,
\[
 \E\log\det(I_p+ntA/\Delta)\le nt\E\tr A/\Delta\le Cp,
 \qquad (n/p)t^{1+\delta}\le1,
\]
Thus $I_{d,n}\le\frac p2\log(1+n/p)+Cp$.
By \eqref{eq:increment}, $n\epsilon_{\log,d,n}\le I_{d,n}$, and hence
\begin{equation}\label{eq:coarse}
 \limsup_d\epsilon_{\log,d}(u)\le C\frac{\log u}{u}
 \qquad(u\text{ sufficiently large}).
\end{equation}

\subsection{Posterior parameter distance and conditional entropy}
Given $\mathcal D_n$, draw independent posterior samples $W,V$; after averaging over the data, both retain the original prior marginals.
Write $B_X=F_{\theta(W)}^{(L)}(X)-F_{\theta(V)}^{(L)}(X)$.
Using the posterior predictive density as an intermediate point, the Hellinger triangle inequality and $H^2\le\KL$ give
\begin{equation}\label{eq:hellinger}
 \E[1-e^{-B_X^2/(8\Delta)}]\le2\epsilon_{\log,d,n},
\end{equation}
The exponential term is the affinity of the two Gaussian densities. For fixed $q>2$, we have
\[
 z^2\le C_q(1+|z|^q)^{2/q}
          (1-e^{-z^2/(8\Delta)})^{1-2/q}
\]
(verified separately for $|z|\le1$ and $|z|>1$). H\"older's inequality and \eqref{eq:moments} give
$\E B_X^2\le C_q\epsilon_{\log,d,n}^{1-2/q}$. Define
\[
 r_{d,n}=\frac1p\E\min_{g\in\mathcal G_d}\|W-gV\|^2,
 \qquad r_*(u)=\limsup_d r_{d,\lfloor ud^2\rfloor}.
\]
Use this posterior coupling in \eqref{eq:matching} and take $q=2/\eta$. Then \eqref{eq:coarse} gives
\begin{equation}\label{eq:rate}
 r_*(u)\le C_\eta(\log u/u)^{1-\eta},\qquad0<\eta<1.
\end{equation}

Choose a measurable optimal label $\widehat g$, and let
$V_t=\sqrt{1-t}W+\sqrt tU$, where $U$ is a standard Gaussian independent of the variables above and $0\le t\le1/2$.
Given $\mathcal D_n$, $V$ and $V_t$ are independent; moreover,
\[
 \E\|V_t-\sqrt{1-t}\widehat gV\|^2=p[t+(1-t)r_{d,n}].
\]
Conditioning on $(\mathcal D_n,V,\widehat g)$, Gaussian maximum entropy and Jensen's inequality give
\begin{equation}\label{info:matched-entropy}
 h(V_t\mid\mathcal D_n)
 \le\log|\mathcal G_d|+\frac p2\log\{2\pi e[t+(1-t)r_{d,n}]\}.
\end{equation}
Here the conditional entropy of the label is at most $\log|\mathcal G_d|$.
At $t=0$, $V_0=W$; subtracting the bound above from $h(W)=\frac p2\log(2\pi e)$ and using \eqref{eq:rate} gives
\begin{equation}\label{eq:sample-information-lower}
 \liminf_d\frac{I_{d,\lfloor\alpha d^2\rfloor}}p
 \ge\frac{1-\eta}{2}\log\alpha
       -\frac{1-\eta}{2}\log\log\alpha-C_\eta.
\end{equation}
Here $\eta\in(0,1)$ is fixed and $\alpha$ is sufficiently large.
For $t>0$, write $J_n(t)=I(W;V_t\mid\mathcal D_n)$.
Subtracting the conditional noise entropy from \eqref{info:matched-entropy} gives
\begin{equation}\label{alt:oracle-cost}
 J_n(t)\le\log|\mathcal G_d|+\frac p2\log\left(1+\frac{(1-t)r_{d,n}}t\right)
 \le\log|\mathcal G_d|+\frac{p r_{d,n}}{2t}.
\end{equation}

\section{Linear Gaussian reference distributions and mutual information bounds}
For $0<t\le1/2$, the reverse representation is
$W=\sqrt{1-t}V_t+\sqrt t\xi$, where $\xi$ is a standard Gaussian independent of $V_t$.
Write $f_W=F_{\theta(W)}^{(L)}$, $a(x)=\nabla f_{V_t}(x)$, and
$A=\E_Xa(X)a(X)^T$. For fixed $t$ and $d\ge d_0(t)$, \eqref{eq:ou-rem} and \eqref{eq:moments} give
\[
 f_W(x)-f_{V_t}(x)=\sqrt t\,a(x)^T\xi+R_t(x),\qquad
 \E\|R_t\|_{L_X^2}^2\le Ct^{1+\delta},\quad
 \E\|a(X)\|^4\le C.
\]


\subsection{Relative entropy decomposition and mutual information increments}
Given $V_t,X_{1:j}$, let $\Phi_j$ have row $i$ equal to $a(X_i)^T$, and define
\[
 Q_j=N\!\left((f_{V_t}(X_i))_{i\le j},S_j\right),
 \qquad S_j=\Delta I_j+t\Phi_j\Phi_j^T.
\]
All reference distributions share an auxiliary coefficient $\zeta\sim N(0,I_p)$ and are generated by the linear model
$f_{V_t}(X_i)+\sqrt t\,a(X_i)^T\zeta+\sqrt\Delta Z_i$,
so their sample marginals are consistent. Inputs follow the true distribution, and the auxiliary coefficient is unobserved.
Let $P_j$ be the true data distribution given $V_t$, and define
\[
 C_j=\E\log\frac{p(Y_{1:j}\mid W,X_{1:j})}
 {q_j(Y_{1:j}\mid V_t,X_{1:j})},\qquad
 B_j=\E_{V_t}\KL(P_j\|Q_j).
\]
The distributions $P_j,Q_j$ include the same input marginal. By relative entropy decomposition and the Markov chain $V_t-W-\mathcal D_j$,
\begin{equation}\label{info:reference-identity}
 I_{d,j}+J_j(t)=\frac p2\log(1/t)+C_j-B_j.
\end{equation}
Marginal consistency gives $B_n\ge B_j\ge0$; using $J_n(t)\ge0$ then yields
\begin{equation}\label{alt:block-entropy}
 I_{d,n}\le\frac p2\log(1/t)+C_n,\qquad
 I_{d,n}-I_{d,j}\le C_n-C_j+J_j(t).
\end{equation}

\subsection{Nonlinear error in cross entropy}
Let $L_j=\sqrt t\Phi_j\xi$ and $R_j=(R_t(X_i))_{i\le j}$, and set
$H_j=\E\log\det(I_p+t\Phi_j^T\Phi_j/\Delta)$.
The Gaussian density formula and Sylvester's identity give
\[
 C_j=H_j/2+\E L_j^TS_j^{-1}R_j+\tfrac12\E R_j^TS_j^{-1}R_j,
\]
The linear and noise terms cancel because
$\E[L_j^TS_j^{-1}L_j\mid V_t,X_{1:j}]+\Delta\tr S_j^{-1}=j$.
Moreover, since $S_j\succeq\Delta I_j$,
\[
 \E L_j^TS_j^{-1}L_j=\E\tr(t\Phi_j\Phi_j^TS_j^{-1})\le p,
 \qquad \E R_j^TS_j^{-1}R_j\le Cjt^{1+\delta}.
\]
Weighted Cauchy--Schwarz gives
\begin{equation}\label{alt:cross-entropy}
 |C_j-H_j/2|\le C\sqrt{pj t^{1+\delta}}+Cjt^{1+\delta}.
\end{equation}
Independence of $R_j$ and $\xi$ is not required here.

\subsection{Comparison of empirical and population determinants}
Set $\overline H_j=\E\log\det(I_p+jtA/\Delta)$. For $M\succeq I_p$,
$D^2(-\log\det)(M)[H,H]=\tr(M^{-1}HM^{-1}H)\le\|H\|_{\rm HS}^2$.
Condition on $V_t$ and apply a second-order Taylor expansion to $I_p+(t/\Delta)\sum_{i\le j}a(X_i)a(X_i)^T$
at its mean. Conditional independence and the fourth moment of the gradient give
\[
 0\le\overline H_j-H_j
 \le\frac{jt^2}{2\Delta^2}\E\|a(X)a(X)^T-A\|_{\rm HS}^2\le Cjt^2.
\]
For $n>j\ge1$, apply $(1+nt\lambda/\Delta)/(1+jt\lambda/\Delta)\le n/j$ to each eigenvalue to obtain
\[
 H_n-H_j\le\overline H_n-\overline H_j+Cjt^2\le p\log(n/j)+Cjt^2.
\]
By the first inequality in \eqref{alt:block-entropy}, \eqref{alt:cross-entropy},
and $H_n\le\overline H_n$, the cumulative information satisfies
\begin{equation}\label{eq:sample-information-upper}\begin{aligned}
 \frac{I_{d,n}}p\le{}&\frac12\log\frac1t
  +\frac1{2p}\E\log\det(I_p+ntA/\Delta)\\
 &+C\sqrt{\frac np t^{1+\delta}}+C\frac np t^{1+\delta}.
\end{aligned}\end{equation}

\section{Information matrix spectrum and predictive lower bound}
Let $a_W(x)=\nabla_WF_\theta^{(L)}(x)$ and
$\mathcal I_W=\E_Xa_Wa_W^T/\Delta$, and write
\[
 D_d(r)=\E\frac1p\tr\{r(p\mathcal I_W+rI)^{-1}\},\qquad
 \nu_d=\frac1p\E\sum_{i=1}^p\delta_{\lambda_i(p\mathcal I_W)}.
\]
The second moment of the gradient gives $\int\lambda\,d\nu_d=\E\tr\mathcal I_W\le C$.

\subsection{Mutual information bounds rule out a positive fraction of spectral mass near zero}
Write $q_d(s)=\nu_d([0,s])$. Suppose for contradiction that
$b=\lim_{s\downarrow0}\limsup_dq_d(s)>0$, and fix $0<\beta<b$.
Fix $1/(1+\delta)<a<1$, and take
$n=\lfloor\alpha d^2\rfloor$ and $t=\alpha^{-a}$ in \eqref{eq:sample-information-upper}.
For each sufficiently large fixed $\alpha$, choose a subsequence of dimensions along which
$q=q_d(\alpha^{-1})\ge\beta$.
Since $n/p\to\alpha/c$, write $Q_d=nt/p\le C\alpha^{1-a}$.
The matrices $A/\Delta$ and $\mathcal I_W$ have the same distribution, so Jensen's inequality gives
\[
 \begin{aligned}
 \frac1p\E\log\det(I_p+ntA/\Delta)
 &=\int\log(1+Q_d\lambda)\,d\nu_d\\
 &\le\log(1+Q_d/\alpha)
 +(1-q)\log\left(1+\frac{CQ_d}{1-q}\right)\\
 &\le(1-\beta)(1-a)\log\alpha+C.
 \end{aligned}
\]
At $q=1$, the second term is zero by continuous extension; the last step uses
$-(1-q)\log(1-q)\le1/e$. Thus, along this subsequence, the information upper bound is at most
\[
 \frac{1-\beta(1-a)}2\log\alpha+C+o_\alpha(1).
\]
Fix $\eta=\beta(1-a)/2$ in \eqref{eq:sample-information-lower}.
Its coefficient of $\log\alpha$ exceeds that above by $\beta(1-a)/4$, and the remaining loss is
$O_\eta(\log\log\alpha)$, giving a contradiction as $\alpha\to\infty$. Hence $b=0$.
Finally, use $D_d(r)\le q_d(s)+r/s$ and take $d\to\infty$,
$r\downarrow0$, and $s\downarrow0$ in this order to obtain
\begin{equation}\label{eq:deficit}
 D_*(r):=\limsup_dD_d(r),\qquad \lim_{r\downarrow0}D_*(r)=0.
\end{equation}

\subsection{Predictive KL lower bound}
Appendix~\ref{app:information} applies OU smoothing to a regularized test field and proves
\begin{equation}\label{eq:lower-unified}
\epsilon_{\log,d,n}\ge
\frac{p[1-D_d(r)-C\sqrt r]_+^2}
{2\{n(1+C\sqrt r)^2+Cp/r^{1+2/\delta}\}},\quad 0<r\le1,\ d\ge d_0(r).
\end{equation}
Fix $r$ and take $d\to\infty$ followed by $\alpha\to\infty$ to obtain
\[
 \liminf_{\alpha\to\infty}\alpha\liminf_d\epsilon_{\log,d}(\alpha)
 \ge \frac{c[1-D_*(r)-C\sqrt r]_+^2}{2(1+C\sqrt r)^2}.
\]
Finally, let $r\downarrow0$ using \eqref{eq:deficit} to obtain
\begin{equation}\label{eq:lower-final}
\liminf_{\alpha\to\infty}\alpha\liminf_d\epsilon_{\log,d}(\alpha)\ge c/2.
\end{equation}

\section{Analytic estimates and integration-by-parts domains for fixed-depth networks}\label{an:appendix}\label{an:section}\label{app:local}
Use $h^{(\ell)},F$ from the model and write $u^{(\ell)}=W^{(\ell)}h^{(\ell-1)}/\sqrt{k_{\ell-1}}$.
Fix the depth $L$, with $cd\le k_\ell\le Cd$; expectations are over the independent Gaussian weights and input.
The moment estimates allow different fixed, centered, unit-variance Lipschitz activations in each layer.
Moment orders are fixed first; constants may depend on the activations, depth, and width ratios. OU constants are independent of the perturbation parameter; the dimension threshold may depend on that parameter, uniformly on compact intervals bounded away from zero.

\subsection{All fixed-order moments of the output and hidden norms}
\begin{lemma}\label{an:moments}
Let $q_\ell=\|h^{(\ell)}\|^2/k_\ell$. For every fixed $r>0$,
\[
 \mathbb E q_\ell^r\le C_r,\qquad
 \mathbb E|q_\ell-1|^r\le C_r d^{-r/2},\qquad
 \mathbb E|F|^r\le C_r.
\]
Individual features and products of a fixed number of features have moment bounds of every fixed order, including across layers or for coupled inputs with the Gaussian marginals above.
\end{lemma}
\begin{proof}
Write $b(q)=\E\sigma(\sqrt qG)$ and $v(q)=\E\sigma(\sqrt qG)^2$.
The Lipschitz property, centering, and unit variance give the global bounds
\begin{equation}\label{an:scalar}
 \E|\sigma(\sqrt qG)|^s\le C_s(1+q^{s/2}),\qquad
 |b(q)|+|v(q)-1|\le C|q-1|\quad(q\ge0).
\end{equation}
Here we use $|\sigma(z)|\le C(1+|z|)$ and
$|\sqrt q-1|(\sqrt q+1)=|q-1|$.
If the $Y_i$ are independent and centered conditional on $\mathcal F$, then
\begin{equation}\label{an:sample}
 \E\left[\left|N^{-1}\sum_iY_i\right|^{2m}\middle|\mathcal F\right]
 \le C_mN^{-m}\max_i\E[|Y_i|^{2m}\mid\mathcal F].
\end{equation}
Expand the $2m$th power. Terms with an index appearing once vanish; the remaining terms contain at most $m$ distinct indices.
There are $O_m(N^m)$ such terms, controlled by H\"older's inequality.

Conditional on the preceding layer, the row outputs in this layer are independent copies of $\sigma(\sqrt{q_{\ell-1}}G)$.
Jensen's inequality and~\eqref{an:scalar} first give $\E q_\ell^{2m}\le C_m$ by induction.
Applying~\eqref{an:sample} to $Y_i=(h_i^{(\ell)})^2-v_\ell(q_{\ell-1})$ then gives
\[
 \E|q_\ell-1|^{2m}\le C_md^{-m}+C_m\E|q_{\ell-1}-1|^{2m}=O(d^{-m}).
\]
The initial case $q_0=d^{-1}\sum_iX_i^2$ follows from the same expansion.
The all-ones readout decomposes as
\[
 F=\sqrt{k_L}b_L(q_{L-1})+
 k_L^{-1/2}\sum_i\{h_i^{(L)}-b_L(q_{L-1})\}.
\]
The $2m$th moment of the first term is at most $C_mk_L^m\E|q_{L-1}-1|^{2m}=O(1)$;
the second term is controlled by~\eqref{an:sample} and the positive-moment bounds.
Choose $2m\ge r$ to obtain the required order. The conditional row estimates and H\"older's inequality give the moments of individual features and fixed products, respectively.
\end{proof}

\subsection{All fixed-order moments of the gradient}
Let $A_\ell=W^{(\ell)}/\sqrt{k_{\ell-1}}$ and $B_\ell=\|A_\ell\|_{\rm op}$. Below, $P$ denotes a nonnegative polynomial of fixed degree in $1+\|X\|/\sqrt d$ and finitely many $B_\ell$; it may also include normalized operator norms of a second independent collection of Gaussian matrices. Every fixed-order moment of such a $P$ is uniformly bounded: $1/4$-nets of the two spheres contain at most
$9^m,9^r$ points, respectively. Net approximation and scalar Gaussian tails give, for an $m\times r$ standard Gaussian matrix $G$,
\[
 \Pr\{\|G\|_{\rm op}>2u\}\le2\,9^{m+r}e^{-u^2/2}.
\]
With bounded width ratios, integration gives all fixed-order moments of the normalized operator norm.

\begin{lemma}\label{an:derivatives}
For Lipschitz activations, the weak gradient satisfies $\mathbb E\|\nabla_WF\|^r\le C_r$ for every fixed $r>0$.
\end{lemma}
\begin{proof}
Forward recursion gives $\|h^{(\ell)}\|/\sqrt{k_\ell}\le P$. Define
\[
 \beta_L=k_L^{-1/2}\boldsymbol1,\quad
 \lambda_\ell=\operatorname{diag}(\sigma_\ell'(u^{(\ell)}))\beta_\ell,\quad
 \beta_{\ell-1}=A_\ell^T\lambda_\ell.
\]
Bounded slopes give $\|\beta_\ell\|+\|\lambda_\ell\|\le P$, and
\[
 \nabla_{W^{(\ell)}}F=\lambda_\ell(h^{(\ell-1)})^T/\sqrt{k_{\ell-1}},
 \qquad \|\nabla_WF\|^2\le P. \tag{A3}\label{an:gradient}
\]
The network is locally Lipschitz, so the almost-everywhere chain rule gives its weak gradient.

\end{proof}
\subsection{Gradient variation for the two activation classes}
\begin{lemma}\label{an:gradient-ou}
Let $W'=\sqrt{1-t}W+\sqrt tZ$, where $Z$ is independent standard Gaussian and $0<t\le1/2$.
For any norm polynomial $P$ and fixed $t>0$, both activation classes satisfy, for sufficiently large $d$,
\[
 \E[P\|\nabla F(W',X)-\nabla F(W,X)\|^2]
 \le C_Pt^{1/4}.
\]
The constant is independent of $t$, and the dimension threshold is uniform on compact intervals bounded away from zero.
\end{lemma}
\begin{proof}
For each fixed $s>0$, the backward coordinates satisfy
\begin{equation}\label{an:coordinate}
 \E[P|\beta_{\ell,i}|^{2s}]\le C_{P,s}k_\ell^{-s}.
\end{equation}
This is immediate in the final layer. In other layers, conditional on $h=h^{(\ell)}$, the next Gaussian matrix is invariant in distribution under right multiplication by any orthogonal transformation fixing $h$,
and $\beta_\ell$ rotates accordingly. Its norm is controlled by downstream operator norms independent of $h$,
so $\E[\|\beta_\ell\|^{2s}\mid h]\le C_s$.
Decompose along $h$ and the orthogonal complement of $h$ and use the coordinate moments of the uniform spherical direction in the latter to obtain
\[
 \E[|\beta_{\ell,i}|^{2s}\mid h]
 \le C_s\left(k_\ell^{-s}+|h_i|^{2s}/\|h\|^{2s}\right).
\]
When $h=0$, the distribution is invariant under all rotations, and the ratio above is set to zero.
On $q_\ell\ge1/2$, use the positive feature moments; on its complement, use that the ratio is at most one and
$\Pr(q_\ell<1/2)\le C_md^{-m}$ (Lemma~\ref{an:moments}).
This gives the unweighted coordinate moments. Cauchy--Schwarz applied to $P$ then yields~\eqref{an:coordinate}.
Here, for $U$ uniform on $S^{n-1}$, the radial decomposition of a standard Gaussian gives
$\E|U_1|^{2s}=\Gamma(s+1/2)\Gamma(n/2)/(\sqrt\pi\Gamma(n/2+s))\le C_sn^{-s}$.

Forward recursion gives
$\|A'_\ell-A_\ell\|_{\rm op}\le\sqrt tP$ and
$\|h'^{(\ell)}-h^{(\ell)}\|/\sqrt{k_\ell}\le\sqrt tP$.
Write $D_\ell=\operatorname{diag}(\sigma'(u^{(\ell)}))$.
In the smooth case, conditional on the preceding layer, $u'_i-u_i$ is centered Gaussian with variance at most
$C(tq_{\ell-1}+\|h'-h\|^2/k_{\ell-1})$, so
$\E|\sigma'(u'_i)-\sigma'(u_i)|^4\le C\E|u'_i-u_i|^4\le Ct^2$.
In the piecewise-linear case, let $\mathcal G=\{q_{\ell-1},q'_{\ell-1}\ge1/2\}$.
Conditional on the preceding layer, on $\mathcal G$ the two preactivations are centered Gaussian with correlation
$\sqrt{1-t}\langle h,h'\rangle/(\|h\|\|h'\|)$, and their angle satisfies
\[
 \omega_\ell\le C\left(\sqrt t+
 \frac{\|h-h'\|}{\sqrt{\|h\|\|h'\|}}\right)\le\sqrt tP.
\]
The uniform polar angle of a two-dimensional Gaussian gives conditional sign-disagreement probability $\omega_\ell/\pi$.
Since $\Pr(\mathcal G^c)\le C_md^{-m}$,
$\Pr(u_iu'_i<0)\le C\sqrt t+C_md^{-m}$.
Zero preactivations have probability zero on $\mathcal G$. Choose any bounded slope at zero; its contribution is absorbed by $\Pr(\mathcal G^c)$. Thus
$\E|\sigma'(u'_i)-\sigma'(u_i)|^4\le C\sqrt t+C_md^{-m}$.
Both cases therefore give
\[
 \E[P\|(D'_\ell-D_\ell)\beta_\ell\|^2]
 \le\sum_i(\E[P^2\beta_{\ell,i}^4])^{1/2}
       (\E|\sigma'(u'_i)-\sigma'(u_i)|^4)^{1/2}
 \le C_P(t^{1/4}+d^{-1}).
\]
Starting from the same readout, recurse backward using
$\lambda'_\ell-\lambda_\ell
 =D'_\ell(\beta'_\ell-\beta_\ell)+(D'_\ell-D_\ell)\beta_\ell$ and
$\beta'_{\ell-1}-\beta_{\ell-1}
 =A_\ell^T(\lambda'_\ell-\lambda_\ell)+(A'_\ell-A_\ell)^T\lambda'_\ell$.
Operator norms only change $P$, and the matrix-difference term is $O(t)$. Backward induction over fixed depth and the product differences in the gradient-block formula~\eqref{an:gradient} preserve this bound.
For fixed $t>0$, choose $d^{-1}\le t^{1/4}$ to conclude. In the smooth case the bound is $O(t)$ throughout.
\end{proof}

\subsection{OU gradient variation and remainder}
\begin{lemma}\label{an:ou}
Equations~\eqref{eq:gradient-stability} and~\eqref{eq:ou-rem} hold with the common exponent $\delta=1/4$, with the dimension threshold as specified in this section.
\end{lemma}
\begin{proof}
Let $c_N$ denote the energy, averaged over $X$, of parameter Hermite degree $N$, write $a_t=\sqrt{1-t}$,
and set $G(t)=\E\|\nabla F(W(t),X)-\nabla F(W,X)\|^2$.
Orthogonality and $\partial_iH_\nu=\sqrt{\nu_i}H_{\nu-e_i}$ give
\[
 G(t)=2\sum_{N\ge1}N(1-a_t^{N-1})c_N\le Ct^\delta.
\]
The last step follows from Lemma~\ref{an:gradient-ou}, with $\delta=1/4$ in both cases.
Setting $t=1-e^{-2\tau}$ gives the first equation;
for $t>1/2$, enlarge the constant using the second gradient moment.
For the remainder in~\eqref{eq:ou-rem}, the same orthogonal expansion and integration by parts in $Z$ give
\[
 \E\|E_t\|_{L_X^2}^2=\sum_{N\ge1}\{2(1-a_t^N)+tN-2tNa_t^{N-1}\}c_N.
\]
For $N\ge2$, we have $a_t^N\ge1-Nt/2$; for $N=1$, the expression in braces equals
$(1-a_t)^2\le t^2$. Therefore
\begin{equation}\label{an:hermite}
 \E\|E_t\|_{L_X^2}^2\le t^2\E\|\nabla F\|^2+tG(t)\le Ct^{1+\delta}.
\end{equation}
First compute for finite Hermite sums, then pass to the limit in the Gaussian Sobolev norm.
\end{proof}

\subsection{The domain of Gaussian integration by parts}
\begin{lemma}\label{an:sobolev}
Let $g:\mathbb R^p\to\mathbb R^p$ belong to Gaussian $W^{1,2}$; the range may also carry an independent external $L^2$ variable. Define
$\delta g=\operatorname{div}g-W\cdot g$. Its closed extension satisfies
\[
 \mathbb E(\delta g)^2
 =\mathbb E\|g\|^2+\mathbb E\sum_{i,j}(\partial_i g_j)(\partial_jg_i)
 \le\mathbb E\{\|g\|^2+\|Dg\|_{\rm HS}^2\}. \tag{A13}\label{an:divergence}
\]
\end{lemma}
\begin{proof}
For smooth compactly supported fields, expand $(\delta g)^2$ and integrate by parts twice in each coordinate to obtain the equality, then use $2uv\le u^2+v^2$.
For general $g$, first multiply by a radial cutoff $\chi_R$ ($\|\nabla\chi_R\|_\infty\le C/R$), then convolve on a compact set.
The tails and cutoff derivatives are controlled by $g,Dg\in L^2$. The Gaussian density has positive upper and lower bounds on compact sets,
so the resulting smooth compactly supported fields converge in Gaussian $W^{1,2}$ to $g$.
The inequality above makes their divergences converge in $L^2$. Integration by parts against test functions identifies the limit as the distributional divergence,
and the equality passes to the limit. For the external variable, first approximate by simple functions and then integrate.
\end{proof}


\section{Gram matrix bounds for symmetric tensors}\label{tg:section}
Fix integers $m\ge5$ and $q\ge1$. Suppose the unit rows $w_1,\ldots,w_k$ satisfy
$\max_{i\ne j}|\langle w_i,w_j\rangle|\le\delta$. For a multi-index $r\in\mathbb N^k$ with $|r|=q$, define
\[
 P_r=\operatorname{Sym}\bigotimes_jw_j^{\otimes mr_j},\qquad
 Z_{r,i}=\operatorname{Sym}\bigotimes_jw_j^{\otimes(mr_j-\mathbf1_{j=i})},\quad r_i>0.
\]
Here $\operatorname{Sym}$ averages over all slot permutations, and the norm is the inherited Hilbert tensor norm.
Let $\mathcal F$ and $\mathcal F^-$ have columns $P_r$ and $Z_{r,i}$, respectively. Then
\begin{equation}\label{tg:frame}
 cI\preceq\mathcal F^*\mathcal F\preceq CI,\qquad
 cI\preceq(\mathcal F^-)^*\mathcal F^-\preceq CI,
\end{equation}
provided $\delta+\sum_{h=1}^q k^h\delta^{mh-2}=o(1)$; the constants $c,C>0$ depend only on $m,q$.

The exact pairing formula is
\[
 \left\langle\operatorname{Sym}\bigotimes_{a=1}^Ru_a,
 \operatorname{Sym}\bigotimes_{a=1}^Rv_a\right\rangle
 =\frac1{R!}\sum_{\pi\in\mathfrak S_R}\prod_a\langle u_a,v_{\pi(a)}\rangle.
\]
If a column has label multiplicities $(n_j)$, the diagonal terms without mismatches sum to $\prod_jn_j!/R!$,
which has a positive lower bound depending only on the total degree; the remaining diagonal terms are $O(\delta)$.
The multiplicities of the full and deletion columns are $mr$ and $mr-e_i$, respectively, and are distinct within each family: in the latter, the residues modulo $m$ uniquely determine $i$.
Off-diagonal terms within the same support have at least one mismatch and hence are $O(\delta)$; the number of such columns is bounded by a fixed constant.
If another column contains $h\ge1$ new labels, at least $mh-1$ slots are mismatched,
so its inner product is at most $C\delta^{mh-2}$. There are at most $Ck^h$ such columns.
Thus the absolute row sums of the difference between each Gram matrix and the positive diagonal matrix above are at most
$C(\delta+\sum_hk^h\delta^{mh-2})=o(1)$. The Schur bound for symmetric matrices gives~\eqref{tg:frame}.

If the row lengths lie in a fixed positive interval, each column is merely multiplied by a scalar with fixed positive upper and lower bounds, so the conclusion is unchanged.
For Gaussian rows normalized by the input width, row lengths converge uniformly to one, with
$\delta=O(\sqrt{\log d/d})$ and $k\asymp d$, so the perturbation vanishes. The probability of the complementary event can be bounded by any fixed negative power.

\section{Tensor estimates for computation-tree expansions}
\label{fp:core-section}

All spaces are real Euclidean spaces and their Hilbert tensor products. The notation $\|\cdot\|_{\mathrm{HS}}$ denotes the tensor Hilbert norm,
and $\operatorname{Flat}_{S|S^c}$ denotes matricization by groups of index positions.
A \emph{mode} is an index position of a coefficient tensor; different modes may take the same value or be copies of the same original node index. Layer widths lie in $[cd,Cd]$;
the depth, polynomial degree, number of layers involved, and number of operations are all fixed.
Write $\delta_d=L_d/\sqrt d$ and $\rho_d=L_d/d$, where $L_d$ is a fixed power of a logarithm that may increase over finitely many
operations. Probability estimates use the Gaussian marginal of each network separately;
on the resulting events, the estimates below hold for any coupling of the two networks.

\subsection{Coefficient tensor norm bounds and linear summation}
\label{fp:core-invariant-section}

\paragraph{Coefficient tensors of computation trees.}
Expand downward from the normalized readout vector along the original computation tree.
Current modes represent original node indices. One expansion of mode $p$ is
\begin{equation}
 J e_a=\bigotimes_{t=1}^{q}u_a^{(t)},\qquad q\geq1,
 \qquad \|U_t\|_{\mathrm{op}}\leq C,
 \quad U_t=(u_a^{(t)})_a .
 \label{fp:core-replacement}
\end{equation}
The factor $e_a$ retains the parent index; factors with the same parent index continue to share it. Expansion may stop at any node, retaining the entire tensor map below it.
Maps depending on both networks act only on stopping leaves, and each map outputs one factor. Nonlinear maps with multiple output factors must first be expanded along the original computation tree and incorporated into the coefficient tensor.

\refstepcounter{lemma}
\paragraph{Lemma~\thelemma\ (Norm of the expansion operator).}
The composition of these expansions over all retained factors is called the expansion operator; any fixed number of compositions is bounded. Let
$\Delta_q e_a=e_a^{\otimes q}$. Then
$J=(\bigotimes_tU_t^*)\Delta_q$, and $\Delta_q$ is an isometric embedding, so
$\|J\|\leq\prod_t\|U_t\|$.
Likewise, if $F_j:e_a\mapsto f_a^{(j)}$ are bounded, the product feature with a shared index,
$e_a\mapsto\bigotimes_jf_a^{(j)}$, is bounded.
Its Gram matrix is $\mathop{\circ}_jF_j^*F_j$,
and for positive semidefinite matrices $X\preceq bI,Y\succeq0$,
$X\circ Y\preceq b\operatorname{diag}(Y)$.

\refstepcounter{lemma}
\paragraph{Lemma~\thelemma\ (Expansion preserves norm and slice bounds).}
\label{fp:core-closure}
Suppose $E$ has $r$ modes and satisfies
\begin{equation}
 \|E\|_{\mathrm{HS}}\leq C,\qquad
 \max_{p,a}\|E(e_a\text{ in mode }p,\cdot)\|_{\mathrm{HS}}\leq\delta_d,
 \qquad
 \|\operatorname{Flat}_{S|S^c}E\|_{\mathrm{op}}\leq\delta_d
 \quad(\varnothing\ne S\ne[r]).
 \label{fp:core-invariants}
\end{equation}
Except when $r=q=1$, one expansion preserves these conclusions, allowing constants to increase by a fixed factor.

\emph{Proof.}
The Hilbert norm bound follows from boundedness of the expansion operator. Consider a nontrivial partition of the new kernel.
If the new block lies entirely on one side, this composes bounded operators on the two sides of a nontrivial partition of the old kernel.
If the new block crosses the partition, temporarily retain an orthogonal parent index $a$ on both sides. The central operator is block diagonal in $a$;
each block is a matricization of the old slice $E(e_a,\cdot)$, whose operator norm is at most its Hilbert norm,
$\delta_d$. Apply the row maps on both sides to conclude. This also holds when the old kernel has one mode.

Except when $r=q=1$, the new tensor has at least two indices. Its coordinate slices satisfy
$\|E_{\rm new}(e_a\text{ in }p,\cdot)\|_{\rm HS}
\le\|\operatorname{Flat}_{\{p\}|\{p\}^c}E_{\rm new}\|_{\rm op}\le C\delta_d$,
which follows from the matricization bound. When $r=q=1$, there is no nontrivial matricization, so coordinate estimates for consecutive Gaussian products are needed.
\hfill$\square$

\paragraph{Gaussian matrix products along linear paths.}
Let $A_\ell$ be independent Gaussian matrices normalized by the square root of the input width.
Any fixed consecutive product $P=A_b\cdots A_a$ satisfies
\begin{equation}
 \|P\|_{\mathrm{op}}\leq C,\qquad
 \max_{ij}|P_{ij}|\leq C_M\sqrt{\log d/d}
 \label{fp:core-product-event}
\end{equation}
with failure probability $O(d^{-M})$. The operator norm bound follows by multiplying the layer bounds. Conditional on
$Q=A_{b-1}\cdots A_a$, the variable $(A_bQ)_{ij}$ is Gaussian with variance
$\|Qe_j\|^2/n_{b-1}\leq C/d$; the tail bound and a finite union bound give the second estimate.
When a readout vector is multiplied by a consecutive Gaussian product, conditioning on the last unexposed Gaussian factor gives the same bound on the maximum coordinate.
This handles the $r=q=1$ chain before the first branch. After branching, the number of modes does not decrease, so subsequent steps use the preceding lemma.
Stopping expansion or applying an identity map requires no probability estimate. All kernels of the finite original tree satisfy
\eqref{fp:core-invariants}. The unary chains here consist only of original network matrices.

\paragraph{External maps and reconstruction.}
Bounded single-factor maps on each index preserve nontrivial matricization bounds, but need not preserve coordinate-slice bounds.
After each contraction, the coefficient tensor must be reconstructed along the original computation tree: stop expansion at the corresponding parent nodes and use the resulting cross-network contraction matrices as external maps on the leaves.
Stopping depths may differ; the internal root-to-leaf paths still contain only original network matrices. The coordinate-slice bounds below are used only for these reconstructed coefficient tensors.

\refstepcounter{lemma}
\paragraph{Lemma~\thelemma\ (Exact summation along linear paths).}
\label{fp:core-linear}
Let the bottom-layer row vectors be $b_j$, let the parent-layer matrix be $U=(u_{aj})$, and set
$z_a=\sum_ju_{aj}b_j$. If a parent node has $t$ linear child slots and nonlinear child slots of degrees
$q_1,\ldots,q_h\geq2$, its unsymmetrized tensor is exactly
\begin{equation}
 z_a^{\otimes t}\otimes
 \bigotimes_{\ell=1}^h\left(\sum_j u_{aj}b_j^{\otimes q_\ell}\right).
 \label{fp:core-linear-identity}
\end{equation}
\emph{Proof.} Sum independently over each ordered child slot. By multilinearity, linear slots give $z_a$, while nonlinear slots retain their original expressions.
Symmetrization averages over slot permutations at a fixed degree and can be applied after summation. Thus a purely unary chain is a normalized matrix product, with no extra
$\sqrt{\text{width}}$ factor. Nonlinear factors still share $a$ and cannot be treated as independent arrays.
Coordinate masks generally do not commute with summation. One must first prove that removing masks from the expansion operator over all factors incurs an
$o(1)$ error, then restore the original tensor and apply this identity.\hfill$\square$

\subsection{Rotations of same-layer indices and contractions over two indices}
\label{fp:core-fresh-section}

\paragraph{Probabilistic tool.}
If deterministic matrices $A_j:\mathbb R^u\to\mathbb R^v$ satisfy
$\max\{\|\sum_jA_j^*A_j\|,\|\sum_jA_jA_j^*\|\}\leq s^2$, then independent
standard Gaussian coefficients satisfy
\[
 \mathbb P\!\left(\left\|\sum_jg_jA_j\right\|>t\right)
 \leq2(u+v)\exp[-t^2/(2s^2)].
\]
Use the self-adjoint dilation
$H_j=\left(\begin{smallmatrix}0&A_j^*\\ A_j&0\end{smallmatrix}\right)$,
for which $\|\sum_jH_j^2\|\leq s^2$; the matrix Gaussian Laplace bound
$\mathbb E\operatorname{tr}\exp(\lambda_{\rm c}\sum_jg_jH_j)
\leq\operatorname{tr}\exp(\lambda_{\rm c}^2\sum_jH_j^2/2)$,
with $\lambda_{\rm c}=t/s^2$, gives the upper tail. Repeating with the negative sign gives the two-sided bound.
Below we use only its $O(s\sqrt{\log d})$ consequence when the dimension is a fixed power of $d$.

\refstepcounter{lemma}
\paragraph{Lemma~\thelemma\ (Coordinate slices under same-layer rotations).}
\label{fp:core-fresh}
Expand all unstopped modes simultaneously at the same original network layer. Choose a column-index mode $p$ of the current Gaussian layer matrix,
which rotates when that matrix is multiplied on the right by an orthogonal matrix; exclude the retained parent index $e_a$ and its deterministic copies.
Choose another mode $q$, with total number of modes $N\ge3$. Let $E$ be the coefficient tensor before external maps are applied.
With arbitrarily high polynomial probability,
\begin{equation}
 \max_j\sup_{\|v\|\leq1}
 \|E(e_j\text{ in }p,v\text{ in }q,\cdot)\|_{\mathrm{HS}}
 \leq\rho_d.
 \label{fp:core-fresh-slice}
\end{equation}
At this stage, do not expand lower layers, and keep stopped indices in higher layers fixed.

\emph{Proof.}
Condition on a kernel satisfying \eqref{fp:core-invariants}, and set
$A_jv=E(e_j,v,\cdot)$. Direct summation over matrix entries gives
\[
 \left\|\sum_jA_j^*A_j\right\|
 =\|\operatorname{Flat}_{\{q\}|\{q\}^c}E\|^2\leq\delta_d^2,
 \qquad
 \left\|\sum_jA_jA_j^*\right\|
 =\|\operatorname{Flat}_{\{p,q\}|\{p,q\}^c}E\|^2\leq\delta_d^2.
\]
The second partition is nontrivial precisely because $N\geq3$. The series bound above and
the Gaussian tail bound $\|g\|\geq c\sqrt d$ bound the contraction norm for a uniform sphere vector $u=g/\|g\|$
by $C_M\delta_d\sqrt{\log d/d}$.
Take a union bound over the columns of an independent Haar matrix; independence between columns is unnecessary.
Right multiplication of the Gaussian layer matrix by this orthogonal matrix preserves its distribution and rotates all its column-index modes simultaneously.
Rotations of unfixed modes are isometries; if $q$ is also a column index of this layer, the supremum over $v$ is unchanged.
Thus the Haar estimate gives the original coordinate estimate. There are only a fixed finite number of kernels and modes.
\hfill$\square$

\refstepcounter{lemma}
\paragraph{Corollary~\thelemma\ (Norm of repeated-coordinate terms).}
\label{fp:core-collision}
If at least two modes correspond to column indices of the same Gaussian layer matrix and there are at least three modes in total, then the part where at least two such indices take the same value satisfies
\begin{equation}
 \|E_{\mathrm{coll}}\|_{\mathrm{HS}}^2
 \leq\sum_{p<q}\sum_j\|E(e_j\text{ in }p,e_j\text{ in }q,\cdot)\|^2
 \leq C d\rho_d^2=o(1).
 \label{fp:core-collision-bound}
\end{equation}
Estimate this part when the modes are introduced at that layer, then propagate it through bounded expansion operators; the error remains $o(1)$.
Delete only terms in which indices of distinct original nodes have the same value, retaining deterministic copies of the same parent index.

\refstepcounter{lemma}
\paragraph{Corollary~\thelemma\ (Moment and coordinate bounds for raw coefficients).}
\label{fp:core-raw-collision-expectation}
Let $C_R$ be a raw order-$R$ coefficient of a Gaussian polynomial network of fixed depth and degree with zero constant terms,
allowing ordered factors, symmetrization, and sums of a fixed finite number of expansion terms. For every fixed $0<p<\infty$,
$\sup_d\mathbb E\|C_R\|_{\rm HS}^p<\infty$.
If all indices arise as column indices of the same original Gaussian layer (excluding deterministic copies of parent indices), of width $k\asymp d$, then for any fixed $M$,
with failure probability $O(d^{-M})$ the following hold simultaneously:
\begin{equation}\label{fp:core-coordinate-bounds}
 \max_{i_1,\ldots,i_R\ \text{distinct}}|(C_R)_{i_1\cdots i_R}|
 \le L_d k^{-R/2},\qquad
 \|\operatorname{Coll}_R C_R\|_{\rm HS}^2\le L_d/d\quad(R\ge3),
\end{equation}
where $L_d$ is a fixed power of the logarithm; for $R\ge3$, also $\mathbb E\|\operatorname{Coll}_R C_R\|_{\rm HS}^2=o(1)$.
The moment bound follows by multiplying the norms of the original expansion operators, using fixed-order moments of normalized Gaussian matrix operator norms and H\"older's inequality; independence of the factors is unnecessary.
Column-permutation invariance makes the second moments of distinct-index coordinates equal, each at most
$C/(k)_R=O(k^{-R})$. Each coordinate is a Gaussian polynomial of fixed degree $D_0$,
with $\|Z\|_u\le(u-1)^{D_0/2}\|Z\|_2$; take $u=C_M\log d$, apply Markov's inequality at $e\|Z\|_u$,
and take a union bound over at most $k^R$ coordinates to obtain the first bound.
Apply Lemma~\ref{fp:core-fresh} and \eqref{fp:core-collision-bound} to each ordered expansion term to obtain the second bound;
a fixed finite number of permutations and sums does not change the order. The moment bound and Cauchy--Schwarz give
$\mathbb E\|\operatorname{Coll}_R C_R\|^2\le L_d/d+
(\mathbb E\|C_R\|^4)^{1/2}O(d^{-M/2})=o(1)$.
These statements control only raw coefficients before cross-network external maps are applied.

\refstepcounter{lemma}
\paragraph{Lemma~\thelemma\ (Contraction over two parent indices and second-order off-diagonal terms).}
\label{fp:core-two-parent}
At an unmasked original cut, suppose that contracting the selected linear slots of distinct parent modes $p,q$ with teacher rows
gives $\eta_i,\zeta_i$, with bounded $\ell^2$ norms. Mode $p$ is introduced at this layer
and has a nonempty residual after its selected slot is deleted; all complete residual families are bounded operators.
If the total number of modes is $R\ge3$, the complete remaining source tensor $S_i$ satisfies $\|S_i\|_{\rm HS}\le C\rho_d$.
Indeed, let $B_ie_b=\zeta_{ib}f_b$, where $f_b$ is the complete remaining
feature of the second source index after its selected slot is deleted; set $f_b=1$ when no slots remain. Pointwise,
$\|B_i\|_{\rm HS}^2=\sum_b|\zeta_{ib}|^2\|f_b\|^2\le C$.
View the original slice $E(e_a,\cdot)$ as an operator $T_a$ from index $q$ to the remaining indices.
Equation~\eqref{fp:core-fresh-slice} gives $\|T_a\|_{\rm op}\le\rho_d$, hence
$\|T_aB_i^*\|_{\rm HS}\le C\rho_d$.
Retain index $a$, multiply by $\eta_{ia}$ and sum the squares, then apply the complete remaining features to obtain the bound.
The estimate holds pointwise and does not require $B_i$ to be independent of the network.

If $R=2$, the two indices belong to the same original network layer, and both modes have nonempty residuals after their selected slots are deleted, \eqref{fp:core-coordinate-bounds} gives
$\max_{a\ne b}|D_{ab}|\le L_d/d$; retaining both labels and then applying the complete feature maps gives the off-diagonal bound
\[
 \|S_i^{\rm off}\|\le C\Big(\sum_{a\ne b}|D_{ab}\eta_{ia}\zeta_{ib}|^2\Big)^{1/2}
 \le CL_d/d.
\]
Pairing either bound with the teacher slice sum $C\sqrt d\|C\|$ gives $o(1)\|C\|$.
The second-order conclusion excludes diagonal terms and coordinate--vector slices; indices from different layers require separate treatment with the root index retained.
Deterministic copies of the same parent index do not count as two parent nodes and cannot use this estimate.

\subsection{Local contraction estimates retaining residual factors}\label{fp:core-residual-section}

\begin{lemma}[Residual families and local contractions]\label{fp:core-residual-flat}
\label{fp:core-local-contraction}\label{fp:core-split-hit}
\label{fp:core-hybrid-Rge2}\label{fp:core-hybrid-R1}
Let $F_a=\sum_bU_{ab}Q_b$, where $U$ is an original Gaussian layer or a product of consecutive layers, and $Q_b$ is an original computation tree of positive degree;
stopped leaves may carry bounded adaptive single-slot maps. The complete family $e_a\mapsto F_a$ is bounded,
and when $F_a$ has at least two output factors, every nontrivial output matricization is at most $C\delta_d$.
Product families with the same index are also bounded.

Fix a teacher node with $m\ge2$ factors. Suppose it meets an old contracted factor $\ell_a$ of a source parent index $a$,
that this source column has at least one complete factor besides $\ell_a$, and that $G_{ia}=\langle a_i,\ell_a\rangle$ satisfies
$\|G\|_{\max}\le\kappa_d$ and $\sup_i\|G_{i,:}\|_2\le C$.
Retain all other complete residuals, and assume that maps for other source parent labels do not depend on $a$.
Let $C,D$ be the original teacher and source coefficients, respectively. Then the complete pairing satisfies
\begin{equation}\label{fp:local-unified}
 |\mathrm{pairing}|\le
 \begin{cases}
 C\kappa_d\|C\|\|D\|,&\text{this leaf also meets another source parent node},\\
 C\sqrt d\,\alpha(D)(\delta_d+\kappa_d)\|C\|,&\text{this leaf lies entirely at parent node }a,
 \end{cases}
\end{equation}
In the second line, take $\alpha(D)=\|\operatorname{Flat}_{a|\mathrm{rest}}D\|$ for a multimode kernel;
for a one-mode kernel $D=d$, take $\alpha(D)=\|d\|_\infty$, but require at least two complete factors besides $\ell_a$.
Thus both types of pairing are $o(1)$ when $\alpha(D),\delta_d,\kappa_d\le L_d/\sqrt d$.
\end{lemma}
\begin{proof}
The residual family is the composition of the original expansion operator with $U^*$. For fixed $a$, reconstruct the kernel using $U_{a,:}$ as readout;
\eqref{fp:core-product-event} and Lemma~\ref{fp:core-closure} give the nontrivial matricization bound, which is preserved by external single-slot maps.
The diagonal isometric embedding ensures that product families are bounded.

If the leaf also meets another parent node, first map all $t\ge1$ slots from other parent nodes to the orthogonal teacher label $i$.
The adjoint of this map is $(A^*)^{\otimes t}\Delta_t$, so it is bounded.
After retaining $a,i$, multiplication by $G_{ia}$ has operator norm $\|G\|_{\max}$; then, within each $i$ block,
lift the remaining family of this column and complete the remaining pairing. At least one further complete factor ensures boundedness of the family;
even if this leaf contracts it completely, an $\ell^2$-bounded scalar family remains, and its label cannot be deleted without its weight.
These are bounded operators on complete tensors, with no partial trace or loss from summing over $i$.

Now suppose the leaf lies entirely at $a$. Fix $i$. Contracting the other factors gives families $H_j$: completely contracted factors give
$\ell^2$-bounded vectors; partially contracted factors have column norms at most $C\delta_d$; fully retained factors remain bounded families.
The leaf still has $m-1\ge1$ slots, so at least one factor is partially or completely contracted.

Set $B_ie_a=G_{ia}\bigotimes_jH_j e_a$. If there is a partially contracted factor, use $\|G_{i,:}\|_2$;
otherwise use $\|G\|_{\max}$ and the $\ell^2$ norm of a completely contracted vector. In either case,
$\|B_i\|_{\rm HS}\le C(\delta_d+\kappa_d)$.
In the multimode case, denote the expansion of the other modes by $T$. Then
$\|(B_i\otimes T)D\|_{\rm HS}\le C\alpha(D)(\delta_d+\kappa_d)$.

In the one-mode case, there are at least two $H_j$. If some $H_u$ is partially contracted, choose $j\ne u$ and use
$H_j^*H_j\preceq CI$ and $\operatorname{diag}(H_u^*H_u)\preceq C\delta_d^2I$.
The Schur product order bounds the operator norm of the product family by $C\delta_d$; applying it to $(d_aG_{ia})_a$ gives
$C\|d\|_\infty\delta_d$. Otherwise absorb one completely contracted vector $\eta$,
so that $\|(d_aG_{ia}\eta_a)_a\|_2\le C\|d\|_\infty\kappa_d$, while at least one remaining family is still bounded.
Finally, use $\sum_i\|C(e_i,\cdot)\|\le C\sqrt d\|C\|$ only once to obtain the second line.
\end{proof}
Write a second-order off-diagonal kernel as $D-\operatorname{diag}D$ and apply this bound to each term; neither the operator norm
nor the Hilbert norm increases upon taking the diagonal. Other masks are not covered, and a second-order diagonal need not be small.
The one-mode boundary case with only one other factor uses the retained-index subroutine in Lemma~\ref{fp:permanent-mark}.

\section{Input chaos projection and polynomial approximation for the actual activation}
\label{fp:actual-section}\label{app:closure}
Fix $N>2$, write $A_\ell=W^{(\ell)}/\sqrt{k_{\ell-1}}$ and $u=c_0=\sqrt{2/\pi}$, and set
\begin{equation}\label{fp:actual-PQ}
 P_N(z)=\sum_{j=1}^N\frac{a_jz^j}{\sqrt{j!}},\qquad
 Q_N(z)=\sum_{j=1}^N\frac{b_jz^j}{\sqrt{j!}},\qquad
 b_j=\E[(\sigma(G)^2-1)\psi_j(G)].
\end{equation}
$Q_N$ is used only in the piecewise-linear branch. The polynomial network uses the original matrices $A_1,\ldots,A_t$, readout
$k_t^{-1/2}\mathbf1$, output activation $A$, and lower-layer activations $p_1,\ldots,p_{t-1}$;
denote its formal symmetric coefficient of degree $N$, after applying $I_N$, by $R_{t,N}[A;p]$,
and by $J_{t,N}[A;p]$ when retaining only expansion terms with all indices distinct within each hidden layer.
Input coordinates are unrestricted, so $J_{1,N}=R_{1,N}$; $;P_N$ means that every lower layer uses $P_N$.
Coefficients in fixed auxiliary variables may be extracted first. Formal coefficient extraction precedes $I_N$
and excludes order-$N$ terms produced by Gaussian contractions of higher formal degrees.

\subsection{Conditional expansion and its error}
\label{fp:actual-finite-support}\label{fp:actual-positive}
All $L^p$ norms in this section are over the original Gaussian prior and the input. Lemma~\ref{an:moments} and
the matrix tail bounds in Section~\ref{an:appendix} give
\begin{equation}\label{fp:actual-local-interface}
 \|q_\ell-1\|_p=O(d^{-1/2}),\qquad
 \sup_d\E\left[\|A_\ell\|_{\rm op}^p+|h_i^{(\ell)}|^p\right]<\infty.
\end{equation}
Fixed products (including repeated indices) also have all fixed-order moments, so the probability of $q_\ell\notin[1/2,2]$
decays at any prescribed polynomial rate. Write first-layer rows as $A_{1,i:}=s_iv_i$, with $\|v_i\|=1$.
Gaussian length and inner-product tails and matrix tail bounds give an event of probability at least $1-d^{-M}$ on which
\begin{equation}\label{fp:actual-row-event}
 \max_i|s_i-1|+\max_{i\ne j}|\langle v_i,v_j\rangle|
 \le C_M\sqrt{\log d/d},\qquad \|A_1\|_{\rm op}\le C_M.
\end{equation}

\paragraph{Finite products in the first layer.}
If the Gram matrix $G$ of a fixed number $t$ of unit vectors satisfies $\|G-I\|\le\eta$,
and $\max_i|s_i-1|\le\delta$, then
\begin{equation}\label{fp:actual-ridge-projection}
 \left\|\Pi_N\prod_{i=1}^t\sigma(s_iv_i\cdot X)
 -\sum_{\substack{n_i\ge1\\\sum_i n_i=N}}
 \frac{\prod_i a_{n_i}}{\sqrt{\prod_i n_i!}}
 I_N\!\left(\Sym\bigotimes_i(s_iv_i)^{\otimes n_i}\right)
 \right\|_2\le C_{t,N}(\eta+\delta).
\end{equation}
In $\Sym^N\operatorname{span}\{v_i\}$, take
$B_n=I_N(\Sym\bigotimes_i v_i^{\otimes n_i})$, with $n_i\ge0$ and $\sum n_i=N$.
Their Gram matrix is a polynomial in $G$ and is uniformly invertible near $G=I$; the projection coordinates are
its inverse times $\E[\prod_i\sigma(s_iZ_i)B_n(Z;G)]$, where $Z\sim N(0,G)$.
Covariance derivatives of the Gaussian density are bounded by integrable polynomials times a Gaussian density;
the coefficients of $B_n$ are polynomials in $G$, and the activation is Lipschitz with at most linear growth,
so these integrals are locally Lipschitz in $(G,s)$. At $(I,\mathbf1)$,
independence gives coefficients $\prod_i a_{n_i}/\sqrt{\prod_i n_i!}$, and $a_0=0$ removes zero degrees.
Replacing $v_i$ by $s_iv_i$ gives the displayed formula, which includes all contraction terms of the nonpolynomial product.
In particular, the approximating sum is empty when $t>N$.

\paragraph{Layerwise conditional expansion.}
Let $\chi=\sigma^2-1$, and let $f=\sigma$, or $f=\chi$ in the piecewise-linear branch.
Given previous-layer features $h\in\R^k$, set $q=\|h\|^2/k$ and
$c_r^f(q)=\E f(\sqrt qG)\psi_r(G)$. The conditional Hermite expansion uses
\begin{equation}\label{fp:actual-multivariate}
 f(W_i\cdot h/\sqrt k)=\sum_{r\ge0}c_r^f(q)
 \sum_{|\alpha|=r}\sqrt{\frac{r!}{\alpha!}}
 (h/\|h\|)^\alpha\prod_j\psi_{\alpha_j}(W_{ij}).
\end{equation}
At $q=0$, use the continuous value of the actual random variable. Since $|f(x)|\le C(1+x^2)$,
$q\mapsto f(\sqrt q\,\cdot)$ is a continuous map from $[1/2,2]$ into $L^2(\gamma)$,
so its Hermite tails vanish uniformly. Outside this interval, use the full conditional second moment $C(1+q^2)$,
\eqref{fp:actual-local-interface}, and small-probability bounds of arbitrarily high order.
For a fixed number of outputs from distinct rows, conditional independence factors the conditional squared expectation of each factorwise difference into
the squared expectation of one tail times the second moments of the remaining factors. Thus products can be truncated without higher-moment convergence of single-row tails.

For fixed $r$, substitute $y=\sqrt qG$ and differentiate the density and $\psi_r(y/\sqrt q)$ to see that
$c_r^f(q)q^{-r/2}$ is Lipschitz on $[1/2,2]$, so replacing it by $c_r^f(1)$ incurs a vanishing error. Outside this interval, fixed-order conditional moments of exact finite components are controlled by the growth of $c_r^f(q)$ and
Gaussian Hermite moment bounds, while the replacement polynomials are controlled by feature moments; H\"older's inequality and small-probability bounds
give $o(1)$, without negative moments of $q$ or derivatives of $f$.

When a fixed expansion contains $T$ previous-layer output factors, there are at most $C_Tk^{T-1}$ repeated-index assignments,
and the squared normalization factor is $k^{-T}$. After merging orderings of the same weight Hermite monomial
(with multiplicities depending only on $T$), orthogonality and fixed-product moments show that deleting repeated assignments has squared error $O_T(k^{-1})$.
Each remaining weight variable appears only once, so its Hermite monomial becomes an ordinary product of weights.
Hidden-layer zeroth-order coefficients are zero (in the odd branch) or $bu(\sqrt q-1)=O_{L^p}(d^{-1/2})$
(in the piecewise-linear branch), and can also be deleted from fixed products; the output-layer zeroth-order term is retained separately below.

Local errors still vanish after summing with the readout: at each stage, merge equal upper weight monomials and write
\begin{equation}\label{fp:actual-phase-bound}
 \Delta=\sum_g\beta_g\Psi_g(W_{>\ell})E_g(W_{\le\ell},X),
 \quad \sum_g\beta_g^2\le C_{\rm fixed},
 \quad \E\|\Delta\|_{L^2_X}^2
 \le C_{\rm fixed}\sup_g\E\|E_g\|_{L^2_X}^2.
\end{equation}
Here distinct $\Psi_g$ are orthogonal, each variable has degree one, and these monomials are independent of the lower layers.
The sum of squared coefficients is bounded: squaring the readout $k_t^{-1/2}$ and the input-width factor with exponent $-1/2$ for each edge
cancels the numbers of outputs and free indices; merging multiplicities depend only on the fixed expansion structure.
Choose finite truncations successively from the output layer downward so that the total $L^2$ tail is arbitrarily small, then use
\eqref{fp:actual-ridge-projection} for the distinct first-layer activation factors.
Equation~\eqref{fp:actual-row-event} and the preceding squared-coefficient bound make the first-layer error vanish;
on the bad event, moments of fixed products and fixed-degree polynomials grow at most as a fixed polynomial in the dimension and are absorbed by the arbitrarily high-order failure probability.
If a node has degree exceeding $N$, its distinct subtrees give more than $N$ distinct first-layer factors, and the projection approximation is empty.
Thus the expression stabilizes once every layer is truncated at least at $N$; first let $d\to\infty$, then let the truncation error vanish, obtaining
\begin{equation}\label{fp:actual-positive-local}
 \Pi_NF_{t,+}^f=J_{t,N}[H_f;P_N]+o_{L^2}(1),\qquad
 H_\sigma=P_N,\quad H_\chi=Q_N.
\end{equation}
Here $F_t^f=k_t^{-1/2}\sum_i f((A_th^{(t-1)})_i)$, and $F_{t,+}^f$ is this quantity minus its conditional mean given the previous layer.
For $t=1$, expand each row directly: $\E f(sG)\psi_N(G)$ is Lipschitz in $s\in[1/2,2]$,
so it can be replaced by $s^N\E f(G)\psi_N(G)$. The first-layer vector map $e_i\mapsto v_i^{\otimes N}$ is bounded,
since the absolute row sums of its Gram matrix are at most $\sum_j|\langle v_i,v_j\rangle|^2\le\|V\|_{\rm op}^2$.
The readout has norm one, so the rowwise coefficient errors $O(\max_i|s_i-1|)$ give total error $o(1)$; the bad event is handled as above.

\subsection{Quadratic diagonals and index restoration}
\label{fp:actual-quadratic}\label{fp:actual-telescope}
For the complete upper polynomial, repeated coordinates of order $r\ge3$ are controlled by Corollary~\ref{fp:core-raw-collision-expectation},
but this cannot be applied directly to the index-restricted $J$. Quadratic terms additionally satisfy the following:
for input width $k$, depth $t$, and layer activations $p_j(z)=\sum_{r\ge1}g_{j,r}z^r$,
\begin{equation}\label{fp:actual-diag}
 \E\|\operatorname{Diag}D_2-\lambda I_k/\sqrt k\|_{\rm HS}^2=O(k^{-1}),
 \qquad \lambda=g_{t,2}\prod_{j<t}g_{j,1}^2\sqrt{k_t/k}.
\end{equation}
Each quadratic term of a positive-degree network has exactly one quadratic node; when this node lies in layer $j$, removing fixed scalar factors leaves
$D_2^{(j)}=B_j^T\operatorname{diag}(v_j)B_j$, where
$B_j=A_j\cdots A_1=C_jA_1$ and $v_j=A_{j+1}^T\cdots A_t^T\mathbf1/\sqrt{k_t}$.
Set $H_j=C_j^T\operatorname{diag}(v_j)C_j$. Conditional on $H_j$, the columns of $A_1$ are independent, so
\[
 \E((D_2^{(j)})_{ii}\mid H_j)=\tr H_j/k,\qquad
 \E\sum_i|(D_2^{(j)})_{ii}-\tr H_j/k|^2=2\E\|H_j\|_{\rm HS}^2/k=O(k^{-1}).
\]
The last step uses $\|H_j\|_{\rm HS}\le\|C_j\|_{\rm op}^2\|v_j\|$.
For $j<t$, $v_j$ is independent of $C_j$, with mean zero and covariance $I/k_j$, so
$\E|\tr H_j|^2=k_j^{-1}\E\sum_a\|C_{j,a:}\|^4=O(1)$.
For $j=t$, $\tr H_t=\|C_t\|_F^2/\sqrt{k_t}$ has mean $\sqrt{k_t}$ and variance $O(k_t^{-1})$:
for an independent next layer $A$,
\[
 \E(\|AC\|_F^2\mid C)=\frac{k_{\rm new}}{k_{\rm old}}\|C\|_F^2,
 \qquad \operatorname{Var}(\|AC\|_F^2\mid C)
 =\frac{2k_{\rm new}}{k_{\rm old}^2}\|CC^T\|_{\rm HS}^2=O_{L^1}(1),
\]
and induction along a fixed-length product using the law of total variance gives $\operatorname{Var}\|C_t\|_F^2=O(1)$.
Restoring the scalar factors proves \eqref{fp:actual-diag}, also after coefficient extraction in auxiliary variables.

\begin{lemma}[Removing the distinct-index restriction]\label{fp:actual-restoration}
Let $A,p_1,\ldots,p_{t-1}$ have zero constant terms, fixed degrees, and uniformly bounded coefficients, with $t\ge2$.
Writing $[p]_r$ for the coefficient of $z^r$, we have
\begin{equation}\label{fp:actual-restoration-exact}
 R_{t,N}[A;p]-J_{t,N}[A;p]
 =\sum_{s=1}^{t-1}\lambda_sJ_{s,N}[p_s^2;p_{<s}]+o_{L^2}(1),
 \qquad
 \lambda_s=[A]_2\prod_{j=s+1}^{t-1}[p_j]_1^2\sqrt{k_t/k_s}.
\end{equation}
\end{lemma}
\begin{proof}
Start from the complete expansion and impose the distinct-index restrictions successively at layers $s=1,\ldots,t-1$.
When processing layer $s$, the upper part is still a complete polynomial.
Equation~\eqref{fp:actual-diag} writes its quadratic diagonal as $\lambda_sI/\sqrt{k_s}$ plus
a small Hilbert--Schmidt error. Denote the polynomial output of layer $s$ by $h_s$; the main term identifies two parent labels, giving
$\lambda_sk_s^{-1/2}\sum_i h_{s,i}^2$. The distinct-index restrictions already imposed below make this exactly
$\lambda_sJ_{s,N}[p_s^2;p_{<s}]$, with the two equal factors still occupying one original slot each.
Degree one has no repeated coordinates, and higher-degree repetitions are controlled by the raw collision estimate.
All these errors propagate through the original expansions of the common lower layers, coordinate projections, and symmetrization, whose operator norms are controlled by
fixed powers of finitely many $\|A_j\|_{\rm op}$.
Moment equivalence for fixed-degree Gaussian polynomials upgrades the coefficient errors to small $L^4$ quantities;
H\"older's inequality and \eqref{fp:actual-isometry} give a function error $o_{L^2}(1)$ at each step.
Summing over finitely many interfaces proves the claim, without requiring independence between upper- and lower-layer errors.
\end{proof}
Every $J$ on the right has height strictly less than $t$. Strong induction from $J_{1,N}=R_{1,N}$
therefore writes $J_{t,N}[A;p]$ as $R_{t,N}[A;p]$ plus finitely many shallower bare networks.
Every shallow root still has zero constant term and uniformly bounded coefficients; terms of degree exceeding $N$ do not contribute to the formal degree-$N$ coefficient and can be deleted.

\subsection{Conditional-mean recursion and shallow corrections}
\label{fp:actual-mean}\label{fp:actual-masked-formula}
In the piecewise-linear branch, for $f=\sigma,\chi$ we have the exact formula
\begin{equation}\label{fp:actual-root-mean}
 \E f(\sqrt qG)=\lambda_f(q-1)+c_f(\sqrt q-1)^2,
 \quad (\lambda_\sigma,c_\sigma)=(bu/2,-bu/2),\quad
 (\lambda_\chi,c_\chi)=(1,b^2u^2).
\end{equation}
By $|\sqrt q-1|\le|q-1|$ and Lemma~\ref{an:moments}, the quadratic remainder, after multiplication by the readout factor $\sqrt{k_t}$,
is still $O_{L^2}(d^{-1/2})$. For $t\ge2$, the linear term equals
$\lambda_f\sqrt{k_t/k_{t-1}}F_{t-1}^\chi$; for $t=1$, it is
$\lambda_f\sqrt{k_1}(q_0-1)$, which the $N>2$ projection annihilates exactly.
Combining with \eqref{fp:actual-positive-local} gives
\begin{equation}\label{fp:actual-cancel-recursion}
 \Pi_NF_t^f=J_{t,N}[H_f;P_N]
 +\lambda_f\sqrt{k_t/k_{t-1}}\Pi_NF_{t-1}^\chi+o_{L^2}(1),
 \quad t\ge2,
\end{equation}
with initial value $\Pi_NF_1^f=R_{1,N}[H_f]+o_{L^2}(1)$.
Substitute the shallow representation from Lemma~\ref{fp:actual-restoration}, induct on height, and combine root polynomials of the same height to obtain
\begin{equation}\label{fp:actual-U}
 \mathcal Q_N=R_{L,N}[P_N;P_N]
 +\sum_{s=1}^{L-1}R_{s,N}[A_{s,d};P_N],
\end{equation}
\begin{equation}\label{fp:actual-final}
 \boxed{\E_W\|\Pi_NF_L-\mathcal Q_N\|_{L^2_X}^2\longrightarrow0,\qquad N>2.}
\end{equation}
Here $A_{s,d}$ are deterministic polynomials with zero constant term and degree at most $N$. Their coefficients depend on fixed activation coefficients and
finitely many width ratios, and are therefore uniformly bounded. In the odd branch, both the conditional means and quadratic coefficients vanish,
so all $A_{s,d}=0$ may be taken, without using $\chi$; for $L=1$, the sum is empty.
The comparisons below hold uniformly over arbitrary shallow roots of fixed degree with bounded coefficients, so the specific coefficients of $A_{s,d}$ need not be computed; the single-network $L^2$ error transfers by
Cauchy--Schwarz to any coupling with the prescribed prior marginals.

\section{Layerwise comparison of computation-tree expansion terms}\label{fp:history}\label{app:cuts}
Fix $L$ and $N=m^L$; take odd $m\ge5$ for odd activations and even $m\ge6$ for piecewise-linear activations.
Let $\delta_d,\kappa_d\le K_d/\sqrt d$, where $K_d$ is a fixed power of the logarithm that may increase through finitely many operations.
At each node, select a positive-degree monomial and order its children to form an ordered expansion term; fix a pairing of its $N$ input factors
with the slots of the homogeneous teacher term, and finally sum over the finitely many choices and at most $N!$ permutations.
First fix the original Gaussian events for the teacher and source separately; the cross-network maps may then depend arbitrarily on both networks.

\begin{lemma}[Ordered-tree elimination]\label{fp:lowest-cut}\label{fp:permanent-mark}\label{fp:core-spectator}
For every ordered expansion term of height $t<L$, with an arbitrary zero-constant polynomial $A$ at the root, $P$ at lower levels, and degree $N$,
every fixed slot pairing with a homogeneous teacher term of height $L$ is $o(1)$.
For ordinary terms of height $L$ in the $P$ expansion, the same holds except when every node has degree $m$.
\end{lemma}
\begin{proof}
Use strong induction on the source height $t$; for fixed $t$, process unfinished nodes in order of teacher height.
Completed contractions of height $h-1$ serve as factors at step $h$, and new outputs are used only at the next step.
Place all cross-network contractions on the source side, retaining the original homogeneous coefficient $C$ on the teacher side.
After fixing the current teacher row $i$, the remaining complete teacher factors have norm at most $K_d\|C(e_i,\cdot)\|$, so
\begin{equation}\label{fp:teacher-slices}
 \sum_i\|C(e_i,\cdot)\|\le\sqrt{k}\|C\|\le K_d\sqrt d.
\end{equation}
Reconstruct the coefficient kernel at the stopping nodes of the original source tree, then apply bounded complete feature maps; retain separate labels for outputs at different heights. Copies of the same parent index are also retained, and are treated neither as independent Gaussian column indices nor as removable collisions.
First sum purely unary chains exactly using Lemma~\ref{fp:core-linear}; all contractions act on the recovered unmasked terms.

\paragraph{Complete contractions and the retained-index subroutine.}
A valid complete contraction uses $m$ complete linear children of the same source node, with matrix
$M_{ia}=\prod_{j=1}^m\rho^{(j)}_{ia}$ satisfying
\begin{equation}\label{fp:metric-schur}
 \sup_i\sum_a|M_{ia}|+\sup_a\sum_i|M_{ia}|\le K_d,
 \qquad \|AM\|_{\max}\le K_d\kappa_d,
\end{equation}
where each $\rho^{(j)}$ has bounded row and column $\ell^2$ norms, and $A$ is the next original teacher Gaussian matrix.
The first bound follows from H\"older; the second uses only $\|A\|_{\max}\le\kappa_d$ and does not require independence of $M$ and $A$.
Complete children alone do not ensure an increase in height, since the same original node may produce several old outputs.
Before reusing an old output, first exclude terms whose originating node still has other factors; after this filtering,
the parent contraction is strictly above its completed children. Thus a teacher subtree of height $h$ occupies at least $h$ original nodes along every source path.

Write the special two-mode state as
\begin{equation}\label{fp:marked-state}
 \sum_{a,b}E_{ab}\ell_a\otimes Q_b,
 \qquad \|E\|_{\rm HS}\le K_d,\quad\|E\|_{\rm op}\le\delta_d.
\end{equation}
Here $\ell$ is a complete current factor, and $Q$ is a complete subtree of original height at most $L-1$;
$E$ is either an original second-order stopping kernel or a kernel propagated along an original unary chain, and hence satisfies Lemma~\ref{fp:core-closure}.
This expression includes all uncontracted factors and the output for the next step.
Retain $a$, defer the entire path from $\ell$ to the teacher root, and process teacher subtrees off this path in order of height.
At each step, check old-output reuse before advancing the selected path; old $G$ factors come only from already certified complete steps at lower teacher heights.
Selecting one descendant of $Q$ then gives at least two modes, and selecting two distinct descendants gives at least three.
Thus, when an old contracted factor still has a residual, the mixed-parent or multimode bound in \eqref{fp:local-unified} applies;
when two distinct source parents meet, the $R\ge3$ bound of Lemma~\ref{fp:core-two-parent} applies.
The complete pairing of a mixed-parent old output is directly $O(K_d\kappa_d)=o(1)$;
for a same-parent old output and a two-parent slice, the complete residual tensor after fixing the teacher row is at most
$K_d\delta_d(\delta_d+\kappa_d)$ and $K_d/d$, respectively, both giving $o(1)$ by \eqref{fp:teacher-slices}.
For the remaining terms, advance the selected path to the lowest node with more than one factor; if the factors come from the same nonlinear child,
one should have stopped earlier, so each step can only use all $m$ linear children of that node.
Each contraction increases the number of completed teacher nodes; between contractions, the linear-path cursor only moves upward along the finite original tree.
The filtering of old outputs above ensures that every parent contraction is strictly above its children, so the height count applies to $Q$.
This subroutine invokes neither the conclusion of this lemma nor the two-mode subroutine recursively.

The other $m-1$ teacher children at the root all have height $L-1$. If $Q$ is shallower, elimination cannot finish;
if its height is exactly $L-1$, these $m-1$ contractions can only finish at $Q$'s original root $b$.
Write their matrices as $M^{(j)}$ and $G^{(j)}=AM^{(j)}$; each $G^{(j)}$ has bounded column $\ell^2$ norms
and maximum entry at most $K_d\kappa_d$. Retain the entire deferred path and all residuals; after removing these $m-1$ known factors, we have
\[
 R_b=\sum_aE_{ab}\ell_a\otimes Q_b^{\rm res},\qquad
 \sum_b\|R_b\|\le K_d\sqrt d\|E\|_{\rm HS}.
\]
For a fixed final teacher row, its complete path has bounded norm. Combining the $O(d^{-1/2})$ maximum of the normalized readout
with the bounds for two $G$ factors in column $\ell^2$ norm and the entrywise bounds for the remaining $m-3$ factors, the entire pairing is at most
\begin{equation}\label{fp:marked-terminal}
 K_d\|E\|_{\rm HS}\kappa_d^{m-3}=o(1).
\end{equation}
All $m-1$ factors are used together here; a small scalar kernel is not multiplied separately by an arbitrary residual tensor.

\paragraph{Elimination order and all boundary cases.}
First exclude terms in which an old contracted factor still carries other factors when reused.
If the current teacher node touches another source parent, use the first line of \eqref{fp:local-unified};
otherwise, if the original kernel has multiple modes, or has one mode and at least two other factors, use the second line.
A one-mode kernel with no other factors can continue. If its only other factor is also an old contracted factor,
the two current positions in this column cannot accommodate $m\ge5$ slots, so this is the first case.
If it is an original complete child, positive degrees and retention of other output labels force only an original unary chain above it;
thus the coefficient $d$ satisfies $\|d\|_2\le K_d$ and $\|d\|_\infty\le\kappa_d$.
Expanding the first edge of that child gives \eqref{fp:marked-state}, with
$E_{ab}=d_aU_{ab}$ and $Q$ of height at most $t-1\le L-1$, which is excluded by the retained-index subroutine.
All contracted factors reused thereafter contain every factor of their originating node.

Next advance upward along the $m$ source paths selected by the current teacher node; sum unary nodes exactly,
and stop at the lowest node with more than one factor, deferring higher paths and advancing lower paths to the same height or to a lower branch.
The current-layer column indices of the selected lowest parent rotate with the right rotation at that layer, and residuals remain after the selected slots are removed.
If the selected factors involve distinct parents and the original kernel has at least three modes, use the $R\ge3$ case of the two-parent lemma.
In the two-mode case, if one parent has a single factor and the other has a residual, use the retained-index subroutine.
If both parents have residuals, both belong to the current original Gaussian layer: an old contracted factor from a higher layer no longer carries other factors,
while a deferred higher output retains a separate label, which would give at least three modes.
Thus second-order off-diagonal terms satisfy the two-parent lemma's condition of nonempty residuals on both sides.

By \eqref{fp:actual-diag}, the second-order diagonal reduces to
$\lambda_sI/\sqrt{k_s}$ plus an $o_{L^2}(1)$ Hilbert--Schmidt error;
the error passes through the complete bounded feature maps. The main term identifies the two original parent labels and multiplies the root monomials,
retaining all ordered children and the original teacher-slot assignment, to give a prefix term of height $s<t$ with readout $1/\sqrt{k_s}$.
The root degree may have either parity, so the strengthened induction on source height applies; expanding existing contractions back into their original equalities leaves no incomplete subtrees.
For $s=0$, the input degree is only two, giving no contribution to $N>2$.
This diagonal formula is not used for two modes from different layers: if the higher old factor participates, use the retained-index subroutine;
otherwise, the current step involves only one parent. If both modes are higher old outputs, each contains at most
$m^{L-1}$ original input factors, giving fewer than $N$ in total; if one has already reached height $L$, it has exhausted all $N$ factors.
Deterministic copies are counted only by their original input factors and do not increase the number of original parents.

The remaining $m$ paths must meet at the same lowest source parent and come from
$m$ distinct complete linear children; otherwise one should have stopped at a lower nonlinear child.
Thus every step is a valid complete contraction; edges of the finite original tree are advanced only finitely many times, so the process terminates.
The height count excludes $t<L$. For $t=L$, every path must attain equality, so there are no additional unary nodes and every node has degree exactly $m$;
this is a contradiction except for the all-$m$ terms separated in advance. Second-order fusion strictly lowers the source height, closing the induction from $t=1$.
The coefficients, weights, and normalizations of the all-$m$ terms retain their original values, and the conclusion holds for both activation classes.
\end{proof}


\subsection{From deterministic events to arbitrary prior couplings}\label{fp:coupling-uniform}
\newthought*{Control marginal errors before taking the supremum over couplings.}
For fixed degree and depth, the numbers of ordered expansion terms, factor pairings, stopping positions, fusion heights, and contraction steps are independent of $d$. First impose all original Gaussian events on the two marginals, then allow the cross-network contraction matrices to depend on both networks. On the joint good event, every term is deterministically $o(1)$; on the bad event, combine bounded marginal higher moments of fixed-degree coefficients, H\"older's inequality, and failure probabilities of arbitrarily high polynomial order. The estimates are therefore uniform over all couplings of the prior marginals.

Let $T_W$ be the balanced all-$m$ homogeneous input-chaos term corresponding to the activation, including its full scalar coefficient,
and let $\mathcal Q_N(V)$ be the polynomial approximation in \eqref{fp:actual-U}.
The termwise estimates above give $\mathbb E_\pi|\langle T_W,\mathcal Q_N(V)-T_V\rangle|=o(1)$ uniformly over couplings.
Since $T_W$ lies in the $N$th input chaos, $\langle T_W,F_V\rangle=\langle T_W,\Pi_NF_V\rangle$.
Cauchy--Schwarz bounds the remaining pairing expectation by
$\|T_W\|_{L^2(W,X)}\|\Pi_NF_V-\mathcal Q_N(V)\|_{L^2(V,X)}=o(1)$;
by the bounded marginal second moments and \eqref{fp:actual-final}, this bound is also uniform over couplings. Hence
\begin{equation}\label{fp:global-function-pairing}
 \boxed{\displaystyle
 \sup_{\pi\in\Gamma(\mathsf P_W,\mathsf P_V)}
 \mathbb E_\pi|\langle T_W,F_V-T_V\rangle|\longrightarrow0.}
\end{equation}

\section{Global matching of homogeneous comparison networks}\label{hm:section}\label{app:matching}
Fix $L$, an integer $m\ge5$, $k_0=d$, and $k_s\asymp d$. The homogeneous network in this section is
$h^{(s)}=c_m(W^{(s)}h^{(s-1)}/\sqrt{k_{s-1}})^{\odot m}$,
$c_m=(m!)^{-1/2}$, $h^{(0)}=x$, with output
$\mathcal H_W=k_L^{-1/2}\sum_i h_i^{(L)}$ and parameter count $p=\sum_s k_{s-1}k_s$.
Denote the output after replacing $h^{(s)}$ by formal variables $Y$ by $P_s(Y)$; its degree is $q_s=m^{L-s}$.
Write $P_s(Y)=\langle C_s,Y^{\otimes q_s}\rangle$ and use the inherited Hilbert norm, namely
\begin{equation}\label{hm:coeff-norm}
 \|C_r(P)\|^2=\sum_{|\nu|=r}\frac{\nu!}{r!}|p_\nu|^2
 \quad\text{if }P(Y)=\sum_\nu p_\nu Y^\nu;
 \qquad C_L=k_L^{-1/2}{\bf1}.
\end{equation}
\begin{theorem}[Global matching of homogeneous networks with finitely many labels]\label{hm:global-matching}
There is a deterministic family of parameter alignment maps $\mathcal G_d$ such that any two networks with the original independent standard Gaussian prior marginals satisfy
\begin{equation}\label{hm:main}
 \log|\mathcal G_d|=O(d\log d)=o(p),\qquad
 p^{-1}\mathbb E\min_{g\in\mathcal G_d}\|W-gV\|_F^2
 \le K\mathbb E\|C_0(W)-C_0(V)\|^2+o_d(1).
\end{equation}
Here $K$ depends only on the fixed depth, degree, and width ratios, and the remainder is uniform over couplings.
The right-hand side can also be replaced by $K\mathbb E\|\mathcal H_W-\mathcal H_V\|_{L^2(N(0,I_d))}^2+o_d(1)$.
\end{theorem}

\paragraph{Joint typical event.}
The lengths of normalized weight rows converge uniformly to one; the self-coherence of the unit rows is
$\eta_d=O(\sqrt{\log d/d})$, the row arrays have bounded operator norm, and their column sums of squares have positive constant upper and lower bounds.
On this Gaussian concentration event, both Gram bounds from Section~\ref{tg:section} apply.
Let $B_{s,i}=\partial_iP_s$ and $w_j=W^{(s+1)}_{j:}/\sqrt{k_s}$; the chain rule gives
\[
 B_{s,i}(Y)=mc_m\sum_jw_{ji}(w_j\cdot Y)^{m-1}
 B_{s+1,j}\big(c_m(w\cdot Y)^{\odot m}\big).
\]
The expanded terms are precisely the factor-deleted tensors in \eqref{tg:frame} with labels $(\nu,j)$.
For fixed degree, the monomial coefficient norm is equivalent to \eqref{hm:coeff-norm}; backward induction from
$B_{L,i}=k_L^{-1/2}$ gives
$\|C(B_{s,i})\|^2\asymp\sum_j w_{ji}^2\|C(B_{s+1,j})\|^2\asymp k_s^{-1}$.
Thus the complete slices satisfy
\begin{equation}\label{hm:mode}
 a/k_s\le\|C_s(e_i,\cdot)\|^2=q_s^{-2}\|C(\partial_iP_s)\|^2\le b/k_s.
\end{equation}
In particular, $\|C_s\|\le C$. Let $\mathrm{SF}$ project onto square-free coordinates;
the coefficients in its orthonormal basis are $c_I=p_I/\sqrt{q_s!}$ ($|I|=q_s$).
Equation~\eqref{fp:core-coordinate-bounds} and the original coefficient bound~\ref{fp:core-raw-collision-expectation}
give, on the same marginal high-probability event,
\begin{align}
 \max_I|c_I|^2&\le C(\log d)^{D_s}k_s^{-q_s},\qquad D_s=m+\cdots+m^{L-s},\label{hm:dense}\\
 \|(I-\mathrm{SF})C_s\|^2&\le L_d/d=o(1).\label{hm:collision}
\end{align}
At the final layer, take $D_L=0$, and the second quantity is zero; $L_d$ is a suitable power of the logarithm.
Coordinated row and column transformations at internal layers preserve the weight concentration quantities and only permute coefficients and change their signs,
so all estimates remain valid pointwise after data-dependent alignment.

\paragraph{Global symmetry and grouped lifting.}
At one layer, write the normalized rows as $w_i=r_i a_i$, $\|a_i\|=1$, and set
$U_Ae_i=a_i^{\otimes m}$ and $\overline C_s=\operatorname{diag}(r_i^m)^{\otimes q_s}C_s$.
We have $\|U_A\|\le\|A\|^m\le C$ and the exact substitution formula
\begin{equation}\label{hm:substitution}
 C_{s-1}=c_m^{q_s}\operatorname{Sym}_{mq_s}U_A^{\otimes q_s}\overline C_s.
\end{equation}
The radii converge uniformly to one, so $\overline C_s-C_s=o(1)$; the coordinatewise radial multipliers are uniformly $1+o(1)$, so \eqref{hm:mode}--\eqref{hm:collision} remain valid.
Uniformly for any two typical networks and $q=q_s$,
\begin{equation}\label{hm:sym-group}
 \|C_{s-1}^A-C_{s-1}^B\|^2
 =\lambda_{m,q}\|G_A-G_B\|^2+o(1),\qquad
 \lambda_{m,q}=\frac{q!}{(mq)!}.
\end{equation}
Here $G_A=U_A^{\otimes q}\overline C_s^A$ and $G_B=U_B^{\otimes q}\overline C_s^B$.
Define the grouped and fully symmetric lifts of the square-free basis by
$Q_I=\sqrt{q!}\operatorname{Sym}_q\bigotimes_{i\in I}a_i^{\otimes m}$ and
$E_I=\sqrt{(mq)!/(m!)^q}\operatorname{Sym}_{mq}\bigotimes_{i\in I}a_i^{\otimes m}$.
Equation~\eqref{hm:substitution} maps this basis to $\sqrt{\lambda_{m,q}}E_I$.
Set $\rho_{ij}=\langle a_i,b_j\rangle$ and $\tau=d^{-1/8}$.
The bipartite graph $|\rho_{ij}|>\tau$ has degree at most $D=\lceil C/\tau^2\rceil$;
connect vertices on each side that share a neighbor, giving degree at most $D^2$. A $q$-element set containing no such edge is called good.
For good $I=\{i_1,\ldots,i_q\}$ and $\sum_av_a=m$,
\begin{equation}\label{hm:mixed-column}
 \sum_j\prod_a|\rho_{i_aj}|^{v_a}\le
 \begin{cases}C,&|\operatorname{supp}v|=1,\\ C\tau,&|\operatorname{supp}v|\ge2.\end{cases}
\end{equation}
At most one factor in each column exceeds $\tau$; partition by the position of the largest factor,
and retain either its square or one power of it and one power of another factor from the product.
At least one remaining power is at most $\tau$, while the sum of the two retained powers is controlled by the sum-of-squares bound or Cauchy--Schwarz.
Counting pairings gives
\[
 \langle E_I^A,E_J^B\rangle
 =\sum_{\substack{e_{ab}\ge0\\\sum_be_{ab}=\sum_ae_{ab}=m}}
 \frac{(m!)^q}{\prod_{a,b}e_{ab}!}\prod_{a,b}\rho_{i_aj_b}^{e_{ab}}.
\]
The sum of terms with $e=mP$ ($P$ a permutation matrix) is exactly $\langle Q_I^A,Q_J^B\rangle$.
Every other term has at least one column mixing two labels; enlarge the sum over $J$ to all ordered indices and separate the columns,
then \eqref{hm:mixed-column} gives an absolute row sum $O(\tau)$. Interchanging the two sides gives the column sum.
Let $K_{\rm act}=(\langle E_I^A,E_J^B\rangle)_{I,J}$ and
$K_{\rm grp}=(\langle Q_I^A,Q_J^B\rangle)_{I,J}$ be the cross-Gram matrices on good sets. Then
\begin{equation}\label{hm:cross-schur}
 \|K_{\rm act}-K_{\rm grp}\|_{\rm op}\le C\tau.
\end{equation}
Taking the two networks to be identical gives the self-Gram matrices; since $\eta_d<\tau$, every set is good.
There are at most $C_qD^2k^{q-1}$ bad sets, whose total coefficient energy is at most
$Cd^{-1/2}(\log d)^{D_s}=o(1)$ by \eqref{hm:dense}; this count also applies to graphs depending on both networks and their coefficients.
Both lifts are bounded, so deleting bad sets and repeated indices incurs only $o(1)$ error; apply
\eqref{hm:cross-schur} to the three Gram blocks and restore the full coefficients to obtain \eqref{hm:sym-group}.
For $q=1$, no slots split, and the equality is exact.

\paragraph{Single-layer matching and coefficient recovery.}
Let $M=U_A^*U_B=(\rho_{ij}^m)$ and $\rho_i=\max_j|\rho_{ij}|$.
For each row, the student rows with correlation exceeding $\tau$ number at most $D$, and their self-Gram norm is at most $1+D\eta_d$;
for the remaining terms, retain the square and extract $\tau^{m-2}$ to obtain
\begin{equation}\label{hm:cross-power}
 \sum_j|M_{ij}|\le\rho_i^{m-2}+e_d,\qquad
 \sum_i|M_{ij}|\le1+e_d,\qquad
 e_d=C(D\eta_d+\tau^{m-2})=o(1).
\end{equation}
Write the full ordered coefficient arrays as $C=\overline C_s^A=(c_I)$ and $D'=\overline C_s^B=(d_J)$,
and let $\mathsf K=(M/(1+e_d))^{\otimes q}$. Its absolute row and column sums satisfy $r_I,s_J\le1$, and
\begin{equation}\label{hm:transport-identity}
 \begin{aligned}
 \mathcal E&:=\|C\|^2+\|D'\|^2-2\langle C,\mathsf K D'\rangle\\
 &=\sum_I(1-r_I)c_I^2+\sum_J(1-s_J)d_J^2
 +\sum_{I,J}|\mathsf K_{IJ}|(c_I-\operatorname{sgn}(\mathsf K_{IJ})d_J)^2.
 \end{aligned}
\end{equation}
Since $U_A^*U_A,U_B^*U_B=I+o_{\rm op}(1)$ (the off-diagonal absolute row sums are at most $k\eta_d^m=o(1)$),
we have $\mathcal E=\|G_A-G_B\|^2+o(1)$.
As $r_I$ is the product of the slotwise row sums, the deficit in the first slot and the complete-slice lower bound \eqref{hm:mode} give
$k_s^{-1}\sum_i(1-\rho_i^{m-2})\le C\mathcal E+o(1)$.
Fix $\rho_0\in(1/\sqrt2,1)$. Since $2\rho_0^2>1+\eta_d$, edges with absolute correlation exceeding $\rho_0$ form a partial bijection.
Complete it to a permutation and orient each row so that diagonal correlations are nonnegative; the unmatched fraction is controlled by the deficit above,
and applying $1-x^m\le m(1-x^{m-2})/(m-2)$ to matched rows gives
\begin{equation}\label{hm:row-loss}
 k_s^{-1}\sum_i\big(\ell_i+\|a_i-b_i\|^2\big)\le C\mathcal E+o(1),
 \qquad \ell_i=1-\langle a_i,b_i\rangle^m.
\end{equation}
Simultaneously permute the coefficients and multiply them by the corresponding signs to the $m$th power, preserving $G_B$.
Then $\mathsf K_{II}\ge0$ and $1-\mathsf K_{II}\le\sum_{a=1}^q(\ell_{i_a}+e_d)$.
Retaining the diagonal squared terms in \eqref{hm:transport-identity} and using the complete-slice upper bound gives
\[
 \|C-D'\|^2\le\mathcal E+\sum_I(1-\mathsf K_{II})(c_I-d_I)^2
 \le\mathcal E+\frac{Cq}{k_s}\sum_i(\ell_i+e_d)\le C\mathcal E+o(1).
\]
Removing the radial factors and combining with \eqref{hm:sym-group} yields
\begin{equation}\label{hm:recover}
 \|C_s^A-C_s^B\|^2+k_s^{-1}\sum_i\|w_i-v_i\|^2
 \le C\|C_{s-1}^A-C_{s-1}^B\|^2+o(1).
\end{equation}
The rows here are aligned; the radial error is absorbed using
$\|r_ia_i-t_ib_i\|^2\le C\|a_i-b_i\|^2+C\max_j(|r_j-1|+|t_j-1|)^2$.

\paragraph{Depth induction and arbitrary couplings.}
Write the signed row permutation at an internal layer as $S_s=D_sP_s$ and let
$R_s=D_s^{m\bmod2}P_s$. Simultaneously transform the next layer's columns to $V^{(s+1)}R_s^{-1}$,
preserving the entire homogeneous function and the current $C_{s-1}$.
Apply \eqref{hm:recover} to internal layers from bottom to top; subsequent alignments do not affect completed layers,
and at fixed depth the mean normalized row distance at every layer is at most $C\|C_0(W)-C_0(V)\|^2+o(1)$.
At the final layer, $q_L=1$, and only \eqref{hm:row-loss} is needed: for orientation signs $\varsigma_i$,
replace the student coefficients in the analytical representation by $\varsigma_i^m t_i^m/\sqrt{k_L}$, preserving the original tensor and slice bounds.
There is no need to recover the final-layer coefficients; final-layer signs serve only as parameter labels and need not preserve the function.
Take all maps
$(gV)^{(1)}=S_1V^{(1)}$ and $(gV)^{(s)}=S_sV^{(s)}R_{s-1}^{-1}$; then
$\log|\mathcal G_d|\le\sum_s(\log(k_s!)+k_s\log2)=O(d\log d)$.
Breaking ties in a fixed label order makes the choice measurable. Weighting each layer's bound by $k_{s-1}k_s/p$ and summing
gives \eqref{hm:main} on the typical event.
Under any prior coupling, the failure probability is at most the sum of the two marginal failure probabilities; on the bad event, use the identity alignment and
$p^{-1}\|W-V\|^2\le2p^{-1}(\|W\|^2+\|V\|^2)$.
The uniformly bounded second moments of normalized Gaussian squared norms and Cauchy--Schwarz make its expected contribution uniformly $o(1)$.
Finally, the isometry of Gaussian chaos of order $N=m^L$ gives
$N!\|C_0(W)-C_0(V)\|^2=\|\Pi_N(\mathcal H_W-\mathcal H_V)\|_{L^2}^2
\le\|\mathcal H_W-\mathcal H_V\|_{L^2}^2$, proving the function-distance version.\hfill$\square$

\paragraph{Parameter comparison for the actual functions.}
Equation~\eqref{fp:global-function-pairing} and the polarization identity in the main text convert control of the actual function distance into control of the homogeneous coefficient distance;
then apply Theorem~\ref{hm:global-matching}. Row signs are used only for finite-label alignment and need not preserve the function values of the actual activation.

\section{Conditional I--MMSE and a unified smoothed test field}\label{app:information}
Write $f=F_{\theta(W)}^{(L)}(X)$, $a=\nabla_Wf$,
$\mathcal I_W=\E_Xaa^T/\Delta$, and $Z=(\mathcal D_n,X)$.
Take an independent standard Gaussian $N$, and let $U_s=\sqrt sf+N$ and
$m(s)=\E[(f-\E[f\mid Z,U_s])^2]$. The true Gaussian output channel and conditional I--MMSE give
\begin{equation}\label{eq:conditional-immse}
 \epsilon_{\log,d,n}=I(f;U_{1/\Delta}\mid Z)
 =\frac12\int_0^{1/\Delta}m(s)\,ds.
\end{equation}
To prove the second equality, condition on $Z=z$ and let $q_v$ be the density of $f+\sqrt vN$.
The Gaussian heat equation, integration by parts, and the conditional expectation formula give
\[
 \frac d{dv}h(q_v)=\frac12\mathcal J(q_v),\qquad
 \partial_y\log q_v(y)=\frac{\E[f\mid z,f+\sqrt vN=y]-y}{v}.
\]
Orthogonal decomposition gives $\operatorname{mmse}(f\mid z,f+\sqrt vN)=v-v^2\mathcal J(q_v)$.
Subtract the noise entropy from $h(q_v)$, differentiate, and set $v=1/s$ to obtain that the derivative of mutual information equals half the MMSE.
These calculations first hold for truncated signals on bounded spatial domains; using $\mathcal J(q_v)\le1/v$
and uniform integrability of the signal second moments, remove the truncations on any interval $0<s_0\le s\le s_1$.
Finally, use $I(f;U_s\mid Z)\le s\E f^2/2$ and $m(s)\le\E f^2$,
average over $z$, and let $s_0\downarrow0$ to obtain \eqref{eq:conditional-immse}.

\subsection{Integration-by-parts bound for the true channel}
For a vector field depending only on $(W,X)$, denoted by $b$, let
\[
 G_s=\delta b+b^TS_D+\sqrt sN b^Ta,\qquad
 \delta b=\operatorname{div}_Wb-W\cdot b,\quad
 S_D=\nabla_W\log p(\mathcal D_n\mid W).
\]
Fixing the observations and integrating by parts over the parameters gives
$\E[(\E[f\mid Z,U_s]-f)G_s]=\E b^Ta$.
Given $(W,X)$, the training score and auxiliary noise are independent and centered, and the training-score covariance is
$n\mathcal I_W$. Thus Cauchy--Schwarz gives
\begin{equation}\label{eq:vt-mmse}
 m(s)\ge\frac{(\E b^Ta)^2}
 {n\E b^T\mathcal I_Wb+\E(\delta b)^2+s\E(b^Ta)^2}.
\end{equation}
First prove this for smooth compactly supported fields and truncated posterior means, then pass to the limit by the Sobolev approximation in Lemma \ref{an:sobolev} and $L^2$ convergence; integrability is established below.

\subsection{OU smoothing of the regularized test field}
Fix $0<\tau\le r\le1$ and let $\lambda=r/p$,
$M_W=(\mathcal I_W+\lambda I_p)^{-1}$, and $b_0(W,X)=M_Wa(W,X)$.
Take $W_\tau$ from \eqref{eq:gradient-stability} and conditionally average only over parameters:
\[
 b_\tau(W,X)=\E[b_0(W_\tau,X)\mid W,X].
\]
The true network, training likelihood, and auxiliary observation $U_s$ remain unchanged. We have
\[
 \E\|b_0\|^2=\Delta\E\tr(M_W^2\mathcal I_W)
 \le\frac{\Delta p}{4\lambda},\qquad
 \E\|b_\tau\|^4\le\E\|b_0\|^4\le C\lambda^{-4}.
\]
View $b_0$ as an $L^2(P_X;\mathbb R^p)$-valued function, and denote its order-$k$ parameter Hermite
energy by $c_k$. The OU semigroup multiplies each order by $e^{-\tau k}$, so
\[
 \E\|Db_\tau\|_{\rm HS}^2
 =\sum_{k\ge1}ke^{-2\tau k}c_k
 \le C\tau^{-1}\E\|b_0\|^2\le Cp^2/(r\tau).
\]
Thus $b_\tau$ belongs to Gaussian $W^{1,2}$ without differentiating $M_W$.
By Lemma \ref{an:sobolev}, the fourth gradient moment, and $\tau\le r$,
\begin{equation}\label{unified:field-cost}
 \E(\delta b_\tau)^2\le Cp^2/(r\tau),\qquad
 \E(b_\tau^Ta)^2\le
 (\E\|b_\tau\|^4\E\|a\|^4)^{1/2}
 \le Cp^2/r^2\le Cp^2/(r\tau).
\end{equation}

Write $b'(x)=b_0(W_\tau,x)$, $a'(x)=a(W_\tau,x)$, and
$e(x)=a(W,x)-a'(x)$, and take another independent input $\overline X$.
Conditionally on $(W,W_\tau,\overline X)$, $X$ remains independent, so
\[
 \E_Xb'(X)b'(X)^T=\Delta M_{W_\tau}\mathcal I_{W_\tau}M_{W_\tau}
 \preceq\frac{\Delta}{4\lambda}I_p.
\]
By \eqref{eq:gradient-stability}, $\E\|e(X)\|^2\le C\tau^\delta$.
Cauchy--Schwarz and the conditional covariance above give, respectively,
\[
 |\E b'(X)^Te(X)|\le Cp\tau^{\delta/2}/\sqrt r,
 \qquad
 \Delta^{-1}\E[(b'(X)^Te(\overline X))^2]
 \le\frac{\E\|e(\overline X)\|^2}{4\lambda}\le Cp\tau^\delta/r.
\]
Invariance of the Gaussian prior gives $\E b'(X)^Ta'(X)=\Delta p[1-D_d(r)]$;
independent inputs give
$\Delta^{-1}\E[(b'(X)^Ta'(\overline X))^2]
 =\Delta\E\tr[(M_W\mathcal I_W)^2]\le\Delta p$.
Conditional Jensen and the $L^2$ triangle inequality therefore give
\begin{equation}\label{field:target}\begin{aligned}
 \E b_\tau^Ta&=\E b'(X)^T(a'(X)+e(X))
 \ge\Delta p[1-D_d(r)-C\tau^{\delta/2}/\sqrt r],\\
 \E b_\tau^T\mathcal I_Wb_\tau
 &\le\Delta^{-1}\E[(b'(X)^T(a'(\overline X)+e(\overline X)))^2]
 \le\Delta p[1+C\tau^{\delta/2}/\sqrt r]^2.
\end{aligned}\end{equation}
Take the positive part of the target lower bound, substitute it together with \eqref{unified:field-cost} into
\eqref{eq:vt-mmse}, and integrate in \eqref{eq:conditional-immse}.
Finally, take $\tau=r^{2/\delta}\le r$, so that $\tau^{\delta/2}/\sqrt r=\sqrt r$,
to obtain \eqref{eq:lower-unified}.
