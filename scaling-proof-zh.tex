\begin{abstract}
对任意固定深度、各层宽度与输入维数成正比的网络，本文证明平均 Bayes 预测 KL 的首项为 $\Pall(d)/(2n)$。激活类包含中心化归一化的 tanh、sigmoid 和 ReLU。证明先从函数距离控制参数在排列和符号变换下的距离，再用互信息估计得到 Fisher 信息矩阵所需的谱性质。条件 I--MMSE 与高斯分部积分给出下界，线性高斯参考分布给出相同首项的上界。维数极限先于样本比极限；证明不要求固定样本比下的风险极限存在，也不依赖后验的局部高斯近似。
\end{abstract}

\section{模型、激活类与定理}
固定 $L\ge1$，$k_0=d$，$k_\ell/d\to\gamma_\ell\in(0,\infty)$，$\gamma_0=1$。
第 $\ell$ 层权重为 $W^{(\ell)}\in\R^{k_\ell\times k_{\ell-1}}$，其元素独立标准高斯。输入 $x\sim N(0,I_d)$ 独立于参数。本文每层使用同一固定激活 $\sigma_\ell\equiv\sigma$，且
\[
\E\sigma(G)=0,\qquad\E\sigma(G)^2=1,\qquad G\sim N(0,1).
\]
记 $W=(W^{(\ell)})_{\ell=1}^L$、$\theta=\theta(W)=(v,W)$，固定 $v_i=1$，网络定义为
\begin{equation}\begin{aligned}
h^{(0)}(x)&=x,\qquad
h^{(\ell)}(x)=\sigma_\ell\!\left(\frac{W^{(\ell)}h^{(\ell-1)}(x)}{\sqrt{k_{\ell-1}}}\right),\\
F_\theta^{(L)}(x)&=\frac{v^{\mathsf T}h^{(L)}(x)}{\sqrt{k_L}}.
\end{aligned}\end{equation}
真实参数 $\theta^0=(v,W^0)$ 由上述先验 $P_0$ 抽取。训练集 $\mathcal D_n=\{(x_\mu,y_\mu)\}_{\mu=1}^n$ 由
\[
y_\mu=F_{\theta^0}^{(L)}(x_\mu)+\sqrt\Delta z_\mu,\qquad
z_\mu\sim N(0,1),\quad\Delta>0
\]
生成。训练和测试输入独立同分布；噪声相互独立，且独立于参数与全部输入。记高斯通道为
\[
\Pout(y\mid\lambda)=N(y;\lambda,\Delta).
\]
学习者使用真实先验和通道，其后验为
\[
P(d\theta\mid\mathcal D_n)\propto P_0(d\theta)
\prod_{\mu=1}^n\Pout\!\left(y_\mu\mid F_\theta^{(L)}(x_\mu)\right).
\]
只有隐藏矩阵被学习，读出 $v$ 固定；隐藏参数数目为
\[
\Pall(d)=\sum_{\ell=1}^Lk_{\ell-1}k_\ell,\qquad
c=\lim_d\frac{\Pall(d)}{d^2}
=\sum_{\ell=1}^L\gamma_{\ell-1}\gamma_\ell.
\]
后验预测密度及平均预测 KL 定义为
\begin{equation}\begin{aligned}
p_{\mathcal D_n}(y\mid x)
&=\E_{\theta\mid\mathcal D_n}\Pout\!\left(y\mid F_\theta^{(L)}(x)\right),\\
\epsilon_{\log,d,n}
&=\E_{\theta^0,\mathcal D_n,x}\KL\!\left(
\Pout\!\left(\cdot\mid F_{\theta^0}^{(L)}(x)\right)
\,\middle\|\,p_{\mathcal D_n}(\cdot\mid x)\right).
\end{aligned}\end{equation}
写 $\epsilon_{\log,d}(\alpha):=\epsilon_{\log,d,\lfloor\alpha d^2\rfloor}$。
技术证明中简记 $p=\Pall(d)$，并把 $W$ 视为 $p$ 维隐藏权重坐标；梯度、Hessian、高斯积分和 OU 耦合均只作用于这些坐标。写 $\theta'=\theta(V)$ 表示第二组权重对应的参数。先验平均中可将真实隐藏权重 $W^0$ 简记为 $W$；技术附录中的 $F_W=F_{\theta(W)}^{(L)}$ 与 $F=F_W$ 均指同一个实际输出，除非明确使用下文定义的齐次多项式网络。

所有常数允许依赖固定深度、激活、宽度比例与噪声，不能依赖 $d,n$。

\subsection{激活类}
\newthought*{取下列两类的并集 $\mathcal A_{\rm bal}$。}
两支均满足上述中心化及单位方差条件。
\emph{光滑奇函数支：} $\sigma$ 有界、奇、属于 $C^\infty$，每个固定阶导数有界。
\emph{分段线性支：} 令 $c_0=\sqrt{2/\pi}$，
\begin{equation}\label{eq:pl}
\sigma(z)=az+b(|z|-c_0),\qquad
b\ne0,\qquad a^2+b^2(1-2/\pi)=1.
\end{equation}
两支均全局 Lipschitz。条件只涉及一个实函数，不包含 Fisher、后验或信息极限前提。

记 $\operatorname{He}_j$ 为概率论约定的 Hermite 多项式，$\psi_j=\operatorname{He}_j/\sqrt{j!}$、$a_j=\E[\sigma(G)\psi_j(G)]$。
与网络符号一致，$\mu_{\ell,j}=\mu_j=\sqrt{j!}a_j$。以下事实使光滑支无需另加高阶系数假设。
\begin{lemma}[高阶奇系数自动存在]\label{lem:automatic-hermite}
若 $\sigma$ 为有界连续奇函数且 $\E\sigma(G)^2=1$，则存在奇数 $m\ge5$ 使 $a_m\ne0$。
\end{lemma}
\begin{proof}
奇性使所有偶次系数为零。若所有奇数 $m\ge5$ 的系数也为零，
Hermite 系的 $L^2(\gamma)$ 完备性给出
$\sigma=a_1\psi_1+a_3\psi_3$，高斯几乎处处成立。
两边连续且高斯密度处处为正，故处处相等。
有界多项式只能为常数，奇性又使该常数为零，与单位方差矛盾。
\end{proof}
光滑支固定选择上述一个 $m$；分段线性支取 $m=6$，其 $a_6\ne0$ 在下节直接验证。
$m$ 可依赖固定激活，但不随维数或样本数变化；后文常数允许依赖此选择。

\begin{theorem}\label{thm:main}
上述模型中，对任意固定深度 $L\ge1$，任一固定 $\sigma\in\mathcal A_{\rm bal}$ 满足
\begin{equation}\label{eq:main}
\boxed{\lim_{\alpha\to\infty}\limsup_{d\to\infty}
\left|\alpha \epsilon_{\log,d}(\alpha)-\frac c2\right|=0.}
\end{equation}
该结论包含任意固定正斜率的中心化归一化 tanh、logistic sigmoid，以及中心化归一化 ReLU。
\end{theorem}
先取 $d\to\infty$，再取 $\alpha\to\infty$。不要求固定 $\alpha$ 的风险极限存在；不据此扩展到未中心化模型、层间任意混用两支、增长深度或非高斯输出下同一精确常数的上界。

\section{三个激活为何属于此类}
对任意固定 $a>0$，令
\[
s_a^2=\E\tanh^2(aG)>0,\qquad \sigma_a(z)=\tanh(az)/s_a.
\]
它有界、光滑且为奇函数，因而中心化，定义给出单位方差。
由 $\tanh'=1-\tanh^2$ 递推，每个固定阶导数都是 $\tanh(az)$ 的多项式乘固定常数，故有界。
因此 $\sigma_a$ 属于光滑奇函数支，高阶非零系数由引理~\ref{lem:automatic-hermite} 自动保证。

对任意固定 $b>0$，logistic sigmoid $\ell_b(z)=(1+e^{-bz})^{-1}$ 满足
\[
\ell_b(z)-\tfrac12=\tfrac12\tanh(bz/2).
\]
其均值为 $1/2$，方差为 $s_{b/2}^2/4$，故中心化归一化后恰为 $\sigma_{b/2}$。

对 ReLU，直接积分给出
\[
\mu_R=\E G_+=\frac1{\sqrt{2\pi}},\qquad
v_R=\operatorname{Var}(G_+)=\frac12-\frac1{2\pi},\qquad
\sigma_R(z)=\frac{z_+-\mu_R}{\sqrt{v_R}}.
\]
由 $z_+=(z+|z|)/2$，它对应式~\eqref{eq:pl} 的
$a=b=1/(2\sqrt{v_R})$。
整个分段线性支都中心化；奇偶正交性给出其方差
$a^2+b^2(1-2/\pi)=1$。
又由
\[
\operatorname{He}_6(z)=z^6-15z^4+45z^2-15,\qquad
\E|G|^{2j+1}=2^j j!\sqrt{2/\pi},
\]
得到 $\E[|G|\operatorname{He}_6(G)]=3\sqrt{2/\pi}$，所以
\[
a_6=\frac{3b\sqrt{2/\pi}}{\sqrt{6!}}\ne0.
\]
分段线性支的奇次系数 $a_j$ 在 $j\ge3$ 时均为零；光滑奇函数支的偶次系数均为零。
这正是后文两类展开项所使用的奇偶结构。

\section{先验矩与局部估计}
令 $q_\ell=\|h^{(\ell)}\|^2/k_\ell$、$a(W,X)=\nabla_WF_{\theta(W)}^{(L)}(X)$。
附录引理~\ref{an:moments}--\ref{an:ou} 证明：对每个固定 $r>0$，
\begin{equation}\label{eq:moments}
 \E|q_\ell-1|^r\le C_r d^{-r/2},\qquad
 \E|F_{\theta(W)}^{(L)}(X)|^r+\E\|a(W,X)\|^r\le C_r.
\end{equation}
对与 $(W,X)$ 独立的标准高斯 $Z$，还有
\begin{equation}\label{eq:gradient-stability}
 W_\tau=e^{-\tau}W+\sqrt{1-e^{-2\tau}}Z,\qquad
 \E\|a(W_\tau,X)-a(W,X)\|^2\le C\sqrt\tau\quad(0<\tau\le1).
\end{equation}
对 $0<t\le1/2$，写 $W(t)=\sqrt{1-t}W+\sqrt tZ$，则一阶展开满足
\begin{equation}\label{eq:ou-rem}
 \begin{aligned}
 F_{\theta(W(t))}^{(L)}-F_{\theta(W)}^{(L)}&=\sqrt t\,a(W,\cdot)^TZ+E_t,\\
 \E\|E_t\|_{L_X^2}^2&\le Ct^{1+\delta},\qquad
 \delta=1\ \text{（光滑支）},\quad\delta=1/2\ \text{（分段线性支）}.
 \end{aligned}
\end{equation}
常数和维数阈值均对扰动参数一致；后文两支统一使用 $\delta=1/2$。

\section{全局函数比较与参数匹配}
本节证明：两个网络的输出接近时，其参数在适当排列和符号变换后也接近。先用一个齐次多项式网络提取实际输出中可比较的分量。
写 $\psi_j=\operatorname{He}_j/\sqrt{j!}$、$a_j=\E\sigma(G)\psi_j(G)$，并置
\[
N=m^L,\qquad S=1+m+\cdots+m^{L-1},\qquad B=a_m^S\ne0.
\]
以各层齐次激活 $z^m/\sqrt{m!}$、原权重与原读出构造多项式网络，其最高输入次数系数为 $C_N(W)\in\Sym^N\R^d$。定义
\[
H_m(W)=I_N(C_N(W)),\qquad T_W=BH_m(W).
\]
记 $\Pi_N$ 为 $L^2(P_X)$ 到第 $N$ 阶 Hermite 子空间的正交投影。$I_N$ 将对称张量映为该子空间中的多项式：
对 $v\ne0$，$I_N(v^{\otimes N})(X)=\|v\|^N\operatorname{He}_N(\langle v,X\rangle/\|v\|)$，再线性延拓。
其归一化满足
\begin{equation}\label{fp:actual-isometry}
 \langle I_N(C),I_N(D)\rangle_{L_X^2}=N!\langle C,D\rangle.
\end{equation}
张量均取对称张量空间中继承的 Hilbert 范数。

\begin{lemma}[函数内积的统一估计]\label{lem:cross}
若 $\Pi_d$ 为原高斯参数先验，记 $\Gamma(\Pi_d,\Pi_d)$ 为两个边缘均为 $\Pi_d$ 的联合分布集合，则
\begin{equation}\label{eq:cross}
e_d:=\sup_{\pi\in\Gamma(\Pi_d,\Pi_d)}
\E_\pi|\langle T_W,F_{\theta'}^{(L)}-T_V\rangle_{L_X^2}|\longrightarrow0.
\end{equation}
这里允许任意正确先验边缘的耦合，不要求 $W,V$ 独立。
\end{lemma}
\begin{proof}
附录~\ref{app:closure} 将实际输出的第 $N$ 阶 Hermite 投影逼近为有限个多项式网络的系数。
其中分段线性支的较浅网络修正项也全部保留。
附录~\ref{app:cuts} 用统一的深度归纳证明：这些项与 $T_W$ 配对时，
只有深度 $L$、每个节点次数均为 $m$ 的项保留，其 $S$ 个系数给出 $B$，故恰为 $T_V$。
单网络逼近误差由 Cauchy--Schwarz 控制；所有界只使用各自先验边缘，
因而对耦合一致。完整推导见式~\eqref{fp:global-function-pairing}。
\end{proof}

\subsection{从函数比较到参数匹配}
对内积估计展开平方，得到齐次多项式网络的距离控制。
对某个正确边缘耦合令
$D=\E\norm{T_W-T_V}^2$、$B_F=\E\norm{F_\theta^{(L)}-F_{\theta'}^{(L)}}^2$。
在原耦合、转置耦合以及两个对角项中分别使用 \eqref{eq:cross}，得
\[
|\E\langle T_W-T_V,F_\theta^{(L)}-F_{\theta'}^{(L)}\rangle-D|\le4e_d.
\]
所以
\[
D\le\sqrt{DB_F}+4e_d\le(D+B_F)/2+4e_d,
\quad D\le B_F+8e_d.
\]
齐次全局匹配定理给出有限可测映射族 $\mathcal G_d$，
\begin{equation}\label{eq:matching}\begin{aligned}
\log|\mathcal G_d|&=O(d\log d)=o(p),\\
\frac1p\E\min_{g\in\mathcal G_d}\norm{W-gV}^2
&\le C\E\norm{H_m(W)-H_m(V)}^2+o_d(1)\\
&\le C\E\norm{F_\theta^{(L)}-F_{\theta'}^{(L)}}^2+o_d(1).
\end{aligned}\end{equation}
误差对耦合一致。附录 \ref{app:matching} 通过逐层匹配证明式 \eqref{eq:matching}，右侧仍为函数距离的平方。映射可包含行符号变换；后文用这些有限标签控制条件熵，不要求它们保持实际网络输出。

\section{互信息界与后验收缩}
记 $I_{d,n}=I(\theta^0;\mathcal D_n)=I(W;\mathcal D_n)$。Bayes 定义和样本条件独立性给出
\begin{equation}\label{eq:increment}
\epsilon_{\log,d,n}=I_{d,n+1}-I_{d,n},\qquad
\epsilon_{\log,d,n+1}\le \epsilon_{\log,d,n}.
\end{equation}
第二式也可由条件熵递减看出：真实条件噪声熵不随数据量改变，预测熵随观察增加而减小。

\subsection{粗风险上界}
由下一节仅依赖先验矩与 OU 余项证明的 \eqref{eq:sample-information-upper}，
取 $n/p\ge1$、$t=(1+n/p)^{-1}$。此时
\[
 \E\log\det(I_p+ntA/\Delta)\le nt\E\tr A/\Delta\le Cp,
 \qquad (n/p)t^{1+\delta}\le1,
\]
所以 $I_{d,n}\le\frac p2\log(1+n/p)+Cp$。
由 \eqref{eq:increment}，$n\epsilon_{\log,d,n}\le I_{d,n}$，因此
\begin{equation}\label{eq:coarse}
 \limsup_d\epsilon_{\log,d}(u)\le C\frac{\log u}{u}
 \qquad(u\text{ 充分大}).
\end{equation}

\subsection{后验参数距离与条件熵}
给定 $\mathcal D_n$ 独立抽取后验样本 $W,V$；平均数据后两者保持原先验边缘。
记 $B_X=F_{\theta(W)}^{(L)}(X)-F_{\theta(V)}^{(L)}(X)$。
以后验预测密度为中间点，Hellinger 三角不等式和 $H^2\le\KL$ 给
\begin{equation}\label{eq:hellinger}
 \E[1-e^{-B_X^2/(8\Delta)}]\le2\epsilon_{\log,d,n},
\end{equation}
其中指数项是两个同方差高斯密度的亲和度。对固定 $q>2$，标量不等式
\[
 z^2\le C_q(1+|z|^q)^{2/q}
          (1-e^{-z^2/(8\Delta)})^{1-2/q}
\]
由分别考察 $|z|\le1$ 和 $|z|>1$ 得到。H\"older 与 \eqref{eq:moments} 因而给
$\E B_X^2\le C_q\epsilon_{\log,d,n}^{1-2/q}$。定义
\[
 r_{d,n}=\frac1p\E\min_{g\in\mathcal G_d}\|W-gV\|^2,
 \qquad r_*(u)=\limsup_d r_{d,\lfloor ud^2\rfloor}.
\]
在 \eqref{eq:matching} 中取此后验耦合，再取 $q=2/\eta$，由 \eqref{eq:coarse} 得
\begin{equation}\label{eq:rate}
 r_*(u)\le C_\eta(\log u/u)^{1-\eta},\qquad0<\eta<1.
\end{equation}

在同一耦合中选择可测最优标签 $\widehat g$，并令
$V_t=\sqrt{1-t}W+\sqrt tU$，其中 $U$ 独立于上述变量且为标准高斯，$0\le t\le1/2$。
给定 $\mathcal D_n$，$V$ 与 $V_t$ 独立；且
\[
 \E\|V_t-\sqrt{1-t}\widehat gV\|^2=p[t+(1-t)r_{d,n}].
\]
条件于 $(\mathcal D_n,V,\widehat g)$ 后用高斯最大熵，再用 Jensen，得到
\begin{equation}\label{info:matched-entropy}
 h(V_t\mid\mathcal D_n)
 \le\log|\mathcal G_d|+\frac p2\log\{2\pi e[t+(1-t)r_{d,n}]\}.
\end{equation}
这里加入标签的条件熵至多为 $\log|\mathcal G_d|$。
$t=0$ 时 $V_0=W$；从 $h(W)=\frac p2\log(2\pi e)$ 减去上述界并用 \eqref{eq:rate}，得
\begin{equation}\label{eq:sample-information-lower}
 \liminf_d\frac{I_{d,\lfloor\alpha d^2\rfloor}}p
 \ge\frac{1-\eta}{2}\log\alpha
       -\frac{1-\eta}{2}\log\log\alpha-C_\eta.
\end{equation}
这里 $\eta\in(0,1)$ 固定，$\alpha$ 充分大。
对 $t>0$，记 $J_n(t)=I(W;V_t\mid\mathcal D_n)$。
从 \eqref{info:matched-entropy} 减去条件噪声熵，得到
\begin{equation}\label{alt:oracle-cost}
 J_n(t)\le\log|\mathcal G_d|+\frac p2\log\left(1+\frac{(1-t)r_{d,n}}t\right)
 \le\log|\mathcal G_d|+\frac{p r_{d,n}}{2t}.
\end{equation}

\section{线性高斯参考分布与互信息界}
对上述 $V_t$，$0<t\le1/2$ 时可反向写为
$W=\sqrt{1-t}V_t+\sqrt t\xi$，其中 $\xi$ 与 $V_t$ 独立标准高斯。
记 $f_W=F_{\theta(W)}^{(L)}$、$a(x)=\nabla f_{V_t}(x)$、
$A=\E_Xa(X)a(X)^T$。已有 \eqref{eq:ou-rem}、\eqref{eq:moments} 给出
\[
 f_W(x)-f_{V_t}(x)=\sqrt t\,a(x)^T\xi+R_t(x),\qquad
 \E\|R_t\|_{L_X^2}^2\le Ct^{1+\delta},\quad
 \E\|a(X)\|^4\le C.
\]
两支统一可取 $\delta=1/2$。

\subsection{相对熵分解与互信息增量}
给定 $V_t,X_{1:j}$，令 $\Phi_j$ 的第 $i$ 行为 $a(X_i)^T$，定义
\[
 Q_j=N\!\left((f_{V_t}(X_i))_{i\le j},S_j\right),
 \qquad S_j=\Delta I_j+t\Phi_j\Phi_j^T.
\]
这些参考分布由同一个虚拟系数 $\zeta\sim N(0,I_p)$ 的线性模型
$f_{V_t}(X_i)+\sqrt t\,a(X_i)^T\zeta+\sqrt\Delta Z_i$ 产生，
因此其样本边缘相容。输入采用真实的独立分布，虚拟系数不作为额外观察。
令 $P_j$ 为给定 $V_t$ 的真实数据分布，定义
\[
 C_j=\E\log\frac{p(Y_{1:j}\mid W,X_{1:j})}
 {q_j(Y_{1:j}\mid V_t,X_{1:j})},\qquad
 B_j=\E_{V_t}\KL(P_j\|Q_j).
\]
$P_j,Q_j$ 包含相同的输入边缘。由相对熵分解和 Markov 链 $V_t-W-\mathcal D_j$，
\begin{equation}\label{info:reference-identity}
 I_{d,j}+J_j(t)=\frac p2\log(1/t)+C_j-B_j.
\end{equation}
参考分布的边缘相容性给 $B_n\ge B_j\ge0$。结合 $J_n(t)\ge0$，同一恒等式给
\begin{equation}\label{alt:block-entropy}
 I_{d,n}\le\frac p2\log(1/t)+C_n,\qquad
 I_{d,n}-I_{d,j}\le C_n-C_j+J_j(t).
\end{equation}

\subsection{交叉熵的非线性误差}
令 $L_j=\sqrt t\Phi_j\xi$、$R_j=(R_t(X_i))_{i\le j}$。
高斯密度公式给出
\[
 C_j=\frac12\E\log\det(I_j+t\Phi_j\Phi_j^T/\Delta)
     +\E L_j^TS_j^{-1}R_j+\frac12\E R_j^TS_j^{-1}R_j.
\]
这里用到了精确抵消
\[
 \E[L_j^TS_j^{-1}L_j\mid V_t,X_{1:j}]
   +\Delta\tr S_j^{-1}=j.
\]
由于 $S_j\succeq\Delta I_j$，且线性模型的有效维数至多 $p$，
\[
 \E R_j^TS_j^{-1}R_j\le Cjt^{1+\delta},\qquad
 \E L_j^TS_j^{-1}L_j
 =\E\tr(t\Phi_j\Phi_j^TS_j^{-1})\le p.
\]
加权 Cauchy--Schwarz 与 Sylvester 恒等式遂给
\begin{equation}\label{alt:cross-entropy}
 \left|C_j-\frac12H_j\right|
 \le C\sqrt{pj t^{1+\delta}}+Cjt^{1+\delta},\qquad
 H_j=\E\log\det(I_p+t\Phi_j^T\Phi_j/\Delta).
\end{equation}
不要求 $R_j$ 与 $\xi$ 独立。

\subsection{经验与总体行列式的比较}
置 $\overline H_j=\E\log\det(I_p+jtA/\Delta)$。矩阵 $M\succeq I_p$ 上，
$-\log\det M$ 的 Hessian 满足

 D^2(-\log\det)(M)[H,H]
 =\tr(M^{-1}HM^{-1}H)\le\|H\|_{\rm HS}^2.
\]

$I_p+(t/\Delta)\sum_{i\le j}a(X_i)a(X_i)^T$
与其均值间作二阶 Taylor 展开，再用条件独立性，得到
\[
 0\le\overline H_j-H_j
 \le\frac{jt^2}{2\Delta^2}\E\|a(X)a(X)^T-A\|_{\rm HS}^2
 \le Cjt^2.
\]
对 $n>j\ge1$，逐特征值使用
$(1+nt\lambda/\Delta)/(1+jt\lambda/\Delta)\le n/j$，即有
\[
 H_n-H_j\le\overline H_n-\overline H_j+Cjt^2
 \le p\log(n/j)+Cjt^2.
\]
由 \eqref{alt:block-entropy} 的第一式、\eqref{alt:cross-entropy}

\end{aligned}\end{equation}

\subsection{输出和隐藏范数的任意固定阶矩}
\begin{lemma}\label{an:moments}
令 $q_\ell=\|h^{(\ell)}\|^2/k_\ell$。每个固定 $r>0$ 均有
\[
 \mathbb E q_\ell^r\le C_r,\qquad
 \mathbb E|q_\ell-1|^r\le C_r d^{-r/2},\qquad
 \mathbb E|F|^r\le C_r.
\]
这些结论适用于混合使用上述激活的固定层数网络。此外，每个固定 $r>0$ 有 $\sup_{\ell,i,d}\mathbb E|h_i^{(\ell)}|^r<\infty$；任意固定个特征的固定次幂乘积具有一致有界的绝对期望，允许这些特征来自不同层或不同输入副本，只须各副本具有规定的高斯边缘分布。
\end{lemma}
\begin{proof}
写 $b(q)=\E\sigma(\sqrt qG)$、$v(q)=\E\sigma(\sqrt qG)^2$。
Lipschitz 性、中心化及单位方差给出全局界
\begin{equation}\label{an:scalar}
 \E|\sigma(\sqrt qG)|^s\le C_s(1+q^{s/2}),\qquad
 |b(q)|+|v(q)-1|\le C|q-1|\quad(q\ge0).
\end{equation}
这里用 $|\sigma(z)|\le C(1+|z|)$ 及
$|\sqrt q-1|(\sqrt q+1)=|q-1|$。
若 $Y_i$ 给定 $\mathcal F$ 后独立且中心化，则
\begin{equation}\label{an:sample}
 \E\left[\left|N^{-1}\sum_iY_i\right|^{2m}\middle|\mathcal F\right]
 \le C_mN^{-m}\max_i\E[|Y_i|^{2m}\mid\mathcal F].
\end{equation}
证明只需展开 $2m$ 次幂：单次出现的指标使该项为零，其余项至多有 $m$ 个指标，
共 $O_m(N^m)$ 项，每项由 H\"older 控制。

给定前层，本层行输出独立同分布为 $\sigma(\sqrt{q_{\ell-1}}G)$。
先由 Jensen 与式~\eqref{an:scalar} 归纳得 $\E q_\ell^{2m}\le C_m$；
再对 $Y_i=(h_i^{(\ell)})^2-v_\ell(q_{\ell-1})$ 用式~\eqref{an:sample}，得到
\[
 \E|q_\ell-1|^{2m}\le C_md^{-m}+C_m\E|q_{\ell-1}-1|^{2m}=O(d^{-m}).
\]
初值 $q_0=d^{-1}\sum_iX_i^2$ 由同一展开给出。
真实全一读出保留其条件均值：
\[
 F=\sqrt{k_L}b_L(q_{L-1})+
 k_L^{-1/2}\sum_i\{h_i^{(L)}-b_L(q_{L-1})\}.
\]
第一项的 $2m$ 阶矩至多 $C_mk_L^m\E|q_{L-1}-1|^{2m}=O(1)$，
第二项由式~\eqref{an:sample} 及正矩界控制。
取 $2m\ge r$ 得全部所需阶数；单个特征矩由条件行估计给出，固定乘积矩由 H\"older 得到。
\end{proof}

\subsection{梯度与 Hessian 的矩界}
令 $A_\ell=W^{(\ell)}/\sqrt{k_{\ell-1}}$，$B_\ell=\|A_\ell\|_{\rm op}$。以后 $P$ 表示 $1+\|X\|/\sqrt d$ 与有限个 $B_\ell$ 的非负固定次数多项式；存在第二组独立高斯矩阵时也允许其归一化算子范数。所有这样的 $P$ 的每个固定阶矩均一致有界。为核实这一点，在 $\mathbb R^m,\mathbb R^r$ 单位球面各取至多 $9^m,9^r$ 个点的 $1/4$ 网。对高斯 $m\times r$ 矩阵 $G$，双线性形式的网逼近给出
\[
 \|G\|_{\rm op}\le2\max_{u,v\text{ 属于网}}|u^TGv|,
 \quad
 \mathbb P\{\|G\|_{\rm op}^2\ge8[\log2+(m+r)\log9+s]\}\le e^{-s}.
\]
积分此尾界并使用有界宽度比，即得所需的全部矩。

\begin{lemma}\label{an:derivatives}
对 Lipschitz 激活，弱梯度满足 $\mathbb E\|\nabla_WF\|^r\le C_r$，每个固定 $r>0$ 均成立。若各激活还属于 $C^2$ 且二阶导数有界，则
\[
 \mathbb E\|\nabla_W^2F\|_{\rm HS}^2\le C.
\]
特别地，归一化 tanh 和中心化归一化 sigmoid 满足这些结论。
\end{lemma}
\begin{proof}
前向递推给出 $\|h^{(\ell)}\|/\sqrt{k_\ell}\le P$。定义
\[
 \beta_L=k_L^{-1/2}\boldsymbol1,\quad
 \lambda_\ell=\operatorname{diag}(\sigma_\ell'(u^{(\ell)}))\beta_\ell,\quad
 \beta_{\ell-1}=A_\ell^T\lambda_\ell.
\]
由有界斜率，$\|\beta_\ell\|+\|\lambda_\ell\|\le P$，且
\[
 \nabla_{W^{(\ell)}}F=\lambda_\ell(h^{(\ell-1)})^T/\sqrt{k_{\ell-1}},
 \qquad \|\nabla_WF\|^2\le P. \tag{A3}\label{an:gradient}
\]
网络在每个有限维空间局部 Lipschitz；链式公式在可微点成立，并代表弱导数。亦可逐层平滑激活，用上述界和弱紧致性识别同一弱梯度。

记 $J_{u,\ell},J_{h,\ell}$ 为对全部权重的 Jacobian。
对前向递推求一次导数，逐层得二者算子范数均不超过 $P$。
二阶链式法则中，每层的激活项为
$J_{u,\ell}^T\operatorname{diag}(\beta_\ell\odot\sigma''_\ell)J_{u,\ell}$；
权重与前层特征的混合项及其转置各有 Hilbert--Schmidt 范数
$k_{\ell-1}^{-1/2}\|\lambda_\ell\|\|J_{h,\ell-1}\|_{\rm HS}$。
因此
\[
 \|\nabla_W^2F\|_{\rm HS}
 \le\sum_\ell\left(
 \|\sigma''_\ell\|_\infty\|\beta_\ell\|\|J_{u,\ell}\|_{\rm op}^2
 +\frac{2\|\lambda_\ell\|\|J_{h,\ell-1}\|_{\rm HS}}{\sqrt{k_{\ell-1}}}\right)
 \le P.
\]
末步用 $\operatorname{rank}J_{h,\ell-1}\le k_{\ell-1}$；取二阶矩即得结论。
\end{proof}
\subsection{分段线性激活的负阶矩与符号变化}
本小节取式~\eqref{eq:pl} 的激活，特别是 $b\ne0$；允许 $a=0$ 或 $a\pm b=0$。

\begin{lemma}\label{an:negative}
存在 $\eta_0,C_0>0$，使得对全部 $r\ge0$、$0<\eta\le\eta_0$，
\[
 \mathbb P\{\sigma(rG)^2\le\eta\}\le C_0\sqrt\eta.
\]
若 $Q=N^{-1}\sum_{i=1}^N\sigma(rG_i)^2$，则每个固定 $p>0$ 都存在 $N_0(p)$ 使
$\sup_{r\ge0,N\ge N_0(p)}\mathbb E Q^{-p}<\infty$。
\end{lemma}
\begin{proof}
置 $c=bc_0\ne0$。两半轴上 $\sigma(z)=(a\pm b)z-c$。
零斜率半轴在 $\sqrt\eta<|c|/2$ 时不贡献事件；其他半轴的小球集合是
长度 $O(\sqrt\eta)$、与零保持固定距离的区间。
而 $r^{-1}\phi(z/r)=|z|^{-1}v\phi(v)$，$v=|z|/r$，在这些区间上对所有 $r>0$ 一致有界。
$r=0$ 时事件为空，故小球界成立。
若 $Q\le\eta$，至少半数样本的平方不超过 $2\eta$；独立性与子集计数给
$\Pr(Q\le\eta)\le(C_1\eta^{1/4})^N$。
缩小 $\eta_0$ 使 $C_1\eta_0^{1/4}\le1/2$，积分尾概率得
\[
 \E Q^{-p}\le\eta_0^{-p}+
 \frac{p\eta_0^{-p}2^{-N}}{N/4-p}\qquad(N>4p).
\]
逐层条件化得到隐藏范数及删去一行后的平均平方范数的全部固定阶负矩。
输入端对整数 $p$ 有 $\E q_0^{-p}=\prod_{j=1}^p d/(d-2j)\le2^p$（$d\ge4p$）；
其他阶数用矩单调性。
\end{proof}

\begin{lemma}[带权符号变化估计]\label{an:weighted-sign}
令 $W'=\sqrt{1-t}W+\sqrt tZ$，其中 $Z$ 为与 $(W,X)$ 独立的标准高斯矩阵族，
$0<t\le1/2$，撇号表示第二个网络。
固定一层 $\ell$，记 $D_\ell=\operatorname{diag}(\sigma'(u^{(\ell)}))$。
给定两网络的前层变量，令本层对应预激活的夹角为 $\omega_\ell$，
$q=q_{\ell-1}$。对前述任意范数多项式 $P$，存在只依赖前层的范数多项式 $P_<$，使
\[
 \E[P\|(D'_\ell-D_\ell)\beta_\ell\|^2\mid\text{前层}]
 \le C_P P_<(1+q)\omega_\ell.
\]
\end{lemma}
\begin{proof}
逐单项把 $P$ 分为前层、本层和下游因子。给定本层输出 $h$，
下游加权二阶矩 $M(h)=\E[P_>\beta_\ell\beta_\ell^T\mid h]$ 满足
\begin{equation}\label{an:covariance}
 M(h)_{ii}\le C\left(k_\ell^{-1}+h_i^2/\|h\|^2\right).
\end{equation}
事实上，同时右旋转下一层的 $W,Z$ 保持联合分布及范数，故 $M$ 正交等变。
固定 $h$ 的正交变换使 $M=c_\perp(I-hh^T/\|h\|^2)+c_\parallel hh^T/\|h\|^2$；
反向范数界给 $\tr M\le C$，即得上式。终层直接用 $\beta_{L,i}^2=1/k_L$。

本层同时删去 $W,Z$ 的第 $i$ 行。算子范数不超过删行范数与该行归一化长度之和，
所以本层因子至多为 $P_{\rm del}Q_i$，其中
$Q_i\le C(1+\|(W_i,Z_i)\|/\sqrt{k_{\ell-1}})^D$，$D$ 固定。
又 $\|h\|^{-2}\le(\sum_{j\ne i}h_j^2)^{-1}$。
给定前层，删行变量独立于第 $i$ 行；引理~\ref{an:negative} 及矩界给
\begin{equation}\label{an:rowdelete}
 \E[P_{\rm del}\mid\text{前层}]\le C,\qquad
 \E\left[\frac{P_{\rm del}}{\sum_{j\ne i}h_j^2}\middle|\text{前层}\right]
 \le C/k_\ell.
\end{equation}
第二式由 Cauchy--Schwarz 和删行平均平方范数的二阶负矩得到。

第 $i$ 行两个预激活可写为
$u=\sqrt qG$、$u'=\sqrt{q'}(\cos\omega_\ell G+\sin\omega_\ell H)$。
在联合行空间中先取这两个方向张成的平面；符号不同对应总角度 $2\omega_\ell$ 的扇区。
上面的 $Q_i$ 上界为径向函数。极坐标积分及正交补上的高斯矩遂给
\begin{equation}\label{an:angular}
 \E[Q_i\mathbf1_{uu'<0}\mid\text{前层}]\le C\omega_\ell,\qquad
 \E[Q_i h_i^2\mathbf1_{uu'<0}\mid\text{前层}]
 \le C(1+q)\omega_\ell.
\end{equation}
后一式使用 $h_i^2\le C(1+qG^2)$，再用 $G^2\le G^2+H^2$ 保持径向上界。
由于 $|\sigma'(u')-\sigma'(u)|^2=4b^2\mathbf1_{uu'<0}$，
合并上述三式并对 $i$ 求和即得结论。这里保留了反向导数与本层符号的依赖。
\end{proof}

\subsection{分段线性网络的梯度变化}
\begin{lemma}\label{an:gradient-ou}
对同一 OU 耦合及任意范数多项式 $P$，分段线性支满足
\[
 \E[P\|\nabla F(W',X)-\nabla F(W,X)\|^2]\le C_P\sqrt t.
\]
常数和维数阈值不依赖 $0<t\le1/2$。
\end{lemma}
\begin{proof}
归一化矩阵差及 Lipschitz 前向递推给出逐点界
\[
 \|A'_\ell-A_\ell\|_{\rm op}\le\sqrt tP,\qquad
 \|h'^{(\ell)}-h^{(\ell)}\|/\sqrt{k_\ell}\le\sqrt tP.
\]
给定前层 $h,h'$，本层预激活的相关系数为
$\sqrt{1-t}\langle h,h'\rangle/(\|h\|\|h'\|)$，故
\[
 \omega_\ell\le C\left(\sqrt t+
 \frac{\|h-h'\|}{\sqrt{\|h\|\|h'\|}}\right),\qquad
 \E[P_<\omega_\ell]\le C_{P_<}\sqrt t.
\]
末式由前向差界、两边缘的负阶矩和 H\"older 得到；第一层 $h=h'=X$ 直接成立。
引理~\ref{an:weighted-sign} 因而给出
\begin{equation}\label{an:pairedgate}
 \E[P\|(D'_\ell-D_\ell)\beta_\ell\|^2]\le C_P\sqrt t.
\end{equation}
从相同读出开始，使用精确递推
\[
 \lambda'_\ell-\lambda_\ell
 =D'_\ell(\beta'_\ell-\beta_\ell)+(D'_\ell-D_\ell)\beta_\ell,
 \qquad
 \beta'_{\ell-1}-\beta_{\ell-1}
 =A_\ell^T(\lambda'_\ell-\lambda_\ell)+(A'_\ell-A_\ell)^T\lambda'_\ell.
\]
固定深度反向归纳给
$\E[P(\|\beta'_\ell-\beta_\ell\|^2+\|\lambda'_\ell-\lambda_\ell\|^2)]\le C_P\sqrt t$：
乘上算子范数只改变允许的 $P$，矩阵差项的平方为 $O(t)$。
最后用梯度块公式~\eqref{an:gradient} 作乘积差展开，结合前向差界即可。
\end{proof}

\subsection{OU 梯度变化与余项}
\begin{lemma}\label{an:ou}
令 $\delta=1$（光滑支）或 $\delta=1/2$（分段线性支）。
对 $W(t)=\sqrt{1-t}W+\sqrt tZ$、$0<t\le1/2$，有
\[
 G(t):=\E\|\nabla F(W(t),X)-\nabla F(W,X)\|^2\le Ct^\delta,
\]
\[
 R_t=F(W(t),X)-F(W,X)-\sqrt t\,DF(W,X)[Z],\qquad
 \E|R_t|^2\le Ct^{1+\delta}.
\]
常数和维数阈值不依赖 $t$。
\end{lemma}
\begin{proof}
记 $c_N$ 为对 $X$ 平均后的第 $N$ 阶参数 Hermite 能量，$a_t=\sqrt{1-t}$。
正交性和 $\partial_iH_\nu=\sqrt{\nu_i}H_{\nu-e_i}$ 给
\[
 G(t)=2\sum_{N\ge1}N(1-a_t^{N-1})c_N\le Ct^\delta.
\]
光滑支用 $1-a_t^{N-1}\le Ct(N-1)$ 及
$\sum_NN(N-1)c_N=\E\|\nabla^2F\|_{\rm HS}^2\le C$；
分段线性支用引理 \ref{an:gradient-ou}。
取 $t=1-e^{-2\tau}$ 即得 \eqref{eq:gradient-stability}；
$t>1/2$ 时由梯度二阶矩增大常数即可。
同一正交展开（交叉项对 $Z$ 分部积分）给
\[
 \E|R_t|^2=\sum_{N\ge1}\{2(1-a_t^N)+tN-2tNa_t^{N-1}\}c_N.
\]
当 $N\ge2$ 时 $a_t^N\ge1-Nt/2$；当 $N=1$ 时大括号等于
$(1-a_t)^2\le t^2$。因此
\begin{equation}\label{an:hermite}
 \E|R_t|^2\le t^2\E\|\nabla F\|^2+tG(t)\le Ct^{1+\delta}.
\end{equation}
计算先对有限 Hermite 和成立，再在高斯 Sobolev 范数下取极限。
\end{proof}

\subsection{高斯分部积分的定义域}
\begin{lemma}\label{an:sobolev}
设 $g:\mathbb R^p\to\mathbb R^p$ 属于高斯 $W^{1,2}$，也允许值域带独立的外部 $L^2$ 变量。定义
$\delta g=\operatorname{div}g-W\cdot g$。则其闭扩张满足
\[
 \mathbb E(\delta g)^2
 =\mathbb E\|g\|^2+\mathbb E\sum_{i,j}(\partial_i g_j)(\partial_jg_i)
 \le\mathbb E\{\|g\|^2+\|Dg\|_{\rm HS}^2\}. \tag{A13}\label{an:divergence}
\]
\end{lemma}
\begin{proof}
先取紧支撑光滑场。对高斯密度逐坐标分部积分两次，利用
$\partial_i(W_jg_j)=\boldsymbol1_{i=j}g_j+W_j\partial_i g_j$，交叉项合并后正好得到等式；导数和中每项用 $2uv\le u^2+v^2$ 给出不等式。一般情形先用径向截断 $\chi_R$，其中 $0\le\chi_R\le1$、球内等于一、$\|\nabla\chi_R\|_\infty\le C/R$，再在紧集上卷积。高斯密度在紧集上光滑且有正上下界，因此普通卷积逼近同时给出高斯 $W^{1,2}$ 逼近。尾部由 $g,Dg$ 的平方可积性控制，截断导数项由 $C\|g\|/R$ 控制。故有紧支撑光滑场序列在该 Sobolev 范数收敛；不等式保证其散度在 $L^2$ 中 Cauchy，并识别为分布散度。等式也由该收敛保留。对外部变量先用简单函数逼近并积分，证明不变。
\end{proof}

上述散度不等式用于附录 \ref{app:information} 的正则化测试场。
该场只依赖参数和独立测试输入；训练及辅助高斯观测的得分项另行处理。

\section{对称张量的 Gram 矩阵界}\label{tg:section}
固定整数 $m\ge5$、$q\ge1$。设单位行 $w_1,\ldots,w_k$ 满足
$\max_{i\ne j}|\langle w_i,w_j\rangle|\le\delta$。对多重指标 $r\in\mathbb N^k$、$|r|=q$，定义
\[
 P_r=\operatorname{Sym}\bigotimes_jw_j^{\otimes mr_j},\qquad
 Z_{r,i}=\operatorname{Sym}\bigotimes_jw_j^{\otimes(mr_j-\mathbf1_{j=i})},\quad r_i>0.
\]
$\operatorname{Sym}$ 是全部槽位排列的平均，范数取继承的 Hilbert 张量范数。
令 $\mathcal F$、$\mathcal F^-$ 分别以 $P_r$、$Z_{r,i}$ 为列。则
\begin{equation}\label{tg:frame}
 cI\preceq\mathcal F^*\mathcal F\preceq CI,\qquad
 cI\preceq(\mathcal F^-)^*\mathcal F^-\preceq CI,
\end{equation}
只要 $\delta+\sum_{h=1}^q k^h\delta^{mh-2}=o(1)$；$c,C>0$ 只依赖 $m,q$。

证明使用精确配对公式
\[
 \left\langle\operatorname{Sym}\bigotimes_{a=1}^Ru_a,
 \operatorname{Sym}\bigotimes_{a=1}^Rv_a\right\rangle
 =\frac1{R!}\sum_{\pi\in\mathfrak S_R}\prod_a\langle u_a,v_{\pi(a)}\rangle.
\]
若列的标签重数为 $(n_j)$，对角中的无错配项为 $\prod_jn_j!/R!$，
它有只依赖固定总次数的正下界；其余对角项为 $O(\delta)$。
纯列的重数为 $mr$，删除列的重数为 $mr-e_i$。两族内部不同列具有不同重数，
后者因模 $m$ 的余数唯一确定 $i$。同一有限支撑内的非对角项至少含一个错配，
故为 $O(\delta)$，此类列的数量有固定上界。
若另一列含 $h\ge1$ 个新标签，它至少有 $mh-1$ 个槽必须与不同标签配对，
故内积至多 $C\delta^{mh-2}$；这样的列至多 $Ck^h$ 个。
因此每个 Gram 矩阵与上述正对角矩阵之差的绝对行和至多
$C(\delta+\sum_hk^h\delta^{mh-2})=o(1)$。对称矩阵的 Schur 界即得 \eqref{tg:frame}。

行长在固定正区间时，各列只乘有固定上下界的标量，结论不变。
对按输入宽度归一化的高斯行，行长一致趋于一且
$\delta=O(\sqrt{\log d/d})$、$k\asymp d$，故扰动趋零；事件补集概率可取任意固定负幂。

\section{计算树展开的张量估计}
\label{fp:core-section}

本节估计计算树展开所产生的系数张量。所有空间取实 Euclidean 空间及其 Hilbert
张量积；$\|\cdot\|_{\mathrm{HS}}$ 表示张量 Hilbert 范数，
$\operatorname{Flat}_{S|S^c}$ 表示按索引位置分组的矩阵化。
下文的一个\emph{模式}是系数张量的一个索引位置；不同模式可以取相同数值，
也可以由同一个原节点索引复制得到。各层宽度在 $[cd,Cd]$ 内，
深度、多项式次数、涉及的层数和操作次数均固定。
记 $\delta_d=L_d/\sqrt d$、$\rho_d=L_d/d$，其中 $L_d$ 是可随有限次
运算增大的固定对数幂。概率估计分别在每个网络的高斯边缘下建立；
相应事件成立后，以下估计对两网络间任意耦合均成立。

\subsection{系数张量的范数界与线性求和}
\label{fp:core-invariant-section}

\paragraph{计算树的系数张量。}
从原网络的归一化读出向量开始，按原计算树自上而下展开。
每个当前模式代表一个原节点的指标，模式 $p$ 的一次展开为
\begin{equation}
 J e_a=\bigotimes_{t=1}^{q}u_a^{(t)},\qquad q\geq1,
 \qquad \|U_t\|_{\mathrm{op}}\leq C,
 \quad U_t=(u_a^{(t)})_a .
 \label{fp:core-replacement}
\end{equation}
因子 $e_a$ 表示保留父索引。也可在某节点停止展开，保留该节点以下的整个张量映射。
同一父索引产生的多个因子仍共享该索引。依赖两网络的映射只能作用在停止展开的叶节点上，
每个映射输出一个张量因子；输出多个因子的非线性映射须先按原计算树展开并纳入系数张量。

\refstepcounter{lemma}
\paragraph{引理~\thelemma（展开算子的范数）。}
把上述展开作用于全部保留因子的复合映射称为展开算子。任意固定次数的此类复合均有界。具体地，令
$\Delta_q e_a=e_a^{\otimes q}$，则
$J=(\bigotimes_tU_t^*)\Delta_q$，而 $\Delta_q$ 为等距嵌入，故
$\|J\|\leq\prod_t\|U_t\|$。
同样，若 $F_j:e_a\mapsto f_a^{(j)}$ 有界，则共享指标的乘积特征
$e_a\mapsto\bigotimes_jf_a^{(j)}$ 有界。
也可从 Gram 矩阵看出：其 Gram 为 $\mathop{\circ}_jF_j^*F_j$，
且对半正定矩阵 $X\preceq bI,Y\succeq0$，
$X\circ Y\preceq b\operatorname{diag}(Y)$。

\refstepcounter{lemma}
\paragraph{引理~\thelemma（展开保持范数和切片界）。}
\label{fp:core-closure}
设 $E$ 有 $r$ 个模式，满足
\begin{equation}
 \|E\|_{\mathrm{HS}}\leq C,\qquad
 \max_{p,a}\|E(e_a\text{ 于模式 }p,\cdot)\|_{\mathrm{HS}}\leq\delta_d,
 \qquad
 \|\operatorname{Flat}_{S|S^c}E\|_{\mathrm{op}}\leq\delta_d
 \quad(\varnothing\ne S\ne[r]).
 \label{fp:core-invariants}
\end{equation}
除 $r=q=1$ 外，一次展开保持这些结论，常数可增大固定倍数。

\emph{证明。}
Hilbert 范数由展开算子有界立即得到。考虑新核的一个非平凡分割。
若新产生的整个块在同一侧，该分割就是旧核的非平凡分割两侧复合有界算子。
若新块跨越两侧，在两侧暂时保留正交父指标 $a$。中央算子按 $a$ 分块对角，
每块为旧切片 $E(e_a,\cdot)$ 的某个矩阵化，算子范数至多其 Hilbert 范数
$\delta_d$。再施加两侧的行映射即得结论，包含旧核只有一个模式的情形。

除 $r=q=1$ 外，新张量至少有两个索引；其坐标切片满足
$\|E_{\rm new}(e_a\text{ 于 }p,\cdot)\|_{\rm HS}
\le\|\operatorname{Flat}_{\{p\}|\{p\}^c}E_{\rm new}\|_{\rm op}\le C\delta_d$，
故已由刚证明的矩阵化界控制。$r=q=1$ 时没有非平凡矩阵化，仍须另用
高斯连续乘积的坐标估计。
\hfill$\square$

\paragraph{线性路径上的高斯矩阵乘积。}
令 $A_\ell$ 为独立、按输入宽度平方根归一化的高斯矩阵。
任意固定连续积 $P=A_b\cdots A_a$ 满足
\begin{equation}
 \|P\|_{\mathrm{op}}\leq C,\qquad
 \max_{ij}|P_{ij}|\leq C_M\sqrt{\log d/d}
 \label{fp:core-product-event}
\end{equation}
且可使失败概率为 $O(d^{-M})$。算子范数由各层相乘得到；条件于
$Q=A_{b-1}\cdots A_a$ 后，$(A_bQ)_{ij}$ 是方差
$\|Qe_j\|^2/n_{b-1}\leq C/d$ 的高斯变量，尾界与有限并合即得第二式。
读出向量乘连续高斯积的坐标最大值也由条件于最后尚未暴露的
高斯因子得到相同估计。于是首次分叉以前的 $r=q=1$ 链可单独处理；
首次分叉以后模式数不减，前一引理处理所有后续步骤。
停止展开或施加恒等映射无须概率估计。有限原树核全部满足
\eqref{fp:core-invariants}。这里的一元链仅由原网络矩阵组成。

\paragraph{外部映射及重建。}
对系数张量的每个索引施加有界、输出一个因子的映射，保持所有非平凡矩阵化界，
但不必保持坐标切片界。完成一次收缩后，应从原计算树重新构造系数张量：
在对应父节点处停止展开，将已算出的跨网络收缩矩阵作为该叶节点的外部映射。
不同深度均可停止展开，但根到叶的内部路径仍只含原网络矩阵。
以下坐标切片估计始终用于这样重建的原系数张量。

\refstepcounter{lemma}
\paragraph{引理~\thelemma（线性路径的精确求和）。}
\label{fp:core-linear}
令底层行向量为 $b_j$，父层矩阵为 $U=(u_{aj})$，并设
$z_a=\sum_ju_{aj}b_j$。一个父节点有 $t$ 个线性子槽以及次数
$q_1,\ldots,q_h\geq2$ 的非线性子槽，则其未对称化张量严格等于
\begin{equation}
 z_a^{\otimes t}\otimes
 \bigotimes_{\ell=1}^h\left(\sum_j u_{aj}b_j^{\otimes q_\ell}\right).
 \label{fp:core-linear-identity}
\end{equation}
\emph{证明。} 给每个有序子槽分配独立求和指标，再逐个使用多线性性；
线性槽的求和正好是 $z_a$，非线性槽保留原式。固定次数的对称化为槽排列的
平均，可在求和后施加。纯一元链因此是归一化矩阵的普通乘法，无额外
$\sqrt{\text{宽度}}$ 因子。式中的所有非线性因子仍共享 $a$，不得替换为独立阵列。
坐标掩码一般不与此求和交换；必须先在全部张量因子的展开算子中证明掩码删除的误差为
$o(1)$，再恢复原张量并用本等式。\hfill$\square$

\subsection{同层索引的旋转与双索引收缩}
\label{fp:core-fresh-section}

\paragraph{概率工具。}
若确定性矩阵 $A_j:\mathbb R^u\to\mathbb R^v$ 满足
$\max\{\|\sum_jA_j^*A_j\|,\|\sum_jA_jA_j^*\|\}\leq s^2$，则独立
标准高斯系数满足
\[
 \mathbb P\!\left(\left\|\sum_jg_jA_j\right\|>t\right)
 \leq2(u+v)\exp[-t^2/(2s^2)].
\]
这是有限维矩阵高斯级数界。其自伴膨胀为
$H_j=\left(\begin{smallmatrix}0&A_j^*\\ A_j&0\end{smallmatrix}\right)$，
$\|\sum_jH_j^2\|\leq s^2$；矩阵高斯 Laplace 界
$\mathbb E\operatorname{tr}\exp(\lambda_{\rm c}\sum_jg_jH_j)
\leq\operatorname{tr}\exp(\lambda_{\rm c}^2\sum_jH_j^2/2)$
配合 $\lambda_{\rm c}=t/s^2$ 得到上尾，对负号重复即得双侧界。
以下只用其维数为固定 $d$ 次幂时的 $O(s\sqrt{\log d})$ 结论。

\refstepcounter{lemma}
\paragraph{引理~\thelemma（同层旋转下的坐标切片）。}
\label{fp:core-fresh}
在同一原网络层，同时展开所有尚未停止的模式。选择当前高斯层矩阵的列索引所对应的模式 $p$：
它随该层矩阵右乘正交矩阵而旋转，不包括保留的父索引 $e_a$ 或其确定性复制。
再选不同模式 $q$；总模式数为 $N\ge3$。令 $E$ 为尚未施加外部映射的系数张量。
以任意高多项式概率有
\begin{equation}
 \max_j\sup_{\|v\|\leq1}
 \|E(e_j\text{ 于 }p,v\text{ 于 }q,\cdot)\|_{\mathrm{HS}}
 \leq\rho_d.
 \label{fp:core-fresh-slice}
\end{equation}
此时尚未展开更低层；在较高层停止展开的索引保持不动。

\emph{证明。}
条件于一个满足 \eqref{fp:core-invariants} 的核，置
$A_jv=E(e_j,v,\cdot)$。直接按矩阵元素求和可知
\[
 \left\|\sum_jA_j^*A_j\right\|
 =\|\operatorname{Flat}_{\{q\}|\{q\}^c}E\|^2\leq\delta_d^2,
 \qquad
 \left\|\sum_jA_jA_j^*\right\|
 =\|\operatorname{Flat}_{\{p,q\}|\{p,q\}^c}E\|^2\leq\delta_d^2.
\]
第二分割非平凡恰好使用 $N\geq3$。前述级数界、
$\|g\|\geq c\sqrt d$ 的高斯尾界，给均匀球面向量 $u=g/\|g\|$
的收缩范数至多 $C_M\delta_d\sqrt{\log d/d}$。
对独立 Haar 矩阵的各列作并合，不要求列独立。
将当前高斯层矩阵右乘该正交矩阵不改变分布，并同时旋转对应其列索引的全部模式。
未固定模式上的旋转是等距算子；若 $q$ 也对应该层的列索引，对 $v$ 的上确界仍不变。
故 Haar 后的估计转回原坐标即为所需式。所有核和模式只有固定有限种。
\hfill$\square$

\refstepcounter{lemma}
\paragraph{推论~\thelemma（重复坐标项的范数）。}
\label{fp:core-collision}
若至少两个模式对应同一高斯层矩阵的列索引，且总模式数至少三，则其中至少两个索引取相同值的部分满足
\begin{equation}
 \|E_{\mathrm{coll}}\|_{\mathrm{HS}}^2
 \leq\sum_{p<q}\sum_j\|E(e_j\text{ 于 }p,e_j\text{ 于 }q,\cdot)\|^2
 \leq C d\rho_d^2=o(1).
 \label{fp:core-collision-bound}
\end{equation}
在该层引入这些模式时先估计这一部分，再将误差通过有界的展开算子传递，最终误差仍为 $o(1)$。
这里只删除不同原节点的索引恰取相同值的项；同一父索引的确定性复制仍须保留。

\refstepcounter{lemma}
\paragraph{推论~\thelemma（原始系数的矩与坐标界）。}
\label{fp:core-raw-collision-expectation}
设 $C_R$ 为固定深度、固定次数、零常数高斯多项式网络的原始 $R$ 阶系数，
允许保留有序因子、对称化及固定有限个展开项的和。对每个固定 $0<p<\infty$，
$\sup_d\mathbb E\|C_R\|_{\rm HS}^p<\infty$。
若全部索引由同一原高斯层的列索引产生（不含父索引的确定性复制），宽度为 $k\asymp d$，则对任意固定 $M$，
以失败概率 $O(d^{-M})$ 同时有
\begin{equation}\label{fp:core-coordinate-bounds}
 \max_{i_1,\ldots,i_R\ \text{互异}}|(C_R)_{i_1\cdots i_R}|
 \le L_d k^{-R/2},\qquad
 \|\operatorname{Coll}_R C_R\|_{\rm HS}^2\le L_d/d\quad(R\ge3),
\end{equation}
其中 $L_d$ 为固定对数幂；$R\ge3$ 时还有 $\mathbb E\|\operatorname{Coll}_R C_R\|_{\rm HS}^2=o(1)$。
为证矩界，将原展开算子的范数相乘，再用归一化高斯矩阵算子范数的固定阶矩和 H\"older 不等式，
不需乘积中各因子独立。列置换不变性使互异坐标的二阶矩相同，故每个至多
$C/(k)_R=O(k^{-R})$。每个坐标是固定次数 $D_0$ 的高斯多项式，
$\|Z\|_u\le(u-1)^{D_0/2}\|Z\|_2$；取 $u=C_M\log d$，在 $e\|Z\|_u$ 处用 Markov
并对至多 $k^R$ 个坐标作并集界，即得第一界。
第二界由同层旋转引理~\ref{fp:core-fresh}及式~\eqref{fp:core-collision-bound}应用于每个有序展开项得到；
固定有限个排列与展开项的和不改变阶数。矩界及 Cauchy--Schwarz 给
$\mathbb E\|\operatorname{Coll}_R C_R\|^2\le L_d/d+
(\mathbb E\|C_R\|^4)^{1/2}O(d^{-M/2})=o(1)$。
这里只控制原始系数，尚未施加依赖两个网络的外部映射。

\refstepcounter{lemma}
\paragraph{引理~\thelemma（两个父索引的收缩与二阶非对角项）。}
\label{fp:core-two-parent}
在一个未加掩码的原始切口中，设不同父模式 $p,q$ 的所选线性槽与教师行
收缩后给出 $\eta_i,\zeta_i$，其 $\ell^2$ 范数有界。模式 $p$ 是本层引入的模式，
且删去所选槽后仍有非空残余；所有完整残余家族均为有界算子。
若总模式数 $R\ge3$，完整剩余源张量 $S_i$ 满足 $\|S_i\|_{\rm HS}\le C\rho_d$。
事实上，令 $B_ie_b=\zeta_{ib}f_b$，其中 $f_b$ 是第二个源索引删除所选槽后的完整
剩余特征；无剩余槽时取 $f_b=1$。逐点有
$\|B_i\|_{\rm HS}^2=\sum_b|\zeta_{ib}|^2\|f_b\|^2\le C$。
将原始切片 $E(e_a,\cdot)$ 看作从 $q$ 索引到其余索引的算子 $T_a$。
式~\eqref{fp:core-fresh-slice}给 $\|T_a\|_{\rm op}\le\rho_d$，故
$\|T_aB_i^*\|_{\rm HS}\le C\rho_d$。
保留第一个索引 $a$，乘 $\eta_{ia}$ 后平方求和，再施加完整剩余特征，即得所需界。
这些估计逐点成立，不要求 $B_i$ 与网络独立。

若 $R=2$、两个索引属于同一原网络层，且删除所选槽后两个模式均有非空残余，式~\eqref{fp:core-coordinate-bounds}给
$\max_{a\ne b}|D_{ab}|\le L_d/d$；保留两标签后施加完整特征映射，非对角部分满足
\[
 \|S_i^{\rm off}\|\le C\Big(\sum_{a\ne b}|D_{ab}\eta_{ia}\zeta_{ib}|^2\Big)^{1/2}
 \le CL_d/d.
\]
两种界与教师切片和 $C\sqrt d\|C\|$ 配对均为 $o(1)\|C\|$。
该二阶结论不控制对角或坐标--向量切片；不同层的两个索引亦不适用，须保留根索引另行处理。
同一父索引的确定性复制不是两个不同父节点，不能使用本估计。

\subsection{保留剩余因子的局部收缩估计}
\label{fp:core-residual-section}

\refstepcounter{lemma}
\paragraph{引理~\thelemma（未收缩因子的矩阵化界）。}\label{fp:core-residual-flat}
设 $F_a=\sum_bU_{ab}Q_b$，其中 $U$ 是原高斯层或固定连续积，
$Q_b$ 为正次数计算树展开项；不同高度的停止展开的叶节点允许任意有界自适应单槽映射。
完整家族 $e_a\mapsto F_a$ 有界。若 $F_a$ 至少有两个张量因子位置，则其每个
非平凡对输出因子的矩阵化至多 $C\delta_d$，同时对所有 $a$ 成立：以
$U_{a,:}$ 为读出重建原系数张量，式~\eqref{fp:core-product-event} 给读出的范数与
最大坐标，展开保持范数和切片界的引理给矩阵化界，最后的单槽映射保持这些界。
家族有界性来自展开算子与 $U^*$ 的复合。单槽收缩因子同样是有界家族，
不需要非平凡矩阵化估计。同指标乘积家族由对角嵌入保持有界。
以上不对自适应叶映射作独立性或坐标切片保持性假设。

\begin{lemma}[完整余张量的局部收缩]\label{fp:core-local-contraction}
\label{fp:core-split-hit}\label{fp:core-hybrid-Rge2}\label{fp:core-hybrid-R1}
固定当前教师节点，并选定与其配对的一组共享源父索引的因子，下文称这组因子为一个源列。所有尚未收缩的因子均保留。
教师叶含 $m\ge2$ 个张量因子位置。教师行矩阵和源完整家族有界；其他源列按各自标签分别施加展开算子，且不额外依赖所选列标签。教师系数为 $C$，源系数为 $D$（一模时记为 $d$）。
设该叶接触源列中的一个收缩因子 $\ell_a$，其交叉矩阵
$G_{ia}=\langle a_i,\ell_a\rangle$ 满足
$\max|G|\le\kappa_d$ 及 $\sup_i\|G_{i,:}\|_2\le C$。
该列的其他因子为上段的原始残余或单槽收缩因子。
则有下列三个完整配对界：
\begin{enumerate}
\item
若当前教师节点还与另一个源父节点的因子收缩，则界为
$C\kappa_d\|C\|\|D\|$。
\item
若本叶全部位于该列，重建源核心有 $R\ge2$ 个模式，且
$\|\operatorname{Flat}_{1|\mathrm{rest}}D\|\le\delta_d$，则界为
$C\sqrt d\,\delta_d(\delta_d+\kappa_d)\|C\|$。
\item
若本叶全部位于该列，核心只有一个模式 $d_a$，
$\|d\|_\infty\le\kappa_d$，且除指定收缩因子外至少有两个完整因子，
则界为 $C\sqrt d\,\kappa_d(\delta_d+\kappa_d)\|C\|$。
\end{enumerate}
各界在 $\delta_d,\kappa_d\le L_d/\sqrt d$ 时均为 $o(1)$。
一模仅一个残余因子不在本引理第三项内。
\end{lemma}
\begin{proof}
第一项先保留源索引 $a$，从其他源父节点的因子中取出与当前教师节点配对的全部 $t\ge1$ 个因子。
映射 $x\mapsto(\langle a_i^{\otimes t},x\rangle)_i$ 有界，因为其伴随
为 $(A^*)^{\otimes t}\Delta_t$。保留所得正交标签 $i$ 后，乘以
$G_{ia}$ 的范数至多 $\kappa_d$。随后在每个 $i$ 块内提升本列剩余家族，
收缩本叶余槽，并分析其余教师叶。这是从完整源系数到完整教师系数的
有界算子复合，故没有逐 $i$ 求和或部分迹损失。

以下考虑本叶全部位于该列。固定 $i$，按本叶如何收缩各剩余因子，分为
完全收缩、部分收缩、完全保留。完全收缩的因子产生 $\ell^2$ 有界向量 $\eta$；
部分收缩的因子产生逐列范数至多 $C\delta_d$ 的有界家族；完全保留的因子族仍有界。
记本列收缩后算子为 $B_i$。若存在部分收缩的因子，则
$\|B_i\|_{\rm HS}^2\le C\delta_d^2\sum_a|G_{ia}|^2$。
否则至少一个因子完全收缩，故
$\|B_i\|_{\rm HS}^2\le C\kappa_d^2\sum_a|\eta_a|^2$。
于是 $\|B_i\|_{\rm HS}\le C(\delta_d+\kappa_d)$。
对第二项，其他源模式的展开算子记为 $T$，Hilbert--Schmidt 理想不等式给
\[
 \|(B_i\otimes T)D\|_{\rm HS}
 \le\|T\|\,\|B_i\|_{\rm HS}
       \|\operatorname{Flat}_{1|\mathrm{rest}}D\|_{\rm op}
 \le C\delta_d(\delta_d+\kappa_d).
\]

对第三项，记收缩后的家族为 $H_j$，完全收缩者取值于 $\mathbb R$。
若有部分收缩因子 $H_u$，其 Gram 对角元至多 $C\delta_d^2$；选任意
$j\ne u$，由 $H_j^*H_j\preceq CI$ 和 Schur 乘积半正定序，整个乘积
家族的算子范数至多 $C\delta_d$。因此
$\|S_i\|\le C\delta_d\|(d_aG_{ia})_a\|_2\le C\kappa_d\delta_d$。
若无部分收缩，至少一个因子完全收缩为 $\ell^2$ 有界向量 $\eta$；
将它移入系数后，$\|(d_aG_{ia}\eta_a)_a\|_2\le C\kappa_d^2$，
其余至少一个家族的乘积有界。两种情况均给
$\|S_i\|\le C\kappa_d(\delta_d+\kappa_d)$。
最后只需一次
$\sum_i\|C(e_i,\cdot)\|\le C\sqrt d\|C\|$，即得第二、三项。
\end{proof}

二阶源矩阵若先限制到非对角，可分别处理 $D$ 与 $\operatorname{diag}D$，
因为后者的算子范数及 Hilbert 范数不增；这里没有将其对角宣称为小量。
其他掩码不能据此处理。系数只有一个索引且仅剩一个其他因子时，须用第~\ref{fp:permanent-mark} 节保留根索引的论证。

\section{实际激活的输入混沌投影与多项式近似}
\label{fp:actual-section}\label{app:closure}

本节沿用模型与 $\Pi_N,I_N$ 的定义，令 $A_\ell=W^{(\ell)}/\sqrt{k_{\ell-1}}$、
$F_L=F_{\theta(W)}^{(L)}$、$u=c_0=\sqrt{2/\pi}$，并记
$b_j=\E[(\sigma(G)^2-1)\psi_j(G)]$。
目标是在单网络先验下，用有限个多项式网络的系数逼近 $\Pi_NF_L$。
下文始终先取形式 $N$ 次系数，再施加 $I_N$；这与先计算整个多项式网络再投影不同，
后者还包含更高形式次数经高斯收缩得到的第 $N$ 阶项。

\paragraph{局部矩估计。}
以下估计均只涉及单网络边缘分布。
这些结论分别由引理~\ref{an:moments}、条件行估计~\ref{an:scalar}
及~\ref{an:derivatives} 中的高斯矩阵网格尾界证明。
对固定层数、固定 $p<\infty$，有
\begin{equation}\label{fp:actual-local-interface}
 \|q_\ell-1\|_{L^p(W,X)}\leq C_p d^{-1/2},\quad
 \sup_{d,\ell,i}\mathbb E|h_i^{(\ell)}|^p<\infty,\quad
 \sup_d\mathbb E\|A_\ell\|_{\rm op}^p<\infty.
\end{equation}
同样的矩界适用于固定多个神经元输出的乘积，允许神经元索引重复。
从第一式和任意大的 $p$，$q_\ell\notin[1/2,2]$ 的概率可取任意多项式衰减。
首层单位方向 $v_i=A_{1,i:}/\|A_{1,i:}\|$、半径 $s_i=\|A_{1,i:}\|$
同时满足
\begin{equation}\label{fp:actual-row-event}
 \max_i|s_i-1|+\max_{i\ne j}|\langle v_i,v_j\rangle|
 \leq C_M\sqrt{\log d/d},\qquad \|A_1\|_{\rm op}\leq C_M
\end{equation}
的失败概率至多 $d^{-M}$。这些结论也只涉及单个网络先验。
证明分三步：截断条件 Hermite 展开，计算输出层的条件均值，再补回展开时删去的重复索引项。
最后一步使用的多项式系数估计将在下文列出。

\subsection{首层激活乘积的输入混沌投影}
\label{fp:actual-finite-support}
固定 $t,N$，设 $v_1,\ldots,v_t$ 为单位向量，$G=(\langle v_i,v_j\rangle)$，
$\|G-I\|\leq\eta$，$\max_i|s_i-1|\leq\delta$。则
\begin{equation}\label{fp:actual-ridge-projection}
 \left\|\Pi_N\prod_{i=1}^t\sigma(s_iv_i\cdot X)
 -\sum_{\substack{n_i\geq1\\\sum_i n_i=N}}
 \frac{\prod_i a_{n_i}}{\sqrt{\prod_i n_i!}}
 I_N\!\left(\operatorname{Sym}\bigotimes_i(s_iv_i)^{\otimes n_i}\right)
 \right\|_{L^2_X}\leq C_{t,N}(\eta+\delta).
\end{equation}
$t>N$ 时右侧的近似和为空。

证明如下。所有相关张量位于 $\operatorname{Sym}^N\operatorname{span}\{v_i\}$。
用 $n_i\geq0$、$\sum n_i=N$ 的
$B_n=I_N(\operatorname{Sym}\bigotimes_i v_i^{\otimes n_i})$ 作基。
其 Gram 矩阵是 $G$ 的多项式，且在 $G=I$ 附近一致可逆：从标准正交基
到 $v_i$ 的线性映射有一致有界的最大、最小奇异值，其固定对称张量幂也如此。
投影的系数向量为此 Gram 矩阵的逆乘
$\mathbb E[\prod_i\sigma(s_iZ_i) B_n(Z;G)]$，其中 $Z\sim N(0,G)$。
在 $G$ 的固定紧邻域内，对高斯密度关于协方差求导，得到一致可积的
多项式乘高斯密度；$B_n$ 的系数也多项式依赖于 $G$。
激活至多线性增长，故这些有限个积分关于 $G$ 局部 Lipschitz。
关于 $s_i$ 的 Lipschitz 性来自 $\sigma$ 的 Lipschitz 性和高斯加权矩。
在 $G=I,s_i=1$，独立性给出系数 $\prod_i a_{n_i}/\sqrt{\prod_i n_i!}$；
$a_0=0$ 去掉所有零次数。最后把 $v_i^{\otimes n_i}$ 改回
$(s_iv_i)^{\otimes n_i}$ 只产生 $O(\delta)$。有限维范数等价证明
\eqref{fp:actual-ridge-projection}。这个证明对完整非多项式乘积积分，
因此已包括高阶 Hermite 项和非正交方向引起的所有收缩项。

\subsection{条件 Hermite 展开的截断与误差}
\label{fp:actual-positive}
本小节先从网络输出中减去给定前层输出后的条件均值，记所得函数为 $F_+$；
条件均值将在下一小节单独处理。给定前层特征 $h\in\mathbb R^k$，$q=\|h\|^2/k$，当前行预激活为
$W_i\cdot h/\sqrt k$。对 $q>0$ 写
\[
 \sigma(\sqrt q G)=\sum_{r\geq0}c_r(q)\psi_r(G),\qquad
 c_r(q)=\mathbb E\sigma(\sqrt qG)\psi_r(G).
\]
多元 Hermite 加法公式为
\begin{equation}\label{fp:actual-multivariate}
 \psi_r(W_i\cdot h/\|h\|)=
 \sum_{|\alpha|=r}\sqrt{\frac{r!}{\alpha!}}
 (h/\|h\|)^\alpha\prod_j\psi_{\alpha_j}(W_{ij}).
\end{equation}
$q=0$ 按连续的实际随机变量取值，不引入 $q^{-1}$ 的矩。

对本节出现的有限函数集 $f\in\{\sigma,\chi=\sigma^2-1\}$，其中 $\chi$
只用作分段线性网络的输出层函数，均有 $|f(x)|\le C(1+|x|^2)$。
映射 $q\mapsto f(\sqrt q\,\cdot)$ 在 $[1/2,2]$ 上为 $L^2(\gamma)$ 连续，
故其 Hermite 投影尾在此紧区间上一致趋零。在区间外，投影的 $L^2$ 范数
不超过完整函数，条件二阶矩至多 $C(1+q^2)$；式~\eqref{fp:actual-local-interface}
给任意高阶的 $q$ 矩和区间外概率，从而区间外误差趋零。
固定多个不同神经元输出的乘积也可截断：条件于前层，相应权重行独立，逐因子写差后，
每项的条件平方期望等于一个尾的平方期望乘其余因子的条件二阶矩。
在紧区间上用一致尾界，区间外用上述多项式矩界即可，不需要单行尾的高阶
$L^p$ 收敛。从输出层向输入层依次选择截断阶数；每次所需处理的神经元数已经固定。

固定保留次数 $r$，令 $c_r^f(q)=\mathbb E f(\sqrt qG)\psi_r(G)$。
将积分变量换成 $y=\sqrt qG$ 后，可对高斯密度和
$\psi_r(y/\sqrt q)$ 求导；在 $[1/2,2]$ 上导数由可积高斯多项式控制。
故 $c_r^f(q)q^{-r/2}$ 局部 Lipschitz，可替换为 $c_r^f(1)$，误差由
$\|q-1\|_{L^p}=O(d^{-1/2})$ 控制。区间外，精确有限分量的条件固定阶矩
由高斯 Hermite 矩界控制，系数是 $c_r^f(q)$，没有逆方差；替换后的
固定多项式由实际特征矩控制。H\"older 不等式与任意高阶区间外概率使误差
趋零。此论证从不使用负阶 $q$ 矩，也不要求分段线性 $f$ 可微。

接着删去每层展开中重复使用前层神经元索引的项。此时关于当前权重的系数
仍是多元 Hermite 多项式，前层输出仍是实际激活函数。若一个展开项含 $T$ 个
前层输出因子，其中至少两个索引相同的赋值至多有 $C_Tk^{T-1}$ 个；
系数的平方归一化因子为 $k^{-T}$。由权重 Hermite 多项式的正交性和
前层输出乘积的矩界，这些项的总平方误差为 $O_T(k^{-1})$。
不同排序若对应同一组权重变量，先将它们合并；重数只依赖固定的 $T$。
剩余各项的每个权重变量只出现一次，故其 Hermite 多项式等于普通权重乘积。
分段线性激活的零阶系数 $c_0(q)=bu(\sqrt q-1)$ 在每个固定 $L^p$ 中为
$O(d^{-1/2})$。对输出层以下的固定乘积，上述乘积估计允许删去含零阶系数的项；
输出层的零阶项带有读出求和，须另行计算。

为说明这些局部误差不会被宽度放大，将某阶段的整个误差写为
\begin{equation}\label{fp:actual-phase}
 \Delta=\sum_g\beta_g\Psi_g(W_{>\ell}) E_g(W_{\leq\ell},X),
 \qquad \sum_g\beta_g^2\leq C_{\rm fixed}.
\end{equation}
这里 $\Psi_g$ 是上方已展开各层的权重单项式，每个权重变量的次数均为一；
相同单项式对应的项先合并。不同 $\Psi_g$ 在高斯权重下正交，
且均与 $E_g$ 所依赖的下层权重及输入独立。每个权重归一化因子为相应输入宽度的
$-1/2$ 次幂，输出读出因子为 $k_L^{-1/2}$。对输出索引和各个自由的下层索引求和，
这些因子的平方恰好抵消索引数目，只留下依赖固定展开结构和次数的常数。
这证明系数平方和有界，并给出
\begin{equation}\label{fp:actual-phase-bound}
 \mathbb E\|\Delta\|_{L^2_X}^2
 \leq C_{\rm fixed}\sup_g\mathbb E\|E_g\|_{L^2_X}^2.
\end{equation}
对应同一上层权重单项式的全部下层误差合并为一个 $E_g$；上述正交性仅用于
不同单项式之间。固定 $\ecut>0$ 后，从输出层向输入层依次选择有限截断，
得到每层神经元索引均互异的展开项之和 $F_{+,\ecut}$，且
\begin{equation}\label{fp:actual-finite-error}
 \limsup_d\mathbb E\|F_+-F_{+,\ecut}\|_{L^2_X}^2
 \leq C_L\ecut^2.
\end{equation}
以上截断、径向系数替换和重复索引项删除的误差，均在单网络边缘分布下估计。

首层保留实际激活函数，并应用\eqref{fp:actual-ridge-projection}。
对每个固定的首层神经元数，\eqref{fp:actual-row-event} 使其误差趋零；利用
\eqref{fp:actual-phase-bound} 对不同权重单项式求平方和，避免逐项使用三角不等式。
在首层好事件补集上，固定次数多项式矩及实际固定乘积矩至多有固定多项式维数增长，
可选足够大的失败指数使贡献趋零。
若某个展开节点的次数大于 $N$，它就包含超过 $N$ 棵不同子树。每棵子树
至少保留一个首层输出，且这些输出的索引互异。因此该项含超过 $N$ 个首层
激活因子，式~\eqref{fp:actual-ridge-projection} 的近似和为空。
所以各层截断阶数达到 $N$ 后，贡献于最终近似的项不再增加。
先固定各层截断并令 $d\to\infty$，再令 $\ecut\to0$，得到同一个有限多项式表达式。

\subsection{输出层条件均值与逐层递推}
\label{fp:actual-mean}\label{fp:actual-masked-formula}
固定 $N>2$，定义零常数多项式
\begin{equation}\label{fp:actual-PQ}
 P_N(z)=\sum_{j=1}^N\frac{a_j}{\sqrt{j!}}z^j,
 \qquad Q_N(z)=\sum_{j=1}^N\frac{b_j}{\sqrt{j!}}z^j.
\end{equation}
使用网络原有的矩阵 $A_1,\ldots,A_t$ 和读出 $k_t^{-1/2}\mathbf1$，
令输出层激活多项式为 $A$，下层依次为 $p_1,\ldots,p_{t-1}$。
取该网络的形式 $N$ 次对称系数并施加 $I_N$，所得函数记为 $R_{t,N}[A;p]$。
若在展开中只保留每个隐藏层的全部索引互异的项，再作同样操作，记为
$J_{t,N}[A;p]$。这里不限制输入坐标是否重复，故 $J_{1,N}=R_{1,N}$。
下文 $;P_N$ 表示所有下层均用 $P_N$。这些定义也适用于固定辅助变量的系数。

令 $F_t^f=k_t^{-1/2}\sum_i f((A_th^{(t-1)})_i)$，并令 $F_{t,+}^f$ 表示
从 $F_t^f$ 中减去给定前层输出后的条件均值。它恰好保留输出层权重的正阶
Hermite 项。由上述截断、权重单项式的正交性及截断达到 $N$ 后的稳定性，
\begin{equation}\label{fp:actual-positive-local}
 \Pi_NF_{t,+}^{f}=J_{t,N}[H_f;P_N]+o_{L^2(W,X)}(1),
 \qquad H_\sigma=P_N,\quad H_\chi=Q_N.
\end{equation}
这里 $\chi$ 只用于分段线性分支。深度一没有可用于上述正交估计的更高层权重，直接估计如下：
$f=\sigma,\chi$ 满足
$|f(sx)-f(tx)|\le C|s-t|(|x|+x^2)$（$s,t\in[1/2,2]$），故
$\mathbb E[f(sG)\psi_N(G)]$ 对 $s$ 局部 Lipschitz。以其 $s=1$ 值乘
$s^N$ 替换的逐行误差为 $O(\max_i|s_i-1|)$；映射
$e_i\mapsto v_i^{\otimes N}$ 有界，因为其 Gram 绝对行和至多
$\sum_j|\langle v_i,v_j\rangle|^2\le\|V\|_{\rm op}^2$。
归一化读出范数为一，故首层激活求和的误差也趋零；坏事件的贡献由固定矩和任意高阶概率界控制。

分段线性支有下列对所有 $q\ge0$ 成立的精确恒等式：
\begin{equation}\label{fp:actual-root-mean}
 \mathbb E f(\sqrt qG)=\lambda_f(q-1)+c_f(\sqrt q-1)^2,
 \quad
 (\lambda_\sigma,c_\sigma)=(bu/2,-bu/2),\qquad
 (\lambda_\chi,c_\chi)=(1,b^2u^2).
\end{equation}
由 $|\sqrt q-1|\le|q-1|$ 及四阶局部矩，乘 $\sqrt{k_t}$ 后的二次余项
为 $O_{L^2}(d^{-1/2})$。又 $F_{t-1}^\chi=\sqrt{k_{t-1}}(q_{t-1}-1)$，所以
\begin{equation}\label{fp:actual-root-recursion}
 \Pi_NF_t^f=J_{t,N}[H_f;P_N]
 +\lambda_f\sqrt{k_t/k_{t-1}}\,\Pi_NF_{t-1}^\chi+o_{L^2}(1),
 \qquad t\ge2.
\end{equation}
$t=1$ 时直接有 $\Pi_NF_1^f=R_{1,N}[H_f]+o_{L^2}(1)$：条件均值的线性项
为 $\lambda_f\sqrt{k_1}(q_0-1)$，它属于输入二阶混沌，故被 $N>2$ 投影
精确消去；二次余项仍为 $o(1)$。这也证明
\eqref{fp:actual-positive-local} 的深度一版本。奇分支所有条件均值为零，
只需 $f=\sigma$，递推中的第二项不存在。

\subsection{重复索引项与多项式近似公式}
\label{fp:actual-quadratic}\label{fp:actual-telescope}
以下对完整上部多项式使用推论~\ref{fp:core-raw-collision-expectation} 的
$r\ge3$ 重复索引估计及固定阶矩界；该推论不能直接用于受索引限制的 $J$。
二阶对角另作如下计算。

设输入宽度为 $k$ 的深度 $t$ 上部网络各层多项式为
$p_j(z)=\sum_{r\ge1}g_{j,r}z^r$。其全二次系数满足
\begin{equation}\label{fp:actual-diag}
 \mathbb E\left\|\operatorname{Diag}D_2-
 \lambda I_k/\sqrt k\right\|_{\rm HS}^2=O(k^{-1}),\qquad
 \lambda=g_{t,2}\prod_{j<t}g_{j,1}^{2}\sqrt{k_t/k}.
\end{equation}
正次网络的二次项恰有一个二次节点。若在第 $j$ 层，去掉固定标量后的系数为
$D_2^{(j)}=B_j^T\operatorname{diag}(v_j)B_j$，其中
$B_j=A_j\cdots A_1$、$v_j=A_{j+1}^T\cdots A_t^T\mathbf1/\sqrt{k_t}$。
写 $B_j=C_jA_1$、$H_j=C_j^T\operatorname{diag}(v_j)C_j$。
条件于 $H_j$，$A_1$ 各列为独立 $N(0,I/k)$，故
\[
 \mathbb E(D_2^{(j)})_{ii}=\operatorname{tr}H_j/k,
 \qquad
 \mathbb E\sum_i|(D_2^{(j)})_{ii}-\operatorname{tr}H_j/k|^2
 =2\|H_j\|_{\rm HS}^2/k.
\]
$\|H_j\|_{\rm HS}\le\|C_j\|_{\rm op}^2\|v_j\|$ 与固定矩界控制后式。
$j<t$ 时 $v_j$ 独立于 $C_j$，均值零、协方差 $I/k_j$，从而
$\mathbb E|\operatorname{tr}H_j|^2=k_j^{-1}\mathbb E\sum_a\|C_{j,a:}\|^4=O(1)$，
标量对角平方范数亦为 $O(k^{-1})$。$j=t$ 时
$\operatorname{tr}H_t=\|C_t\|_F^2/\sqrt{k_t}$，均值为 $\sqrt{k_t}$，
且 $\operatorname{Var}\|C_t\|_F^2=O(1)$：沿固定长度乘积逐层使用
\[
 \mathbb E(\|AC\|_F^2\mid C)=\frac{k_{\rm new}}{k_{\rm old}}\|C\|_F^2,
 \qquad
 \operatorname{Var}(\|AC\|_F^2\mid C)
 =\frac{2k_{\rm new}}{k_{\rm old}^2}\|CC^T\|_{\rm HS}^2=O_{L^1}(1)
\]
及全方差公式即可。因此只有输出层的二次节点贡献非消失的确定对角项。乘回固定标量后
得到\eqref{fp:actual-diag}；辅助标记取系数不改变证明。

\begin{lemma}[去掉索引互异限制]\label{fp:actual-restoration}
记 $[p]_r$ 为 $z^r$ 的系数。对固定、常数项为零的多项式
$A,p_1,\ldots,p_{t-1}$，$t\ge2$，有
\begin{equation}\label{fp:actual-restoration-exact}
 J_{t,N}[A;p]=R_{t,N}[A;p]
 -[A]_2\sqrt{k_t/k_{t-1}}\,
 R_{t-1,N}[p_{t-1}^2;p_{<t-1}]+o_{L^2}(1).
\end{equation}
\end{lemma}
\begin{proof}
令 $\gamma=[A]_2\sqrt{k_t/k_{t-1}}$。在取形式 $N$ 次系数前，
从完整多项式中减去 $\gamma k_{t-1}^{-1/2}\sum_i y_i y_i$，
其中 $y$ 为第 $t-1$ 层的多项式输出；保留两个因子各自的索引。
从第 $1$ 层开始向上，依次删除同层索引重复的项。

处理第 $s$ 层时，上方尚未施加索引限制。
若 $s<t-1$，减项等同于输出激活为 $p_{t-1}^2$ 的浅一层网络；
上方两个完整多项式的确定二次对角系数由
\eqref{fp:actual-diag} 给出，分别为
\[
 [A]_2\prod_{j=s+1}^{t-1}[p_j]_1^2\sqrt{k_t/k_s},
 \qquad
 \gamma[p_{t-1}^2]_2\prod_{j=s+1}^{t-2}[p_j]_1^2
       \sqrt{k_{t-1}/k_s}.
\]
由于 $[p_{t-1}^2]_2=[p_{t-1}]_1^2$，二者相等；$s=t-1$ 时，减项
恰好就是第一个多项式的确定二次对角部分。因此其差的二次对角为 $o_{L^2}(1)$；
高于二次的重复索引项由推论~\ref{fp:core-raw-collision-expectation} 分别为 $o_{L^2}(1)$，
一次项则没有重复索引。

两项共用同一个下层展开映射，故确定对角部分在代入前精确抵消。
此映射的范数至多为有限个 $\|A_j\|_{\rm op}$ 的固定次幂；
此前加入的坐标限制和对称化均为收缩。固定次数高斯多项式的矩等价、
H\"older 不等式及\eqref{fp:actual-isometry} 保证每一步删除的误差仍为 $o_{L^2}(1)$。
最后，第 $t-1$ 层的互异限制将减项完全删去，原多项式则成为 $J_{t,N}[A;p]$。
有限步误差相加即得结论。
\end{proof}

对至多多项式增长的 $f$，高斯密度关于方差求导给
$\frac{d}{dq}\mathbb E f(\sqrt qG)|_{q=1}
=\frac12\mathbb E[f(G)(G^2-1)]$，所以
$[H_f]_2=\lambda_f$（$f=\sigma,\chi$）。
将条件均值递推与上述系数恒等式合并，得到
\begin{equation}\label{fp:actual-cancel-recursion}
 \Pi_NF_t^f=R_{t,N}[H_f;P_N]
 +\lambda_f\sqrt{k_t/k_{t-1}}
 \bigl(\Pi_NF_{t-1}^{\chi}-R_{t-1,N}[P_N^2;P_N]\bigr)+o_{L^2}(1).
\end{equation}
置 $U_N=[Q_N-P_N^2]_{\deg\le N}$，
$E_t=\Pi_NF_t^\chi-R_{t,N}[P_N^2;P_N]$。
由 $\lambda_\chi=1$ 和深度一结论，
\[
 E_1=R_{1,N}[U_N]+o_{L^2}(1),\qquad
 E_t=R_{t,N}[U_N;P_N]+\sqrt{k_t/k_{t-1}}E_{t-1}+o_{L^2}(1).
\]
固定深度归纳，再将 $f=\sigma$ 代入\eqref{fp:actual-cancel-recursion}，得到
\begin{equation}\label{fp:actual-U}
 \mathcal Q_N=R_{L,N}[P_N;P_N]
 +\mu'\sum_{t=1}^{L-1}\sqrt{k_L/k_t}\,R_{t,N}[U_N;P_N],
 \qquad \mu'=bu/2,
\end{equation}
以及
\begin{equation}\label{fp:actual-final}
 \boxed{\displaystyle
 \mathbb E_W\|\Pi_NF_L-\mathcal Q_N\|_{L^2_X}^2\longrightarrow0,
 \qquad N>2.}
\end{equation}
奇函数分支的条件均值为零，且各层展开的总次数为奇数，不出现二次对角项，故取
$\mu'=0$，不需引入 $\chi$ 或 $E_t$。
所有激活多项式均没有常数项；逐层代入不能降低形式次数，故可统一删去超过 $N$ 次的项。

以上误差均在单网络先验下以 $L^2$ 控制。下一节用 Cauchy--Schwarz 将其用于任意先验边缘耦合，
并比较式~\eqref{fp:actual-U} 中的深层项和全部较浅 $U_N$ 修正项。

\section{计算树展开项的逐层比较}\label{fp:history}\label{app:cuts}
\newthought*{先固定单项式和因子配对，再对所有选择求和。}
本节固定深度 $L$、次数 $N=m^L$ 和比例宽度 $k_s\asymp d$。奇激活取奇数 $m\ge5$；分段线性激活取偶数 $m\ge6$。所有常数允许依赖这些固定参数。用 $K_d$ 表示一个固定的对数幂，令 $\delta_d=K_d/\sqrt d$、$\kappa_d=K_d/\sqrt d$。以下所有有限次数、高概率界均在源、教师各自的原始高斯边缘事件上成立；两个网络之间没有独立性假设。

在计算树的每个节点选择一个正次数单项式，并给孩子排序，便得到一个\emph{有序展开项}。
完全展开后，它包含 $N$ 个有序输入向量因子。先固定这些因子与比较张量中 $N$ 个因子的配对；
全局对称化是在至多 $N!$ 种配对上取平均。各节点可选不同次数。
不同节点使用的神经元索引即使数值相同，仍须与同一父索引的确定性复制区分。
下文称齐次比较张量所在的网络为教师网络，称被比较展开项所在的网络为源网络；这些名称只区分配对两侧。

第 $h$ 步把每个已比较的高度 $h-1$ 子树视为一个当前因子；第 $1$ 步的因子就是输入向量。
每个待处理教师节点有 $m$ 个当前因子。收缩这 $m$ 个因子后，得到矩阵 $M_{ia}$，
称为\emph{收缩因子}：$a$ 是源节点索引，$i$ 是已处理教师子树的根索引，
其值包含原展开的全部权重和归一化。它依赖已有权重，不是新引入的高斯边。
本步得到的收缩因子先保留为输出，到下一步才参与后续收缩。

\subsection{剩余张量与最低非线性节点}
教师当前原始齐次系数记作 $C$。完成的输出位置采用恒等映射，未完成位置采用有界齐次幂列映射。固定当前教师行 $i$ 后，其余全部输出因子的范数因此至多为 $K_d\|C(e_i,\cdot)\|_{\rm HS}$，而
\begin{equation}\label{fp:teacher-slices}
 \sum_i\|C(e_i,\cdot)\|_{\rm HS}
 \le \sqrt{k}\|C\|_{\rm HS}\le K_d\sqrt d.
\end{equation}
跨网络收缩矩阵始终放在源侧；已经自适应收缩的教师张量不再被声称是新的原始齐次系数。

在源网络每个停止展开的节点处，把下方原子树和已算出的输出合为一个有界张量映射。
高于停止层的每个收缩因子分别保留一个输出索引；同一源索引对应多个输出时，
用恒等复制映射保留这些输出，而不删除它们的对角部分。
在当前层须先保留原系数张量，再施加这些映射和复制，才能使用该层的右旋转估计。

\begin{lemma}[首次非线性节点的收缩]\label{fp:lowest-cut}
假定已经证明：已有收缩因子再次参与收缩时，若其来源节点还带其他因子，则包含全部剩余因子的配对为 $o(1)$。又假定在每次停止展开处，所选因子涉及两个不同源父节点的项也已由相应估计控制为 $o(1)$。那么对其余项，每个教师节点的收缩恰好在同一个源父节点使用 $m$ 个完整孩子，且每个孩子关于当前因子为线性。
\end{lemma}
\begin{proof}
固定与当前教师节点配对的 $m$ 个源因子，沿它们到源根的路径按高度递增处理；起点较高的路径暂缓。
到第一个乘法分支之前，每条路径上的表达式都是原归一化矩阵的连续乘积，末端接已有收缩矩阵，故为有界行映射。
若一个父节点在取出所选线性因子后，不再含其他因子或留待下一步处理的输出，称其为单因子节点。
来自同一父节点的所选因子一起处理；若至少两个所选因子到达该节点，该节点就不是单因子节点。

若遇到的都是单因子节点，就对它们的索引精确求和，沿原线性路径继续。
否则，在高度最低的非单因子节点停止。到达该节点的每个所选因子均为完整线性因子；若此前已经过非线性分支，本应更早停止。
若所选因子分属至少两个源父节点，选择最低的非单因子节点，并考虑其对应当前高斯层矩阵列索引的位置。
另一条较低的线性路径推进到该高度；若先遇到分支，则改在该更低分支处停止。
较高层已有收缩因子再次使用时，按前提已不带其他因子，因此不会同时是另一条所选路径的祖先；本步刚算出的收缩因子暂不参与后续收缩。
在这些停止位置由原计算树重建系数张量，并先保留当前层的原索引，再施加特征映射和复制。
这样选定的索引随当前高斯层矩阵右旋转，满足同层旋转估计的条件。

按假设，该混合父位置的情况被排除。因此全部标记在同一父节点相遇。如果其中两个标记来自同一个非线性原始孩子，它们在严格更低节点就已相遇并应停止，矛盾。所以它们来自 $m$ 个不同的完整原始孩子，且每个孩子关于当前因子为线性。
\end{proof}

若停止展开后的系数张量至少有三个索引，最低非单因子节点所对应的当前层列索引随该层右旋转；
取出一个所选线性因子后仍有其他因子，因此适用双父收缩估计~\ref{fp:core-two-parent}。
所有配对都在恢复后的原展开项上进行，包含相同数值标签，
不再在本步骤插入碰撞掩码。每项完整余张量界为 $K_d/d$，与
式~\eqref{fp:teacher-slices} 配对的误差为 $K_d/\sqrt d=o(1)$。

\paragraph{完整收缩所得矩阵的范数。}
每次已经证明完整的 $m$ 因子收缩，其矩阵是 $m$ 个线性行核的逐项乘积。
若核为 $\rho^{(j)}$，由各行各列的有界 $\ell^2$ 范数和 H\"older 不等式，
\begin{equation}\label{fp:metric-schur}
 \sup_i\sum_a\left|\prod_{j=1}^m\rho^{(j)}_{ia}\right|
 +\sup_a\sum_i\left|\prod_{j=1}^m\rho^{(j)}_{ia}\right|\le K_d.
\end{equation}
故所得矩阵 $M$ 的绝对行列和有界。下一教师原始高斯矩阵 $A$ 满足
$\max|A_{ij}|\le\kappa_d$，从而 $\max|AM|\le K_d\kappa_d$，
即使 $M$ 依赖两个网络也成立。

\subsection{保留一个索引的二阶情形}\label{fp:permanent-mark}\label{fp:core-spectator}
\newthought*{先处理其余子树，最后收缩索引 $a$ 所在的路径。}
考虑系数含两个索引的张量
\begin{equation}\label{fp:marked-state}
 \sum_{a,b}E_{ab}\,\ell_a\otimes Q_b.
\end{equation}
其中 $\ell$ 是当前教师树末端的完整因子，$Q$ 是原始高度至多 $L-1$ 的完整子树；所有尚未收缩的因子及留待下一步处理的输出均保留。$E$ 是原计算树停止展开后得到的二阶系数矩阵，或由该矩阵沿原网络线性路径乘原归一化矩阵得到，因而满足引理~\ref{fp:core-closure} 的范数和矩阵化界。特别地 $\|E\|_{\rm HS}\le K_d$，各非平凡展平至多 $\delta_d$。一个较短源根留下一个收缩因子与一个完整孩子时，就是 $E_{ab}=d_aU_{ab}$，其中 $\|d\|_2\le K_d$、$\|d\|_\infty\le\kappa_d$，$U$ 为原始归一化高斯边。

保留索引 $a$，暂不完成从 $\ell$ 到教师根的路径上的收缩。
其他教师子树按高度递增处理，其对应的源因子均位于 $Q$ 中。
每次重建系数张量都保留 $a$，并在当前层的复制和特征映射之前停止展开。
因此选择一个后代索引时，总模式数至少二；选择两个不同后代索引时，总模式数至少三。

因此，估计 $Q$ 中同一节点仍有剩余因子的项时，总可使用源系数的矩阵化界。
固定教师行后，包含全部剩余因子的源张量范数至多 $K_d\delta_d(\delta_d+\kappa_d)$。
暂缓处理的路径上的因子也包含在该张量中，结合式~\eqref{fp:teacher-slices}，总误差趋零。
涉及不同父节点的收缩用估计~\ref{fp:core-two-parent}。
余下的各次收缩均在 $Q$ 内使用完整孩子，不能直接发生在索引 $a$ 处：
若 $Q$ 非线性，所选因子先在其内部相遇；若 $Q$ 只有一个当前因子，则不足以提供所需的 $m$ 个因子。

完成上述路径之外的高度 $h$ 教师子树，需要 $Q$ 的原始高度至少为 $h$。
若 $Q$ 高度小于 $L-1$，根部其余 $m-1$ 个高度 $L-1$ 的孩子不能全部完成。
若 $Q$ 高度为 $L-1$，这些孩子的收缩都必须在 $Q$ 的原根 $b$ 处完成。
记相应收缩矩阵为 $M^{(j)}$，最终教师矩阵为 $A$，置 $G^{(j)}=AM^{(j)}$。
由式~\eqref{fp:metric-schur}，每个 $G^{(j)}$ 的列 $\ell^2$ 范数有界，最大条目至多 $K_d\kappa_d$。

保持该路径上的教师子树所含的全部因子。固定最终教师行 $i$，它的范数有界。去掉 $m-1$ 个已知收缩因子后，完整源余张量为
\[
 R_b=\sum_aE_{ab}\ell_a\otimes Q_b^{\rm res},\qquad
 \|R_b\|\le K_d\|E_{:b}\|_2,
 \quad\sum_b\|R_b\|\le K_d\sqrt d\|E\|_{\rm HS}.
\]
这里每个余因子仍原样保留，$\ell$ 的列映射有界，$Q_b^{\rm res}$ 的列范数有界。归一化最终读出最大值为 $O(d^{-1/2})$；利用两个 $G$ 的列 $\ell^2$ 界及其余 $m-3$ 个的条目界，完整标量配对至多
\begin{equation}\label{fp:marked-terminal}
 K_d\|E\|_{\rm HS}\kappa_d^{m-3}=o(1).
\end{equation}
这一步同时使用 $m-1$ 个收缩因子，未把一个小标量核与任意余张量直接相乘。它解决了完整剩余深度下的单因子/非单因子二模式边界，也解决较短任意根的一个收缩因子加一个完整孩子的边界。

\subsection{两类激活展开项的统一归纳}
\newthought*{对源高度作强归纳。}
令 $P$ 为相应激活的普通零常数多项式，$m$ 按本节开头选定。
同时证明：高度 $t<L$、根为任意零常数多项式 $A$、下层各节点为 $P$ 的
每个有序展开项，在每个固定槽分配下配对趋零；高度 $L$ 时先分出全 $m$
展开项，证明其余普通 $P$ 展开项配对趋零。不同孩子可以选择不同单项式，
根次数不限制奇偶性。
以下只用节点次数为正，以及上一节的局部收缩估计和二阶对角公式。

停止展开后的系数张量若至少有三个索引，已由上面的估计处理；只有一个索引时，不涉及两个不同父节点。
剩下系数为二阶的情形。若两个父节点中一个为单因子节点、另一个不是，使用式~\eqref{fp:marked-state}--\eqref{fp:marked-terminal}。
若两者都不是单因子节点，则二者都在当前同一层展开：较高层收缩因子再次使用时已经不带其他因子，而较高层尚未处理的输出会额外保留一个索引，使总阶数至少为三。
因此这里的 $D$ 是在施加下方特征映射之前、两个索引属于同一原网络层的二阶系数矩阵。

非对角部分由双父收缩估计~\ref{fp:core-two-parent}的二阶情形控制，完整配对为 $o(1)$。

对角由式~\eqref{fp:actual-diag} 分为 $\lambda_s I/\sqrt{k_s}$ 与
Hilbert--Schmidt 范数为 $o_{L^2}(1)$ 的误差；后者由源侧完整特征映射的算子范数界控制。
主项识别两个原始父标签，把它们的两个选中根单项式相乘，保留所有有序孩子与完全相同的教师槽分配。得到高度 $s<t$、读出 $1/\sqrt{k_s}$ 的原始前缀展开项，乘精确确定性系数 $\lambda_s$。其根可有任意奇偶性，所以必须使用上述加强归纳类。已完成的收缩矩阵可展开回原始槽收缩；它们是等式，故融合不产生残缺子树。若 $s=0$，只剩输入次数二，不能贡献 $N>2$。

两个索引也可能分别对应较高层已算出的收缩因子和当前层刚展开的因子。
它们来自不同层，不使用上述同层对角删除。若较高层因子参与本次收缩，采用保留根索引的论证；若暂不参与，本次只涉及一个父索引。
若两个索引都对应较高层的收缩因子，则在尚未完成时，每个至多代表 $m^{L-1}$ 个原输入因子，合计少于 $N$，不能组成整个展开项；若其中一个已达到高度 $L$，它已独自使用全部 $N$ 个因子。
同一父索引的确定性复制也按其代表的原输入因子数计数，不增加不同父节点的数目。

现在考虑已有收缩因子用于下一步时，同一源节点还带其他完整因子的项。
若当前教师节点的 $m$ 个因子还涉及另一个源父节点，使用引理~\ref{fp:core-local-contraction} 的第 1 项；
若全部来自同一个源父节点，而停止展开后的原系数张量至少有两个索引，使用引理~\ref{fp:core-local-contraction} 的第 2 项。
因此只剩原系数张量有一个索引的情形。指定收缩因子之外若至少还有两个因子，
使用引理~\ref{fp:core-local-contraction} 的第 3 项；若没有其他因子，该项保留。

若恰有一个其他因子，只需判断它是完整原孩子还是已有收缩因子。
这是因为这里只使用先前已证明完整的步骤，每次仅将 $m$ 个完整原孩子
替换为一个收缩因子。若另一个也是收缩因子，本列至多提供两个当前因子位置，
不能容纳一个 $m\ge5$ 因子的教师叶，已属于访问其他列的情形。
若另一个是原孩子，则所有较高层已完成输出及未收缩子树仍分别保留索引，
且原子树次数为正；上方仅一个索引因此迫使当前节点以上只有原始一元链。
其系数 $d$ 是归一化读出乘原高斯连续积，故
$\|d\|_2\le K_d$、$\|d\|_\infty\le\kappa_d$。展开唯一原孩子的第一条边得到
\[
 \sum_a d_a\ell_a\otimes\Bigl(\sum_bU_{ab}Q_b\Bigr)
 =\sum_{a,b}d_aU_{ab}\,\ell_a\otimes Q_b.
\]
$Q$ 的原高度至多 $t-1\le L-1$，正是
式~\eqref{fp:marked-state}--\eqref{fp:marked-terminal}处理的情形。
因此留下的每个收缩因子再次使用时，都已包含其来源节点的全部因子。
此结论对普通节点与任意次数的较短根同时成立，给出引理~\ref{fp:lowest-cut}的前提。



从 $t=1$ 开始归纳。每次二阶对角合并使源高度严格降低，故由归纳假设控制；
高度降为零时只剩输入次数二，不能贡献 $N>2$。
上述估计排除了涉及不同父节点的项，以及再次使用收缩因子时仍有其他因子的项。
余下每次收缩均使用 $m$ 个完整孩子。沿线性路径求和会经过原树的一条边，
每次 $m$ 因子收缩完成一个教师子树；原树和教师树均有限，因此步骤有限终止。
最终收缩使用全部 $N$ 个输入因子，不再留下正次数因子。

高度 $h$ 的教师子树在每条源路径上至少需要 $h$ 个原节点：$h=1$ 需要分支节点；
从 $h-1$ 到 $h$ 时，$m$ 个孩子已各自完成高度 $h-1$ 的收缩，且再次使用时
不带剩余因子，所以父节点严格位于这些孩子之上。线性节点只增加源高度。
故 $t<L$ 时无法完成高度 $L$ 的教师树，配对趋零。
$t=L$ 时所有二阶对角合并项已由较低高度排除，而每条路径必须取高度等号：
没有额外线性节点，每个节点次数恰为 $m$，与正在考虑的非全 $m$ 展开项矛盾。
因此这些项的配对全都趋零；预先分出的全 $m$ 项恰为原齐次比较项，
其系数、权重和归一化完全未改。这同时完成两类激活的归纳。



\subsection{从确定性事件到任意先验耦合}\label{fp:coupling-uniform}
\newthought*{先控制各自先验下的误差，再对耦合取上确界。}
固定次数和深度时，有序展开项、因子配对、停止展开的位置选择、合并高度及收缩步骤的种类数均与 $d$ 无关。分别在两个边缘上同时施加所有前述原始高斯事件，然后才允许跨网络收缩矩阵依赖两个网络。在共同好事件上，各项估计均为确定性 $o(1)$。在坏事件上，用固定次数系数的有界边缘高阶矩和 H\"older 不等式；坏事件概率可取任意所需多项式阶。该控制不依赖耦合，因此对所有先验边缘耦合一致。

设 $T_W$ 为与激活对应、带完整标量系数的全 $m$ 平衡齐次输入混沌项，
$\mathcal Q_N(V)$ 为式~\eqref{fp:actual-U} 的多项式近似。
上述逐展开项估计给出 $\mathbb E_\pi|\langle T_W,\mathcal Q_N(V)-T_V\rangle|=o(1)$，对耦合一致。
又 $T_W$ 位于第 $N$ 输入混沌，故 $\langle T_W,F_V\rangle=\langle T_W,\Pi_NF_V\rangle$。
剩余配对期望由 Cauchy--Schwarz 不等式控制为
$\|T_W\|_{L^2(W,X)}\|\Pi_NF_V-\mathcal Q_N(V)\|_{L^2(V,X)}=o(1)$；
这里使用有界边缘二阶矩与式~\eqref{fp:actual-final}，同样对耦合一致。因此
\begin{equation}\label{fp:global-function-pairing}
 \boxed{\displaystyle
 \sup_{\pi\in\Gamma(\mathsf P_W,\mathsf P_V)}
 \mathbb E_\pi|\langle T_W,F_V-T_V\rangle|\longrightarrow0.}
\end{equation}

\section{齐次比较网络的全局匹配}\label{hm:section}\label{app:matching}
本节仅讨论激活为 $c_m z^m$、$c_m=(m!)^{-1/2}$ 的齐次网络。固定 $L$ 与整数 $m\ge5$，令 $k_0=d$，$cd\le k_s\le Cd$，并定义
\[
 h^{(0)}=x,\qquad h^{(s)}=c_m(W^{(s)}h^{(s-1)}/\sqrt{k_{s-1}})^{\odot m},\qquad
 \mathcal H_W(x)=k_L^{-1/2}\sum_i h_i^{(L)}(x),\qquad
 p=\sum_{s=1}^L k_{s-1}k_s.
\]
每个网络的所有权重分别具有独立标准高斯先验；两个网络之间可任意耦合。记 $P_s(Y)$ 为将 $h^{(s)}$ 替换为形式变量 $Y$ 后的下游输出，$q_s=m^{L-s}$，$C_s=C_{q_s}(P_s)$。这里 $P(Y)=\langle C_r(P),Y^{\otimes r}\rangle$，张量使用继承的 Hilbert 范数；若 $P=\sum_{|\nu|=r}p_\nu Y^\nu$，则
\begin{equation}\label{hm:coeff-norm}
 \|C_r(P)\|^2=\sum_{|\nu|=r}\frac{\nu!}{r!}|p_\nu|^2.
\end{equation}
特别地 $C_L=k_L^{-1/2}{\bf1}$。

\begin{theorem}[齐次网络的有限标签全局匹配]\label{hm:global-matching}
存在一族确定性的参数对齐映射 $\mathcal G_d$，以及仅依赖固定深度、次数和宽度比例的常数 $K$，使得
\begin{equation}\label{hm:main}
 \log|\mathcal G_d|=O(d\log d)=o(p),\qquad
 \frac1p\mathbb E\min_{g\in\mathcal G_d}\|W-gV\|_F^2
 \le K\mathbb E\|C_0(W)-C_0(V)\|^2+o_d(1).
\end{equation}
余项对所有具有上述先验边缘分布的耦合一致。若 $X\sim N(0,I_d)$，则右侧亦可替换为
$K\mathbb E\|\mathcal H_W-\mathcal H_V\|_{L^2_X}^2+o_d(1)$。
\end{theorem}

\subsection{系数切片的范数界}

第~\ref{tg:section} 节取 $w_i=a_i$、$q=r$。因 $m\ge5$、$k=O(d)$、
$\eta=O(\sqrt{\log d/d})$，其扰动条件成立，故完整张量 $P_\nu$ 及删除一个因子的张量 $Z_{\nu,j}$
两族均满足式~\eqref{tg:frame} 的维度无关 Gram 上下界；行长在 $[1/2,2]$ 时亦然。

令 $B_{s,i}=\partial_iP_s$。在各层行范数、相干度和列平方和的通常高斯集中事件上，
\begin{equation}\label{hm:derivative}
 a/k_s\le\|C(B_{s,i})\|^2\le b/k_s\quad\hbox{对所有 }s,i.
\end{equation}
为证明式~\eqref{hm:derivative}，从终层 $B_{L,i}=k_L^{-1/2}$ 反向归纳。置 $w_j=W^{(s+1)}_{j:}/\sqrt{k_s}$，链式法则给出
\[
 B_{s,i}(Y)=mc_m\sum_j w_{ji}(w_j\cdot Y)^{m-1}
 B_{s+1,j}\big(c_m(w\cdot Y)^{\odot m}\big).
\]
展开 $B_{s+1,j}$ 后，每项恰为标签 $(\nu,j)$ 的第二类张量；固定次数下式 \eqref{hm:coeff-norm} 与普通单项式系数范数等价。第二类张量的 Gram 矩阵界因此给出
\[
 \|C(B_{s,i})\|^2\asymp\sum_j w_{ji}^2\|C(B_{s+1,j})\|^2
 \asymp k_{s+1}^{-1}\sum_j w_{ji}^2\asymp k_s^{-1}.
\]
这一估计在行向量满足上述条件时逐点成立。记 $H_s$ 为切片 $C_s(e_i,\cdot)$ 的 Gram 矩阵，即 $(H_s)_{ij}=\langle C_s(e_i,\cdot),C_s(e_j,\cdot)\rangle$。精确微分恒等式给出
\begin{equation}\label{hm:mode}
 (H_s)_{ii}=q_s^{-2}\|C(\partial_iP_s)\|^2,\qquad
 a'/k_s\le(H_s)_{ii}\le b'/k_s.
\end{equation}

\subsection{最大系数与重复索引项的控制}
\newthought*{用最大系数界控制删除少量行的误差。}
对行矩阵 $A$ 定义 $U_Ae_i=a_i^{\otimes m}$。由
$U_A=(A^T)^{\otimes m}J$、$Je_i=e_i^{\otimes m}$，有 $\|U_A\|_{\rm op}\le\|A\|_{\rm op}^m$。
对 $s<L$，$C_s$ 是原下游齐次网络的系数，阶数 $q_s=m^{L-s}\ge5$；
可直接应用原始系数界~\ref{fp:core-raw-collision-expectation}。终层 $C_L=k_L^{-1/2}{\bf1}$ 单独处理。
因此对所有 $s$ 和固定 $0<u<\infty$，
\begin{equation}\label{hm:moments}
 \sup_d\mathbb E\|C_s\|^u<\infty.
\end{equation}
平方自由单项式中每个变量次数至多为一，记 $\mathrm{SF}$ 为其正交投影。
对 $|I|=q_s$，写 $Y_I$ 的系数为 $p_I$，令 $c_I=p_I/\sqrt{q_s!}$，则
$\sum_Ic_I^2=\|\mathrm{SF}C_s\|^2$，且 $c_I$ 是对应互异张量坐标的 $\sqrt{q_s!}$ 倍。
式~\eqref{fp:core-coordinate-bounds}及其证明给任意固定 $M$：
\begin{equation}\label{hm:dense}
 \mathbb P\left\{\max_I|c_I|^2>C_M(\log d)^{D_s}k_s^{-q_s}\right\}\le d^{-M},
 \qquad D_s=m+\cdots+m^{L-s},\quad s<L.
\end{equation}
同时，在各层共同的边缘高概率事件上，
\begin{equation}\label{hm:collision}
 \|(I-\mathrm{SF})C_s\|^2\le L_d/d=o(1)\quad(s<L),
 \qquad (I-\mathrm{SF})C_L=0.
\end{equation}
后文只需该绝对误差；施加有界的完整提升仍为 $o(1)$。
式~\eqref{hm:mode} 的下界针对完整切片，故应用前须恢复完整张量。

所有这些边缘事件可同时选取，失败概率趋于零。有符号行列置换保持行范数、算子范数、相干度、最大坐标及列平方和；下游系数张量只发生坐标重排和符号变化。因此其范数、最大系数绝对值、式~\eqref{hm:mode} 的切片界，以及重复指标部分的范数界均保持。即使根据两个网络选择对齐，这些确定性界仍成立。

\subsection{全局对称与分组张量的精确换算}
在第 $s$ 层写 $w_i=W^{(s)}_{i:}/\sqrt{k_{s-1}}=r_i a_i$，$\|a_i\|=1$，
$R=\operatorname{diag}(r_i^m)$，$\overline C_s=R^{\otimes q_s}C_s$。精确代入为
\begin{equation}\label{hm:substitution}
 C_{s-1}=c_m^{q_s}\operatorname{Sym}_{mq_s}U_A^{\otimes q_s}\overline C_s.
\end{equation}
均匀半径误差 $\max_i|r_i-1|\le\delta_d\to0$ 意味着
$\|\overline C_s-C_s\|\le C\delta_d\|C_s\|$。
式~\eqref{hm:mode} 的切片界、最大系数界与重复指标部分的估计仍成立，因为每个张量坐标只乘上固定个 $1+o(1)$ 半径因子。

固定 $q=q_s$，对 $|I|=q$ 定义
\[
 Q_I=\sqrt{q!}\operatorname{Sym}_q\bigotimes_{i\in I}a_i^{\otimes m},\qquad
 E_I=\sqrt{(mq)!/(m!)^q}\operatorname{Sym}_{mq}\bigotimes_{i\in I}a_i^{\otimes m}.
\]
$Q_I$ 属于分组空间，$E_I$ 属于实际全局对称空间。平方自由坐标基向量经分组提升变为 $Q_I$，再经 \eqref{hm:substitution} 变为 $\sqrt{\lambda_{m,q}}E_I$，其中
\begin{equation}\label{hm:lambda}
 \lambda_{m,q}=q!/(mq)!.
\end{equation}
两种归一化使不拆分 $m$ 个槽的跨网络配对项完全相同。

\subsection{两种张量表示的 Gram 矩阵比较}
\newthought*{一次 Schur 比较同时处理自身与交叉 Gram。}
给定两个单位行阵列 $A,B$，自身相干度至多 $\eta_d=O(\sqrt{\log d/d})$，
算子范数至多 $K$。置 $\rho_{ij}=\langle a_i,b_j\rangle$、$\tau=d^{-1/8}$。
二部图 $|\rho_{ij}|>\tau$ 的度数至多 $D=\lceil K^2/\tau^2\rceil$；
在各侧连接共享邻居的顶点，所得图度数至多 $D^2$。不含图边的 $q$ 元集合称为好集合。
对好 $I=\{i_1,\ldots,i_q\}$，任一列中至多一个 $|\rho_{i_a j}|>\tau$，故
\begin{equation}\label{hm:mixed-column}
 \sum_j\prod_a|\rho_{i_a j}|^{v_a}\le
 \begin{cases}K^2,&\#\{a:v_a>0\}=1,\\ C\tau,&\#\{a:v_a>0\}\ge2,
 \end{cases}\qquad \sum_a v_a=m.
\end{equation}
为证后一界，在 $v_a>0$ 的位置中选取 $|\rho_{i_a j}|$ 最大者，并按其位置分区：其指数至少二时保留该项的平方，
否则保留它与另一项各一幂；剩余因子中至少一幂至多 $\tau$。
所保留项的列和分别由平方和界及 Cauchy--Schwarz 控制。
全槽配对计数给出
\[
 \langle E_I^A,E_J^B\rangle
 =\sum_{\substack{e_{ab}\ge0\\\sum_b e_{ab}=\sum_a e_{ab}=m}}
 \frac{(m!)^q}{\prod_{a,b}e_{ab}!}\prod_{a,b}\rho_{i_a j_b}^{e_{ab}}.
\]
矩阵 $e=(e_{ab})$ 记录每对因子组之间的配对数。当 $e=mP$、$P$ 为置换矩阵时，整组因子保持在一起，这些项之和恰为 $\langle Q_I^A,Q_J^B\rangle$；其余矩阵至少有一列含两个正元。记两族张量在好集合上的交叉 Gram 矩阵分别为 $K_{\rm act}=(\langle E_I^A,E_J^B\rangle)_{I,J}$、$K_{\rm grp}=(\langle Q_I^A,Q_J^B\rangle)_{I,J}$。
固定好 $I$，把对好 $J$ 的绝对行和放大为对所有有序 $J$ 的和，各列分离，
由 \eqref{hm:mixed-column} 得 $C\tau$；交换两侧得到同一列和界。因此
\begin{equation}\label{hm:cross-schur}
 \|K_{\rm act}-K_{\rm grp}\|_{\rm op}\le C\tau
\end{equation}
在双方好集合上成立。这也适用于 $A=A$、$B=B$；因 $\eta_d<\tau$，这两种自身比较的所有集合均好。
最大度数 $D^2$ 的图中，含边 $q$ 元集合至多 $C_qD^2k^{q-1}$ 个。
由 \eqref{hm:dense}，同时对所有可能依赖完整网络及系数的图，删除能量至多
$C D^2k^{q-1}\max_I|c_I|^2\le C d^{-1/2}(\log d)^{D_s}=o(1)$。
实际与分组提升的算子范数均有界，故删除以及 \eqref{hm:collision} 的平方自由截断
各只产生 $o(1)$ 范数误差。对三个 Gram 块 $AA,AB,BB$ 使用同一比较，再恢复完整系数，得
\begin{equation}\label{hm:sym-group}
 \|C_{s-1}^A-C_{s-1}^B\|^2
 =\lambda_{m,q}\|G_A-G_B\|^2+o(1),\qquad
 G_A=U_A^{\otimes q}\overline C_s^A,\quad
 G_B=U_B^{\otimes q}\overline C_s^B.
\end{equation}
误差在联合典型事件上一致；$q=1$ 时无拆槽，此等式精确成立。

\subsection{一个平方差恒等式同时给出匹配与系数恢复}
\newthought*{交叉矩阵的绝对行列和控制匹配误差。}
记 $U=U_A,V=U_B,M=U^*V=(\rho_{ij}^m)$，$\rho_i=\max_j|\rho_{ij}|$。
一行中 $|\rho_{ij}|>\tau$ 的子集大小至多 $D$，其自身 Gram 算子范数至多 $1+D\eta_d$；
低相关项保留平方后提出 $\tau^{m-2}$，故
\begin{equation}\label{hm:cross-power}
 \sum_j|M_{ij}|\le\rho_i^{m-2}(1+D\eta_d)+K^2\tau^{m-2}
 \le\rho_i^{m-2}+e_d,\qquad \sum_i|M_{ij}|\le1+e_d,
\end{equation}
其中 $e_d\to0$ 对两阵列一致；列界由同一论证得到。
置 $\widetilde M=M/(1+e_d)$、$\mathsf K=\widetilde M^{\otimes q}$，
并将完整系数张量 $C=\overline C_s^A,D'=\overline C_s^B$ 写为有序指标数组 $(c_I),(d_J)$。
$\mathsf K$ 的绝对行列和 $r_I,s_J$ 均不超过一，且有精确恒等式
\begin{equation}\label{hm:transport-identity}
 \begin{aligned}
 \mathcal E&:=\|C\|^2+\|D'\|^2-2\langle C,\mathsf K D'\rangle\\
 &=\sum_I(1-r_I)c_I^2+\sum_J(1-s_J)d_J^2
   +\sum_{I,J}|\mathsf K_{IJ}|
       (c_I-\operatorname{sgn}(\mathsf K_{IJ})d_J)^2.
 \end{aligned}
\end{equation}
零权项不影响等式。因为 $U^*U,V^*V=I+o_{\rm op}(1)$（非对角绝对行和至多 $k\eta_d^m=o(1)$），
且系数范数有界，$\mathcal E=\|G_A-G_B\|^2+o(1)$。
$r_I$ 是各槽行和的乘积，故 $1-r_I$ 至少为一减去第一槽对应的行绝对和；由 \eqref{hm:cross-power}
及 \eqref{hm:mode}，\eqref{hm:transport-identity} 给出
\[
 k_s^{-1}\sum_i(1-\rho_i^{m-2})\le C\|G_A-G_B\|^2+o(1).
\]
固定 $\rho_0\in(1/\sqrt2,1)$。由于 $2\rho_0^2>1+\eta_d$，绝对相关超过 $\rho_0$ 的边构成部分双射。
将其补为排列并逐行定向，使对角相关非负。未匹配行的比例由上一式除以
$1-\rho_0^{m-2}$ 控制；已匹配行使用 $1-x^m\le\frac m{m-2}(1-x^{m-2})$。
令 $\ell_i=1-\langle a_i,b_i\rangle^m$，遂得
\begin{equation}\label{hm:row-loss}
 k_s^{-1}\sum_i\ell_i+k_s^{-1}\sum_i\|a_i-b_i\|^2
 \le C\|G_A-G_B\|^2+o(1).
\end{equation}
定向时同步置换、变号学生系数以保持 $G_B$ 不变，式~\eqref{hm:mode} 的切片界亦保持。
在此表示中 $\mathsf K_{II}=\prod_{a=1}^q(1-\ell_{i_a})/(1+e_d)^q\ge0$，
且 $1-\mathsf K_{II}\le\sum_a(\ell_{i_a}+e_d)$。
保留 \eqref{hm:transport-identity} 的对角平方项，记切片族 $C(e_i,\cdot)$、$D'(e_i,\cdot)$ 的 Gram 矩阵分别为 $H_C,H_{D'}$，得到
\[
 \begin{aligned}
 \|C-D'\|^2
 &\le\mathcal E+\sum_I(1-\mathsf K_{II})(c_I-d_I)^2\\
 &\le\mathcal E+2q\sum_i(\ell_i+e_d)
       ((H_C)_{ii}+(H_{D'})_{ii})
 \le C\|G_A-G_B\|^2+o(1).
 \end{aligned}
\]
最后去除一致趋于一的半径因子，并用 \eqref{hm:sym-group}，得到
\begin{equation}\label{hm:recover}
 \|C_s^A-C_s^B\|^2
 \le C\|G_A-G_B\|^2+o(1)
 \le C\|C_{s-1}^A-C_{s-1}^B\|^2+o(1).
\end{equation}

\subsection{内部符号传播与终层对齐}
内部层 $s<L$ 的有符号行置换写作 $S_s=D_sP_s$，其中 $D_s$ 对角元为 $\pm1$。隐特征相应变为
$R_sh^{(s)}$，$R_s=D_s^{m\bmod2}P_s$。同时以 $V^{(s+1)}R_s^{-1}$ 替换下一层矩阵，整个齐次函数精确保持不变。奇数 $m$ 将符号传播到下一层列，偶数 $m$ 仅传播排列；形式下游系数相应重排、变号，前述典型事件保持。

终层也允许有符号排列，但仅用作参数对齐标签。若匹配符号为 $\varsigma_i$、学生半径为 $t_i$，
把学生行定向后，同时在分析表示中取
$\overline c_i^B=\varsigma_i^{m}t_i^m/\sqrt{k_L}$，教师系数为
$\overline c_i^A=r_i^m/\sqrt{k_L}$。这样表示的原学生张量不变，式~\eqref{hm:mode} 的切片界仍成立。
此时 $q_L=1$，由 \eqref{hm:sym-group} 和 \eqref{hm:row-loss}，定向后的终层平均行距离
直接由 $C\|C_{L-1}^A-C_{L-1}^B\|^2+o(1)$ 控制。无需恢复终层系数，
也不要求这个终层标签保持网络函数。

\subsection{深度归纳与任意先验耦合}
令 $E_0=\|C_0(W)-C_0(V)\|^2$。自下而上在内部层选择有符号匹配、传播至下一层列，记当前对齐后的形式张量误差为 $E_s$。内部对称变换保持当前 $C_{s-1}$，且以后对齐不再改变已完成的较低层。因此 \eqref{hm:sym-group}、\eqref{hm:row-loss}、\eqref{hm:recover} 在每个内部层给出
\[
 E_s\le CE_{s-1}+e_d,\qquad
 \frac1{k_s}\sum_i\|w_{s,i}-(gv)_{s,i}\|^2\le CE_{s-1}+e_d,
 \qquad e_d\to0.
\]
半径误差由
$\|r_ia_i-t_ib_i\|^2\le C\|a_i-b_i\|^2+C\delta_d^2$
吸收。终层使用上一小节，固定深度归纳得到每层平均归一化行距离至多 $C_{L,m}E_0+o(1)$。
恢复原始权重需要乘 $k_{s-1}$，故以 $k_{s-1}k_s/p$ 加权求和，得到联合典型事件上的 \eqref{hm:main} 确定性版本。

具体映射族为
\[
 (gV)^{(1)}=S_1V^{(1)},\qquad (gV)^{(s)}=S_sV^{(s)}R_{s-1}^{-1}\quad(s\ge2),
\]
所有 $S_s$ 均可为有符号排列，终层标签不必是函数对称。因此
\[
 \log|\mathcal G_d|\le\sum_s\{\log(k_s!)+k_s\log2\}=O(d\log d).
\]
可固定标签次序打破所有平局，获得可测的匹配选择。任意先验边缘耦合下，两个各自典型事件的并集界保证联合失败概率趋零，不需要两个网络相互独立。坏事件上取恒等对齐，
\[
 p^{-1}\min_g\|W-gV\|^2\le2p^{-1}(\|W\|^2+\|V\|^2).
\]
归一化高斯权重平方范数有一致二阶矩；Cauchy--Schwarz 将坏事件期望变为 $o(1)$，且对耦合一致。取期望即证 \eqref{hm:main}。最后最高输入次数 $N=m^L$ 的高斯混沌满足
\[
 N!\|C_0(W)-C_0(V)\|^2
 =\|\Pi_N(\mathcal H_W-\mathcal H_V)\|_{L^2_X}^2
 \le\|\mathcal H_W-\mathcal H_V\|_{L^2_X}^2,
\]
证明函数距离版本。\hfill$\square$


\paragraph{实际函数的参数比较。}
定理 \ref{fp:global-function-pairing} 和正文的极化恒等式将实际函数距离控制转为齐次系数距离控制，再应用定理 \ref{hm:global-matching}。行符号只用于有限标签参数对齐，并不要求保持实际激活的函数值。

\section{条件 I--MMSE 与统一平滑测试场}\label{app:information}
记 $f=F_{\theta(W)}^{(L)}(X)$、$a=\nabla_Wf$、
$\mathcal I_W=\E_Xaa^T/\Delta$、$Z=(\mathcal D_n,X)$。
另取独立标准高斯 $N$，令 $U_s=\sqrt sf+N$、
$m(s)=\E[(f-\E[f\mid Z,U_s])^2]$。真实高斯输出通道及条件 I--MMSE 给
\begin{equation}\label{eq:conditional-immse}
 \epsilon_{\log,d,n}=I(f;U_{1/\Delta}\mid Z)
 =\frac12\int_0^{1/\Delta}m(s)\,ds.
\end{equation}
为验证第二等号，条件于 $Z=z$，令 $q_v$ 为 $f+\sqrt vN$ 的密度。
高斯热方程、分部积分及条件期望公式给
\[
 \frac d{dv}h(q_v)=\frac12\mathcal J(q_v),\qquad
 \partial_y\log q_v(y)=\frac{\E[f\mid z,f+\sqrt vN=y]-y}{v}.
\]
正交分解得到 $\operatorname{mmse}(f\mid z,f+\sqrt vN)=v-v^2\mathcal J(q_v)$。
从 $h(q_v)$ 减去噪声熵后微分，再令 $v=1/s$，互信息导数即为 MMSE 的一半。
上述计算先对截断信号及有界空间成立；由 $\mathcal J(q_v)\le1/v$
和信号二阶矩的一致可积性，可在任意 $0<s_0\le s\le s_1$ 上去掉截断。
最后用 $I(f;U_s\mid Z)\le s\E f^2/2$、$m(s)\le\E f^2$，
平均 $z$ 并令 $s_0\downarrow0$，即得 \eqref{eq:conditional-immse}。

\subsection{真实通道的分部积分界}
对只依赖 $(W,X)$ 的向量场 $b$，令
\[
 G_s=\delta b+b^TS_D+\sqrt sN b^Ta,\qquad
 \delta b=\operatorname{div}_Wb-W\cdot b,\quad
 S_D=\nabla_W\log p(\mathcal D_n\mid W).
\]
固定观察并对参数分部积分得
$\E[(\E[f\mid Z,U_s]-f)G_s]=\E b^Ta$。
给定 $(W,X)$，训练得分与辅助噪声独立且中心化，训练得分协方差为
$n\mathcal I_W$。因此 Cauchy--Schwarz 给
\begin{equation}\label{eq:vt-mmse}
 m(s)\ge\frac{(\E b^Ta)^2}
 {n\E b^T\mathcal I_Wb+\E(\delta b)^2+s\E(b^Ta)^2}.
\end{equation}
先对紧支撑光滑场及截断的后验均值证明，再用引理 \ref{an:sobolev}
的 Sobolev 逼近和 $L^2$ 收敛取极限；以下估计保证所需可积性。

\subsection{正则化测试场的 OU 平滑}
固定 $0<\tau\le r\le1$，令 $\lambda=r/p$、
$M_W=(\mathcal I_W+\lambda I_p)^{-1}$、$b_0(W,X)=M_Wa(W,X)$。
取 \eqref{eq:gradient-stability} 中的 $W_\tau$，仅对参数作条件平均：
\[
 b_\tau(W,X)=\E[b_0(W_\tau,X)\mid W,X].
\]
真实网络、训练似然及辅助观测 $U_s$ 均不变。首先
\[
 \E\|b_0\|^2=\Delta\E\tr(M_W^2\mathcal I_W)
 \le\frac{\Delta p}{4\lambda},\qquad
 \E\|b_\tau\|^4\le\E\|b_0\|^4\le C\lambda^{-4}.
\]
将 $b_0$ 视为 $L^2(P_X;\mathbb R^p)$ 值函数，记第 $k$ 阶参数 Hermite
能量为 $c_k$。OU 半群逐阶乘 $e^{-\tau k}$，所以
\[
 \E\|Db_\tau\|_{\rm HS}^2
 =\sum_{k\ge1}ke^{-2\tau k}c_k
 \le C\tau^{-1}\E\|b_0\|^2\le Cp^2/(r\tau).
\]
这直接使 $b_\tau$ 属于高斯 $W^{1,2}$，不需要对 $M_W$ 求导。
由引理 \ref{an:sobolev}、梯度四阶矩及 $\tau\le r$，
\begin{equation}\label{unified:field-cost}
 \E(\delta b_\tau)^2\le Cp^2/(r\tau),\qquad
 \E(b_\tau^Ta)^2\le
 (\E\|b_\tau\|^4\E\|a\|^4)^{1/2}
 \le Cp^2/r^2\le Cp^2/(r\tau).
\end{equation}

由高斯先验不变性、\eqref{eq:gradient-stability} 和 Cauchy--Schwarz，
\begin{equation}\label{field:target}\begin{aligned}
 \E b_\tau^Ta(W,X)
 &=\E b_0(W_\tau,X)^Ta(W,X)\\
 &\ge\Delta p[1-D_d(r)]
   -(\E\|b_0\|^2)^{1/2}
       (\E\|a(W_\tau,X)-a(W,X)\|^2)^{1/2}\\
 &\ge\Delta p[1-D_d(r)-C(\tau/r^2)^{1/4}].
\end{aligned}\end{equation}

为控制训练得分项，另取独立输入 $\overline X$。条件 Jensen 给
\[
 \E b_\tau^T\mathcal I_Wb_\tau
 \le\Delta^{-1}\E[(b_0(W_\tau,X)^Ta(W,\overline X))^2].
\]
写 $M'=M_{W_\tau}$、$\mathcal I'=\mathcal I_{W_\tau}$。
条件于 $(W,W_\tau,\overline X)$，$X$ 仍独立，故
\[
 \E_X b_0(W_\tau,X)b_0(W_\tau,X)^T
 =\Delta M'\mathcal I'M'\preceq\frac{\Delta}{4\lambda}I_p.
\]
在内积中分解 $a(W,\overline X)=a(W_\tau,\overline X)+e$。
第一项的平方平均除以 $\Delta$ 为
$\Delta\E\tr[(M'\mathcal I')^2]\le\Delta p$；第二项满足
\[
 \Delta^{-1}\E[(b_0(W_\tau,X)^Te)^2]
 \le\frac1{4\lambda}\E\|e\|^2\le C\sqrt\tau/\lambda.
\]
由 $L^2$ 三角不等式，得到
\begin{equation}\label{field:training}
 \E b_\tau^T\mathcal I_Wb_\tau
 \le\Delta p[1+C(\tau/r^2)^{1/4}]^2.
\end{equation}
将 \eqref{field:target} 的右侧取正部，把
\eqref{unified:field-cost}、\eqref{field:training} 代入
\eqref{eq:vt-mmse}，再对 \eqref{eq:conditional-immse} 积分。
最后取 $\tau=r^4$，使 $(\tau/r^2)^{1/4}=\sqrt r$、$r\tau=r^5$，
即得 \eqref{eq:lower-unified}。
