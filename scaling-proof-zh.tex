\begin{abstract}
对任意固定深度、各层宽度与输入维数成正比的网络，本文证明平均 Bayes 预测 KL 的首项为 $\Pall(d)/(2n)$。激活类包含中心化归一化的 tanh、sigmoid 和 ReLU。证明先从函数距离控制参数在排列和符号变换下的距离，再用互信息估计得到 Fisher 信息矩阵所需的谱性质。条件 I--MMSE 与高斯分部积分给出下界，线性高斯参考分布给出相同首项的上界。维数极限先于样本比极限；证明不要求固定样本比下的风险极限存在，也不依赖后验的局部高斯近似。
\end{abstract}

\section{模型、激活类与定理}
固定 $L\ge1$，$k_0=d$，$k_\ell/d\to\gamma_\ell\in(0,\infty)$，$\gamma_0=1$。
第 $\ell$ 层权重为 $W^{(\ell)}\in\R^{k_\ell\times k_{\ell-1}}$，其元素独立标准高斯。输入 $x\sim N(0,I_d)$ 独立于参数。本文每层使用同一固定激活 $\sigma_\ell\equiv\sigma$，且
\[
\E\sigma(G)=0,\qquad\E\sigma(G)^2=1,\qquad G\sim N(0,1).
\]
记 $W=(W^{(\ell)})_{\ell=1}^L$、$\theta=\theta(W)=(v,W)$，固定 $v_i=1$，网络定义为
\begin{equation}\begin{aligned}
h^{(0)}(x)&=x,\qquad
h^{(\ell)}(x)=\sigma_\ell\!\left(\frac{W^{(\ell)}h^{(\ell-1)}(x)}{\sqrt{k_{\ell-1}}}\right),\\
F_\theta^{(L)}(x)&=\frac{v^{\mathsf T}h^{(L)}(x)}{\sqrt{k_L}}.
\end{aligned}\end{equation}
真实参数 $\theta^0=(v,W^0)$ 由上述先验 $P_0$ 抽取。训练集 $\mathcal D_n=\{(x_\mu,y_\mu)\}_{\mu=1}^n$ 由
\[
y_\mu=F_{\theta^0}^{(L)}(x_\mu)+\sqrt\Delta z_\mu,\qquad
z_\mu\sim N(0,1),\quad\Delta>0
\]
生成。训练和测试输入独立同分布；噪声相互独立，且独立于参数与全部输入。
记 $\Pout(y\mid\lambda)=N(y;\lambda,\Delta)$，真实先验和通道给出的后验为
\[
P(d\theta\mid\mathcal D_n)\propto P_0(d\theta)
\prod_{\mu=1}^n\Pout\!\left(y_\mu\mid F_\theta^{(L)}(x_\mu)\right).
\]
只有隐藏矩阵被学习，读出 $v$ 固定；隐藏参数数目为
\[
\Pall(d)=\sum_{\ell=1}^Lk_{\ell-1}k_\ell,\qquad
c=\lim_d\frac{\Pall(d)}{d^2}
=\sum_{\ell=1}^L\gamma_{\ell-1}\gamma_\ell.
\]
后验预测密度及平均预测 KL 定义为
\begin{equation}\begin{aligned}
p_{\mathcal D_n}(y\mid x)
&=\E_{\theta\mid\mathcal D_n}\Pout\!\left(y\mid F_\theta^{(L)}(x)\right),\\
\epsilon_{\log,d,n}
&=\E_{\theta^0,\mathcal D_n,x}\KL\!\left(
\Pout\!\left(\cdot\mid F_{\theta^0}^{(L)}(x)\right)
\,\middle\|\,p_{\mathcal D_n}(\cdot\mid x)\right).
\end{aligned}\end{equation}
写 $\epsilon_{\log,d}(\alpha):=\epsilon_{\log,d,\lfloor\alpha d^2\rfloor}$。
下文简记 $p=\Pall(d)$、$F_W=F=F_{\theta(W)}^{(L)}$、$\theta'=\theta(V)$，先验平均中的 $W^0$ 简记为 $W$。微分、高斯积分和 OU 耦合仅作用于 $W\in\R^p$；齐次多项式网络另行定义。

所有常数允许依赖固定深度、激活、宽度比例与噪声，不能依赖 $d,n$。

\subsection{激活类}
\newthought*{取下列两类的并集 $\mathcal A_{\rm bal}$。}
两支均满足上述中心化及单位方差条件。
\emph{光滑奇函数支：} $\sigma$ 有界、奇、属于 $C^\infty$，每个固定阶导数有界。
\emph{分段线性支：} 令 $c_0=\sqrt{2/\pi}$，
\begin{equation}\label{eq:pl}
\sigma(z)=az+b(|z|-c_0),\qquad
b\ne0,\qquad a^2+b^2(1-2/\pi)=1.
\end{equation}
两支均全局 Lipschitz。

记 $\operatorname{He}_j$ 为概率论约定的 Hermite 多项式，$\psi_j=\operatorname{He}_j/\sqrt{j!}$、$a_j=\E[\sigma(G)\psi_j(G)]$。
与网络符号一致，$\mu_{\ell,j}=\mu_j=\sqrt{j!}a_j$。以下事实使光滑支无需另加高阶系数假设。
\begin{lemma}[高阶奇系数自动存在]\label{lem:automatic-hermite}
若 $\sigma$ 为有界连续奇函数且 $\E\sigma(G)^2=1$，则存在奇数 $m\ge5$ 使 $a_m\ne0$。
\end{lemma}
\begin{proof}
若结论不成立，奇性和 Hermite 系的完备性给出
$\sigma=a_1\psi_1+a_3\psi_3$ 高斯几乎处处成立；连续性使等式处处成立。
有界奇多项式只能为零，与单位方差矛盾。
\end{proof}
光滑支固定选择上述一个 $m$；分段线性支取 $m=6$，其 $a_6\ne0$ 在下节直接验证。
$m$ 可依赖固定激活，但不随维数或样本数变化；后文常数允许依赖此选择。

\begin{theorem}\label{thm:main}
上述模型中，对任意固定深度 $L\ge1$，任一固定 $\sigma\in\mathcal A_{\rm bal}$ 满足
\begin{equation}\label{eq:main}
\boxed{\lim_{\alpha\to\infty}\limsup_{d\to\infty}
\left|\alpha \epsilon_{\log,d}(\alpha)-\frac c2\right|=0.}
\end{equation}
该结论包含任意固定正斜率的中心化归一化 tanh、logistic sigmoid，以及中心化归一化 ReLU。
\end{theorem}
先取 $d\to\infty$，再取 $\alpha\to\infty$；不要求固定 $\alpha$ 的风险极限存在。

\subsection{tanh、sigmoid 与 ReLU}
对任意固定 $a>0$，令
\[
s_a^2=\E\tanh^2(aG)>0,\qquad \sigma_a(z)=\tanh(az)/s_a.
\]
它是有界光滑奇函数，且方差为一。由 $\tanh'=1-\tanh^2$ 递推，
每阶导数都是 $\tanh(az)$ 的多项式，因而有界，故属于光滑支。
对固定 $b>0$，logistic sigmoid $\ell_b(z)=(1+e^{-bz})^{-1}$ 满足
$\ell_b(z)-1/2=\tanh(bz/2)/2$，其均值为 $1/2$、方差为 $s_{b/2}^2/4$，
中心化归一化后恰为 $\sigma_{b/2}$。

对 ReLU，直接积分给出
\[
\mu_R=\E G_+=\frac1{\sqrt{2\pi}},\qquad
v_R=\operatorname{Var}(G_+)=\frac12-\frac1{2\pi},\qquad
\sigma_R(z)=\frac{z_+-\mu_R}{\sqrt{v_R}}.
\]
由 $z_+=(z+|z|)/2$，它对应式~\eqref{eq:pl} 的
$a=b=1/(2\sqrt{v_R})$。
整个分段线性支都中心化；奇偶正交性给出其方差
$a^2+b^2(1-2/\pi)=1$。
由 $\operatorname{He}_6(z)=z^6-15z^4+45z^2-15$
及 $\E|G|^{2j+1}=2^j j!\sqrt{2/\pi}$，得到
\[
\E[|G|\operatorname{He}_6(G)]=3\sqrt{2/\pi},\qquad
a_6=\frac{3b\sqrt{2/\pi}}{\sqrt{6!}}\ne0.
\]
分段线性支的奇次系数 $a_j$ 在 $j\ge3$ 时均为零；光滑奇函数支的偶次系数均为零。
这正是后文两类展开项所使用的奇偶结构。

\section{先验矩与局部估计}
令 $q_\ell=\|h^{(\ell)}\|^2/k_\ell$、$a(W,X)=\nabla_WF_{\theta(W)}^{(L)}(X)$。
附录引理~\ref{an:moments}--\ref{an:ou} 证明：对每个固定 $r>0$，
\begin{equation}\label{eq:moments}
 \E|q_\ell-1|^r\le C_r d^{-r/2},\qquad
 \E|F_{\theta(W)}^{(L)}(X)|^r+\E\|a(W,X)\|^r\le C_r.
\end{equation}
存在只依赖固定模型参数的 $0<\delta\le1$，使对与 $(W,X)$ 独立的标准高斯 $Z$，有
\begin{equation}\label{eq:gradient-stability}
 W_\tau=e^{-\tau}W+\sqrt{1-e^{-2\tau}}Z,\qquad
 \E\|a(W_\tau,X)-a(W,X)\|^2\le C\tau^\delta\quad(0<\tau\le1).
\end{equation}
对 $0<t\le1/2$，写 $W(t)=\sqrt{1-t}W+\sqrt tZ$，则一阶展开满足
\begin{equation}\label{eq:ou-rem}
 \begin{aligned}
 F_{\theta(W(t))}^{(L)}-F_{\theta(W)}^{(L)}&=\sqrt t\,a(W,\cdot)^TZ+E_t,\\
 \E\|E_t\|_{L_X^2}^2&\le Ct^{1+\delta}.
 \end{aligned}
\end{equation}
常数和指数不依赖扰动参数；维数阈值允许依赖该参数，并在其紧离零的区间上一致。以下局部界均在对应阈值以上使用。

\section{全局函数比较与参数匹配}
本节证明：两个网络的输出接近时，其参数在适当排列和符号变换后也接近。先用一个齐次多项式网络提取实际输出中可比较的分量。
沿用 $\psi_j,a_j$ 的定义，置
\[
N=m^L,\qquad S=1+m+\cdots+m^{L-1},\qquad B=a_m^S\ne0.
\]
以各层齐次激活 $z^m/\sqrt{m!}$、原权重与原读出构造多项式网络，其最高输入次数系数为 $C_N(W)\in\Sym^N\R^d$。定义
\[
H_m(W)=I_N(C_N(W)),\qquad T_W=BH_m(W).
\]
记 $\Pi_N$ 为 $L^2(P_X)$ 到第 $N$ 阶 Hermite 子空间的正交投影。$I_N$ 将对称张量映为该子空间中的多项式：
对 $v\ne0$，$I_N(v^{\otimes N})(X)=\|v\|^N\operatorname{He}_N(\langle v,X\rangle/\|v\|)$，再线性延拓。
其归一化满足
\begin{equation}\label{fp:actual-isometry}
 \langle I_N(C),I_N(D)\rangle_{L_X^2}=N!\langle C,D\rangle.
\end{equation}
张量均取对称张量空间中继承的 Hilbert 范数。

\begin{lemma}[函数内积的统一估计]\label{lem:cross}
若 $\Pi_d$ 为原高斯参数先验，记 $\Gamma(\Pi_d,\Pi_d)$ 为两个边缘均为 $\Pi_d$ 的联合分布集合，则
\begin{equation}\label{eq:cross}
e_d:=\sup_{\pi\in\Gamma(\Pi_d,\Pi_d)}
\E_\pi|\langle T_W,F_{\theta'}^{(L)}-T_V\rangle_{L_X^2}|\longrightarrow0.
\end{equation}
这里允许任意正确先验边缘的耦合，不要求 $W,V$ 独立。
\end{lemma}
\begin{proof}
附录~\ref{app:closure} 给出实际输出的多项式近似，附录~\ref{app:cuts}
逐项比较并控制耦合误差，得到式~\eqref{fp:global-function-pairing}，即本引理。
\end{proof}

对内积估计展开平方，得到齐次多项式网络的距离控制。
对某个正确边缘耦合令
$D=\E\norm{T_W-T_V}^2$、$B_F=\E\norm{F_\theta^{(L)}-F_{\theta'}^{(L)}}^2$。
在原耦合、转置耦合以及两个对角项中分别使用 \eqref{eq:cross}，得
\[
|\E\langle T_W-T_V,F_\theta^{(L)}-F_{\theta'}^{(L)}\rangle-D|\le4e_d.
\]
所以
\[
D\le\sqrt{DB_F}+4e_d\le(D+B_F)/2+4e_d,
\quad D\le B_F+8e_d.
\]
齐次全局匹配定理给出有限可测映射族 $\mathcal G_d$，
\begin{equation}\label{eq:matching}\begin{aligned}
\log|\mathcal G_d|&=O(d\log d)=o(p),\\
\frac1p\E\min_{g\in\mathcal G_d}\norm{W-gV}^2
&\le C\E\norm{H_m(W)-H_m(V)}^2+o_d(1)\\
&\le C\E\norm{F_\theta^{(L)}-F_{\theta'}^{(L)}}^2+o_d(1).
\end{aligned}\end{equation}
误差对耦合一致，逐层匹配的证明见附录 \ref{app:matching}。映射可包含行符号变换；后文仅用这些有限标签控制条件熵，不要求它们保持实际网络输出。

\section{互信息界与后验收缩}
记 $I_{d,n}=I(\theta^0;\mathcal D_n)=I(W;\mathcal D_n)$。Bayes 定义和样本条件独立性给出
\begin{equation}\label{eq:increment}
\epsilon_{\log,d,n}=I_{d,n+1}-I_{d,n},\qquad
\epsilon_{\log,d,n+1}\le \epsilon_{\log,d,n}.
\end{equation}
第二式也可由条件熵递减看出：真实条件噪声熵不随数据量改变，预测熵随观察增加而减小。

\subsection{粗风险上界}
固定充分大的 $u$，令 $n=\lfloor ud^2\rfloor$、$t=(1+n/p)^{-1}$。
此时 $t\to(1+u/c)^{-1}>0$，故维数充分大时 $t$ 位于紧离零的区间；
下一节仅由先验矩与 OU 余项证明的 \eqref{eq:sample-information-upper} 因而适用。又
\[
 \E\log\det(I_p+ntA/\Delta)\le nt\E\tr A/\Delta\le Cp,
 \qquad (n/p)t^{1+\delta}\le1,
\]
所以 $I_{d,n}\le\frac p2\log(1+n/p)+Cp$。
由 \eqref{eq:increment}，$n\epsilon_{\log,d,n}\le I_{d,n}$，因此
\begin{equation}\label{eq:coarse}
 \limsup_d\epsilon_{\log,d}(u)\le C\frac{\log u}{u}
 \qquad(u\text{ 充分大}).
\end{equation}

\subsection{后验参数距离与条件熵}
给定 $\mathcal D_n$ 独立抽取后验样本 $W,V$；平均数据后两者保持原先验边缘。
记 $B_X=F_{\theta(W)}^{(L)}(X)-F_{\theta(V)}^{(L)}(X)$。
以后验预测密度为中间点，Hellinger 三角不等式和 $H^2\le\KL$ 给
\begin{equation}\label{eq:hellinger}
 \E[1-e^{-B_X^2/(8\Delta)}]\le2\epsilon_{\log,d,n},
\end{equation}
其中指数项是两个同方差高斯密度的亲和度。对固定 $q>2$，标量不等式
\[
 z^2\le C_q(1+|z|^q)^{2/q}
          (1-e^{-z^2/(8\Delta)})^{1-2/q}
\]
由分别考察 $|z|\le1$ 和 $|z|>1$ 得到。H\"older 与 \eqref{eq:moments} 因而给
$\E B_X^2\le C_q\epsilon_{\log,d,n}^{1-2/q}$。定义
\[
 r_{d,n}=\frac1p\E\min_{g\in\mathcal G_d}\|W-gV\|^2,
 \qquad r_*(u)=\limsup_d r_{d,\lfloor ud^2\rfloor}.
\]
在 \eqref{eq:matching} 中取此后验耦合，再取 $q=2/\eta$，由 \eqref{eq:coarse} 得
\begin{equation}\label{eq:rate}
 r_*(u)\le C_\eta(\log u/u)^{1-\eta},\qquad0<\eta<1.
\end{equation}

在同一耦合中选择可测最优标签 $\widehat g$，并令
$V_t=\sqrt{1-t}W+\sqrt tU$，其中 $U$ 独立于上述变量且为标准高斯，$0\le t\le1/2$。
给定 $\mathcal D_n$，$V$ 与 $V_t$ 独立；且
\[
 \E\|V_t-\sqrt{1-t}\widehat gV\|^2=p[t+(1-t)r_{d,n}].
\]
条件于 $(\mathcal D_n,V,\widehat g)$ 后用高斯最大熵，再用 Jensen，得到
\begin{equation}\label{info:matched-entropy}
 h(V_t\mid\mathcal D_n)
 \le\log|\mathcal G_d|+\frac p2\log\{2\pi e[t+(1-t)r_{d,n}]\}.
\end{equation}
这里加入标签的条件熵至多为 $\log|\mathcal G_d|$。
$t=0$ 时 $V_0=W$；从 $h(W)=\frac p2\log(2\pi e)$ 减去上述界并用 \eqref{eq:rate}，得
\begin{equation}\label{eq:sample-information-lower}
 \liminf_d\frac{I_{d,\lfloor\alpha d^2\rfloor}}p
 \ge\frac{1-\eta}{2}\log\alpha
       -\frac{1-\eta}{2}\log\log\alpha-C_\eta.
\end{equation}
这里 $\eta\in(0,1)$ 固定，$\alpha$ 充分大。
对 $t>0$，记 $J_n(t)=I(W;V_t\mid\mathcal D_n)$。
从 \eqref{info:matched-entropy} 减去条件噪声熵，得到
\begin{equation}\label{alt:oracle-cost}
 J_n(t)\le\log|\mathcal G_d|+\frac p2\log\left(1+\frac{(1-t)r_{d,n}}t\right)
 \le\log|\mathcal G_d|+\frac{p r_{d,n}}{2t}.
\end{equation}

\section{线性高斯参考分布与互信息界}
对上述 $V_t$，$0<t\le1/2$ 时可反向写为
$W=\sqrt{1-t}V_t+\sqrt t\xi$，其中 $\xi$ 与 $V_t$ 独立标准高斯。
记 $f_W=F_{\theta(W)}^{(L)}$、$a(x)=\nabla f_{V_t}(x)$、
$A=\E_Xa(X)a(X)^T$。对固定 $t$ 及 $d\ge d_0(t)$，\eqref{eq:ou-rem}、\eqref{eq:moments} 给出
\[
 f_W(x)-f_{V_t}(x)=\sqrt t\,a(x)^T\xi+R_t(x),\qquad
 \E\|R_t\|_{L_X^2}^2\le Ct^{1+\delta},\quad
 \E\|a(X)\|^4\le C.
\]


\subsection{相对熵分解与互信息增量}
给定 $V_t,X_{1:j}$，令 $\Phi_j$ 的第 $i$ 行为 $a(X_i)^T$，定义
\[
 Q_j=N\!\left((f_{V_t}(X_i))_{i\le j},S_j\right),
 \qquad S_j=\Delta I_j+t\Phi_j\Phi_j^T.
\]
这些参考分布由同一个虚拟系数 $\zeta\sim N(0,I_p)$ 的线性模型
$f_{V_t}(X_i)+\sqrt t\,a(X_i)^T\zeta+\sqrt\Delta Z_i$ 产生，
因此其样本边缘相容。输入采用真实的独立分布，虚拟系数不作为额外观察。
令 $P_j$ 为给定 $V_t$ 的真实数据分布，定义
\[
 C_j=\E\log\frac{p(Y_{1:j}\mid W,X_{1:j})}
 {q_j(Y_{1:j}\mid V_t,X_{1:j})},\qquad
 B_j=\E_{V_t}\KL(P_j\|Q_j).
\]
$P_j,Q_j$ 包含相同的输入边缘。由相对熵分解和 Markov 链 $V_t-W-\mathcal D_j$，
\begin{equation}\label{info:reference-identity}
 I_{d,j}+J_j(t)=\frac p2\log(1/t)+C_j-B_j.
\end{equation}
参考分布的边缘相容性给 $B_n\ge B_j\ge0$。结合 $J_n(t)\ge0$，同一恒等式给
\begin{equation}\label{alt:block-entropy}
 I_{d,n}\le\frac p2\log(1/t)+C_n,\qquad
 I_{d,n}-I_{d,j}\le C_n-C_j+J_j(t).
\end{equation}

\subsection{交叉熵的非线性误差}
令 $L_j=\sqrt t\Phi_j\xi$、$R_j=(R_t(X_i))_{i\le j}$，并置
$H_j=\E\log\det(I_p+t\Phi_j^T\Phi_j/\Delta)$。

\end{equation}

\begin{lemma}\label{an:moments}

\[
 \mathbb E q_\ell^r\le C_r,\qquad
 \mathbb E|q_\ell-1|^r\le C_r d^{-r/2},\qquad
 \mathbb E|F|^r\le C_r.
\]
单个特征及任意固定个特征的乘积也有任意固定阶矩界；特征可来自不同层或具有上述高斯边缘的耦合输入。
\end{lemma}
\begin{proof}

Lipschitz 性、中心化及单位方差给出全局界
\begin{equation}\label{an:scalar}
 \E|\sigma(\sqrt qG)|^s\le C_s(1+q^{s/2}),\qquad
 |b(q)|+|v(q)-1|\le C|q-1|\quad(q\ge0).
\end{equation}
这里用 $|\sigma(z)|\le C(1+|z|)$ 及
$|\sqrt q-1|(\sqrt q+1)=|q-1|$。
若 $Y_i$ 给定 $\mathcal F$ 后独立且中心化，则
\begin{equation}\label{an:sample}
 \E\left[\left|N^{-1}\sum_iY_i\right|^{2m}\middle|\mathcal F\right]
 \le C_mN^{-m}\max_i\E[|Y_i|^{2m}\mid\mathcal F].
\end{equation}
证明只需展开 $2m$ 次幂：单次出现的指标使该项为零，其余项至多有 $m$ 个指标，
共 $O_m(N^m)$ 项，每项由 H\"older 控制。

给定前层，本层行输出独立同分布为 $\sigma(\sqrt{q_{\ell-1}}G)$。
先由 Jensen 与式~\eqref{an:scalar} 归纳得 $\E q_\ell^{2m}\le C_m$；
再对 $Y_i=(h_i^{(\ell)})^2-v_\ell(q_{\ell-1})$ 用式~\eqref{an:sample}，得到
\[
 \E|q_\ell-1|^{2m}\le C_md^{-m}+C_m\E|q_{\ell-1}-1|^{2m}=O(d^{-m}).
\]
初值 $q_0=d^{-1}\sum_iX_i^2$ 由同一展开给出。
真实全一读出保留其条件均值：
\[
 F=\sqrt{k_L}b_L(q_{L-1})+
 k_L^{-1/2}\sum_i\{h_i^{(L)}-b_L(q_{L-1})\}.
\]
第一项的 $2m$ 阶矩至多 $C_mk_L^m\E|q_{L-1}-1|^{2m}=O(1)$，
第二项由式~\eqref{an:sample} 及正矩界控制。
取 $2m\ge r$ 得全部所需阶数；单个特征矩由条件行估计给出，固定乘积矩由 H\"older 得到。
\end{proof}

\subsection{两类激活的梯度变化}
\begin{lemma}\label{an:gradient-ou}
令 $W'=\sqrt{1-t}W+\sqrt tZ$，$Z$ 独立标准高斯，$0<t\le1/2$。
对任意范数多项式 $P$ 和固定 $t>0$，充分大的 $d$ 下两类激活均满足
\[
 \E[P\|\nabla F(W',X)-\nabla F(W,X)\|^2]
 \le C_Pt^{1/4}.
\]
常数独立于 $t$，维数阈值在紧离零区间上一致。
\end{lemma}
\begin{proof}
反向坐标首先满足，对每个固定 $s>0$，
\begin{equation}\label{an:coordinate}
 \E[P|\beta_{\ell,i}|^{2s}]\le C_{P,s}k_\ell^{-s}.
\end{equation}
终层直接成立。其余层给定 $h=h^{(\ell)}$，下一层高斯矩阵在右乘任何固定 $h$ 的正交变换时分布不变，
$\beta_\ell$ 相应旋转；反向范数只由独立于 $h$ 的下游算子范数控制，
故 $\E[\|\beta_\ell\|^{2s}\mid h]\le C_s$。
沿 $h$ 及垂直于 $h$ 分解，并用后者均匀球面方向的坐标矩，得
\[
 \E[|\beta_{\ell,i}|^{2s}\mid h]
 \le C_s\left(k_\ell^{-s}+|h_i|^{2s}/\|h\|^{2s}\right).
\]
$h=0$ 时分布对全部旋转不变，上式的比值取零。
在 $q_\ell\ge1/2$ 上用特征正矩，在补集上用比值不超过一及
$\Pr(q_\ell<1/2)\le C_md^{-m}$（引理~\ref{an:moments}），
得到无权坐标矩；再对 $P$ 用 Cauchy--Schwarz 即得 \eqref{an:coordinate}。
这里 $U$ 在 $S^{n-1}$ 上均匀时，标准高斯径向分解给
$\E|U_1|^{2s}=\Gamma(s+1/2)\Gamma(n/2)/(\sqrt\pi\Gamma(n/2+s))\le C_sn^{-s}$。

前向递推给
$\|A'_\ell-A_\ell\|_{\rm op}\le\sqrt tP$、
$\|h'^{(\ell)}-h^{(\ell)}\|/\sqrt{k_\ell}\le\sqrt tP$。
记 $D_\ell=\operatorname{diag}(\sigma'(u^{(\ell)}))$。
光滑支中，给定前层，$u'_i-u_i$ 是中心高斯，方差不超过
$C(tq_{\ell-1}+\|h'-h\|^2/k_{\ell-1})$，故
$\E|\sigma'(u'_i)-\sigma'(u_i)|^4\le C\E|u'_i-u_i|^4\le Ct^2$。
分段线性支中，令 $\mathcal G=\{q_{\ell-1},q'_{\ell-1}\ge1/2\}$。
给定前层，在 $\mathcal G$ 上两预激活为中心高斯，相关系数为
$\sqrt{1-t}\langle h,h'\rangle/(\|h\|\|h'\|)$，其夹角满足
\[
 \omega_\ell\le C\left(\sqrt t+
 \frac{\|h-h'\|}{\sqrt{\|h\|\|h'\|}}\right)\le\sqrt tP.
\]
二维高斯极角均匀性给条件异号概率 $\omega_\ell/\pi$，
而 $\Pr(\mathcal G^c)\le C_md^{-m}$，因此
$\Pr(u_iu'_i<0)\le C\sqrt t+C_md^{-m}$。
零预激活在 $\mathcal G$ 上概率为零；在预激活为零处任选有界斜率代表，
其可能贡献由 $\Pr(\mathcal G^c)$ 吸收，仍得
$\E|\sigma'(u'_i)-\sigma'(u_i)|^4\le C\sqrt t+C_md^{-m}$。
于是两支均有
\[
 \E[P\|(D'_\ell-D_\ell)\beta_\ell\|^2]
 \le\sum_i(\E[P^2\beta_{\ell,i}^4])^{1/2}
       (\E|\sigma'(u'_i)-\sigma'(u_i)|^4)^{1/2}
 \le C_P(t^{1/4}+d^{-1}).
\]
从相同读出反向使用
$\lambda'_\ell-\lambda_\ell
 =D'_\ell(\beta'_\ell-\beta_\ell)+(D'_\ell-D_\ell)\beta_\ell$ 及
$\beta'_{\ell-1}-\beta_{\ell-1}
 =A_\ell^T(\lambda'_\ell-\lambda_\ell)+(A'_\ell-A_\ell)^T\lambda'_\ell$。
算子范数仅改变允许的 $P$，矩阵差项为 $O(t)$，故固定深度归纳保留上述界。
最后展开梯度块公式~\eqref{an:gradient} 的乘积差，得到同一界。
固定 $t>0$ 后取 $d^{-1}\le t^{1/4}$ 即得结论；光滑支全程为 $O(t)$。
\end{proof}

\subsection{高斯分部积分的定义域}
\begin{lemma}\label{an:sobolev}
设 $g:\mathbb R^p\to\mathbb R^p$ 属于高斯 $W^{1,2}$，也允许值域带独立的外部 $L^2$ 变量。定义
$\delta g=\operatorname{div}g-W\cdot g$。则其闭扩张满足
\[
 \mathbb E(\delta g)^2
 =\mathbb E\|g\|^2+\mathbb E\sum_{i,j}(\partial_i g_j)(\partial_jg_i)
 \le\mathbb E\{\|g\|^2+\|Dg\|_{\rm HS}^2\}. \tag{A13}\label{an:divergence}
\]
\end{lemma}
\begin{proof}
对紧支撑光滑场，展开 $(\delta g)^2$ 并逐坐标两次分部积分得到等式，
再用 $2uv\le u^2+v^2$ 得不等式。
一般 $g$ 先乘径向截断 $\chi_R$（$\|\nabla\chi_R\|_\infty\le C/R$），再在紧集卷积。
尾部及截断导数由 $g,Dg\in L^2$ 控制；高斯密度在紧集有正上下界，
故所得紧支撑光滑场在高斯 $W^{1,2}$ 中收敛于 $g$。
上述不等式使散度在 $L^2$ 中收敛；对测试函数分部积分识别极限为分布散度，
并将等式传至极限。外部变量先用简单函数逼近再积分。
\end{proof}


\section{对称张量的 Gram 矩阵界}\label{tg:section}
固定整数 $m\ge5$、$q\ge1$。设单位行 $w_1,\ldots,w_k$ 满足
$\max_{i\ne j}|\langle w_i,w_j\rangle|\le\delta$。对多重指标 $r\in\mathbb N^k$、$|r|=q$，定义
\[
 P_r=\operatorname{Sym}\bigotimes_jw_j^{\otimes mr_j},\qquad
 Z_{r,i}=\operatorname{Sym}\bigotimes_jw_j^{\otimes(mr_j-\mathbf1_{j=i})},\quad r_i>0.
\]
$\operatorname{Sym}$ 是全部槽位排列的平均，范数取继承的 Hilbert 张量范数。
令 $\mathcal F$、$\mathcal F^-$ 分别以 $P_r$、$Z_{r,i}$ 为列。则
\begin{equation}\label{tg:frame}
 cI\preceq\mathcal F^*\mathcal F\preceq CI,\qquad
 cI\preceq(\mathcal F^-)^*\mathcal F^-\preceq CI,
\end{equation}
只要 $\delta+\sum_{h=1}^q k^h\delta^{mh-2}=o(1)$；$c,C>0$ 只依赖 $m,q$。

证明使用精确配对公式
\[
 \left\langle\operatorname{Sym}\bigotimes_{a=1}^Ru_a,
 \operatorname{Sym}\bigotimes_{a=1}^Rv_a\right\rangle
 =\frac1{R!}\sum_{\pi\in\mathfrak S_R}\prod_a\langle u_a,v_{\pi(a)}\rangle.
\]
若列的标签重数为 $(n_j)$，对角中的无错配项为 $\prod_jn_j!/R!$，
它有只依赖固定总次数的正下界；其余对角项为 $O(\delta)$。
纯列的重数为 $mr$，删除列的重数为 $mr-e_i$。两族内部不同列具有不同重数，
后者因模 $m$ 的余数唯一确定 $i$。同一有限支撑内的非对角项至少含一个错配，
故为 $O(\delta)$，此类列的数量有固定上界。
若另一列含 $h\ge1$ 个新标签，它至少有 $mh-1$ 个槽必须与不同标签配对，
故内积至多 $C\delta^{mh-2}$；这样的列至多 $Ck^h$ 个。
因此每个 Gram 矩阵与上述正对角矩阵之差的绝对行和至多
$C(\delta+\sum_hk^h\delta^{mh-2})=o(1)$。对称矩阵的 Schur 界即得 \eqref{tg:frame}。

行长在固定正区间时，各列只乘有固定上下界的标量，结论不变。
对按输入宽度归一化的高斯行，行长一致趋于一且
$\delta=O(\sqrt{\log d/d})$、$k\asymp d$，故扰动趋零；事件补集概率可取任意固定负幂。

\section{计算树展开的张量估计}
\label{fp:core-section}

本节估计计算树展开所产生的系数张量。所有空间取实 Euclidean 空间及其 Hilbert
张量积；$\|\cdot\|_{\mathrm{HS}}$ 表示张量 Hilbert 范数，
$\operatorname{Flat}_{S|S^c}$ 表示按索引位置分组的矩阵化。
下文的一个\emph{模式}是系数张量的一个索引位置；不同模式可以取相同数值，
也可以由同一个原节点索引复制得到。各层宽度在 $[cd,Cd]$ 内，
深度、多项式次数、涉及的层数和操作次数均固定。
记 $\delta_d=L_d/\sqrt d$、$\rho_d=L_d/d$，其中 $L_d$ 是可随有限次
运算增大的固定对数幂。概率估计分别在每个网络的高斯边缘下建立；
相应事件成立后，以下估计对两网络间任意耦合均成立。

\subsection{系数张量的范数界与线性求和}
\label{fp:core-invariant-section}

\paragraph{计算树的系数张量。}
从原网络的归一化读出向量开始，按原计算树自上而下展开。
每个当前模式代表一个原节点的指标，模式 $p$ 的一次展开为
\begin{equation}
 J e_a=\bigotimes_{t=1}^{q}u_a^{(t)},\qquad q\geq1,
 \qquad \|U_t\|_{\mathrm{op}}\leq C,
 \quad U_t=(u_a^{(t)})_a .
 \label{fp:core-replacement}
\end{equation}
因子 $e_a$ 表示保留父索引。也可在某节点停止展开，保留该节点以下的整个张量映射。
同一父索引产生的多个因子仍共享该索引。依赖两网络的映射只能作用在停止展开的叶节点上，
每个映射输出一个张量因子；输出多个因子的非线性映射须先按原计算树展开并纳入系数张量。

\refstepcounter{lemma}
\paragraph{引理~\thelemma（展开算子的范数）。}
把上述展开作用于全部保留因子的复合映射称为展开算子。任意固定次数的此类复合均有界。具体地，令
$\Delta_q e_a=e_a^{\otimes q}$，则
$J=(\bigotimes_tU_t^*)\Delta_q$，而 $\Delta_q$ 为等距嵌入，故
$\|J\|\leq\prod_t\|U_t\|$。
同样，若 $F_j:e_a\mapsto f_a^{(j)}$ 有界，则共享指标的乘积特征
$e_a\mapsto\bigotimes_jf_a^{(j)}$ 有界。
也可从 Gram 矩阵看出：其 Gram 为 $\mathop{\circ}_jF_j^*F_j$，
且对半正定矩阵 $X\preceq bI,Y\succeq0$，
$X\circ Y\preceq b\operatorname{diag}(Y)$。

\refstepcounter{lemma}
\paragraph{引理~\thelemma（展开保持范数和切片界）。}
\label{fp:core-closure}
设 $E$ 有 $r$ 个模式，满足
\begin{equation}
 \|E\|_{\mathrm{HS}}\leq C,\qquad
 \max_{p,a}\|E(e_a\text{ 于模式 }p,\cdot)\|_{\mathrm{HS}}\leq\delta_d,
 \qquad
 \|\operatorname{Flat}_{S|S^c}E\|_{\mathrm{op}}\leq\delta_d
 \quad(\varnothing\ne S\ne[r]).
 \label{fp:core-invariants}
\end{equation}
除 $r=q=1$ 外，一次展开保持这些结论，常数可增大固定倍数。

\emph{证明。}
Hilbert 范数由展开算子有界立即得到。考虑新核的一个非平凡分割。
若新产生的整个块在同一侧，该分割就是旧核的非平凡分割两侧复合有界算子。
若新块跨越两侧，在两侧暂时保留正交父指标 $a$。中央算子按 $a$ 分块对角，
每块为旧切片 $E(e_a,\cdot)$ 的某个矩阵化，算子范数至多其 Hilbert 范数
$\delta_d$。再施加两侧的行映射即得结论，包含旧核只有一个模式的情形。

除 $r=q=1$ 外，新张量至少有两个索引；其坐标切片满足
$\|E_{\rm new}(e_a\text{ 于 }p,\cdot)\|_{\rm HS}
\le\|\operatorname{Flat}_{\{p\}|\{p\}^c}E_{\rm new}\|_{\rm op}\le C\delta_d$，
故已由刚证明的矩阵化界控制。$r=q=1$ 时没有非平凡矩阵化，仍须另用
高斯连续乘积的坐标估计。
\hfill$\square$

\paragraph{线性路径上的高斯矩阵乘积。}
令 $A_\ell$ 为独立、按输入宽度平方根归一化的高斯矩阵。
任意固定连续积 $P=A_b\cdots A_a$ 满足
\begin{equation}
 \|P\|_{\mathrm{op}}\leq C,\qquad
 \max_{ij}|P_{ij}|\leq C_M\sqrt{\log d/d}
 \label{fp:core-product-event}
\end{equation}
且可使失败概率为 $O(d^{-M})$。算子范数由各层相乘得到；条件于
$Q=A_{b-1}\cdots A_a$ 后，$(A_bQ)_{ij}$ 是方差
$\|Qe_j\|^2/n_{b-1}\leq C/d$ 的高斯变量，尾界与有限并合即得第二式。
读出向量乘连续高斯积的坐标最大值也由条件于最后尚未暴露的
高斯因子得到相同估计。于是首次分叉以前的 $r=q=1$ 链可单独处理；
首次分叉以后模式数不减，前一引理处理所有后续步骤。
停止展开或施加恒等映射无须概率估计。有限原树核全部满足
\eqref{fp:core-invariants}。这里的一元链仅由原网络矩阵组成。

\paragraph{外部映射及重建。}
对系数张量的每个索引施加有界、输出一个因子的映射，保持所有非平凡矩阵化界，
但不必保持坐标切片界。完成一次收缩后，应从原计算树重新构造系数张量：
在对应父节点处停止展开，将已算出的跨网络收缩矩阵作为该叶节点的外部映射。
不同深度均可停止展开，但根到叶的内部路径仍只含原网络矩阵。
以下坐标切片估计始终用于这样重建的原系数张量。

\refstepcounter{lemma}
\paragraph{引理~\thelemma（线性路径的精确求和）。}
\label{fp:core-linear}
令底层行向量为 $b_j$，父层矩阵为 $U=(u_{aj})$，并设
$z_a=\sum_ju_{aj}b_j$。一个父节点有 $t$ 个线性子槽以及次数
$q_1,\ldots,q_h\geq2$ 的非线性子槽，则其未对称化张量严格等于
\begin{equation}
 z_a^{\otimes t}\otimes
 \bigotimes_{\ell=1}^h\left(\sum_j u_{aj}b_j^{\otimes q_\ell}\right).
 \label{fp:core-linear-identity}
\end{equation}
\emph{证明。} 给每个有序子槽分配独立求和指标，再逐个使用多线性性；
线性槽的求和正好是 $z_a$，非线性槽保留原式。固定次数的对称化为槽排列的
平均，可在求和后施加。纯一元链因此是归一化矩阵的普通乘法，无额外
$\sqrt{\text{宽度}}$ 因子。式中的所有非线性因子仍共享 $a$，不得替换为独立阵列。
坐标掩码一般不与此求和交换；必须先在全部张量因子的展开算子中证明掩码删除的误差为
$o(1)$，再恢复原张量并用本等式。\hfill$\square$

\subsection{同层索引的旋转与双索引收缩}
\label{fp:core-fresh-section}

\paragraph{概率工具。}
若确定性矩阵 $A_j:\mathbb R^u\to\mathbb R^v$ 满足
$\max\{\|\sum_jA_j^*A_j\|,\|\sum_jA_jA_j^*\|\}\leq s^2$，则独立
标准高斯系数满足
\[
 \mathbb P\!\left(\left\|\sum_jg_jA_j\right\|>t\right)
 \leq2(u+v)\exp[-t^2/(2s^2)].
\]
这是有限维矩阵高斯级数界。其自伴膨胀为
$H_j=\left(\begin{smallmatrix}0&A_j^*\\ A_j&0\end{smallmatrix}\right)$，
$\|\sum_jH_j^2\|\leq s^2$；矩阵高斯 Laplace 界
$\mathbb E\operatorname{tr}\exp(\lambda_{\rm c}\sum_jg_jH_j)
\leq\operatorname{tr}\exp(\lambda_{\rm c}^2\sum_jH_j^2/2)$
配合 $\lambda_{\rm c}=t/s^2$ 得到上尾，对负号重复即得双侧界。
以下只用其维数为固定 $d$ 次幂时的 $O(s\sqrt{\log d})$ 结论。

\refstepcounter{lemma}
\paragraph{引理~\thelemma（同层旋转下的坐标切片）。}
\label{fp:core-fresh}
在同一原网络层，同时展开所有尚未停止的模式。选择当前高斯层矩阵的列索引所对应的模式 $p$：
它随该层矩阵右乘正交矩阵而旋转，不包括保留的父索引 $e_a$ 或其确定性复制。
再选不同模式 $q$；总模式数为 $N\ge3$。令 $E$ 为尚未施加外部映射的系数张量。
以任意高多项式概率有
\begin{equation}
 \max_j\sup_{\|v\|\leq1}
 \|E(e_j\text{ 于 }p,v\text{ 于 }q,\cdot)\|_{\mathrm{HS}}
 \leq\rho_d.
 \label{fp:core-fresh-slice}
\end{equation}
此时尚未展开更低层；在较高层停止展开的索引保持不动。

\emph{证明。}
条件于一个满足 \eqref{fp:core-invariants} 的核，置
$A_jv=E(e_j,v,\cdot)$。直接按矩阵元素求和可知
\[
 \left\|\sum_jA_j^*A_j\right\|
 =\|\operatorname{Flat}_{\{q\}|\{q\}^c}E\|^2\leq\delta_d^2,
 \qquad
 \left\|\sum_jA_jA_j^*\right\|
 =\|\operatorname{Flat}_{\{p,q\}|\{p,q\}^c}E\|^2\leq\delta_d^2.
\]
第二分割非平凡恰好使用 $N\geq3$。前述级数界、
$\|g\|\geq c\sqrt d$ 的高斯尾界，给均匀球面向量 $u=g/\|g\|$
的收缩范数至多 $C_M\delta_d\sqrt{\log d/d}$。
对独立 Haar 矩阵的各列作并合，不要求列独立。
将当前高斯层矩阵右乘该正交矩阵不改变分布，并同时旋转对应其列索引的全部模式。
未固定模式上的旋转是等距算子；若 $q$ 也对应该层的列索引，对 $v$ 的上确界仍不变。
故 Haar 后的估计转回原坐标即为所需式。所有核和模式只有固定有限种。
\hfill$\square$

\refstepcounter{lemma}
\paragraph{推论~\thelemma（重复坐标项的范数）。}
\label{fp:core-collision}
若至少两个模式对应同一高斯层矩阵的列索引，且总模式数至少三，则其中至少两个索引取相同值的部分满足
\begin{equation}
 \|E_{\mathrm{coll}}\|_{\mathrm{HS}}^2
 \leq\sum_{p<q}\sum_j\|E(e_j\text{ 于 }p,e_j\text{ 于 }q,\cdot)\|^2
 \leq C d\rho_d^2=o(1).
 \label{fp:core-collision-bound}
\end{equation}
在该层引入这些模式时先估计这一部分，再将误差通过有界的展开算子传递，最终误差仍为 $o(1)$。
这里只删除不同原节点的索引恰取相同值的项；同一父索引的确定性复制仍须保留。

\refstepcounter{lemma}
\paragraph{推论~\thelemma（原始系数的矩与坐标界）。}
\label{fp:core-raw-collision-expectation}
设 $C_R$ 为固定深度、固定次数、零常数高斯多项式网络的原始 $R$ 阶系数，
允许保留有序因子、对称化及固定有限个展开项的和。对每个固定 $0<p<\infty$，
$\sup_d\mathbb E\|C_R\|_{\rm HS}^p<\infty$。
若全部索引由同一原高斯层的列索引产生（不含父索引的确定性复制），宽度为 $k\asymp d$，则对任意固定 $M$，
以失败概率 $O(d^{-M})$ 同时有
\begin{equation}\label{fp:core-coordinate-bounds}
 \max_{i_1,\ldots,i_R\ \text{互异}}|(C_R)_{i_1\cdots i_R}|
 \le L_d k^{-R/2},\qquad
 \|\operatorname{Coll}_R C_R\|_{\rm HS}^2\le L_d/d\quad(R\ge3),
\end{equation}
其中 $L_d$ 为固定对数幂；$R\ge3$ 时还有 $\mathbb E\|\operatorname{Coll}_R C_R\|_{\rm HS}^2=o(1)$。
为证矩界，将原展开算子的范数相乘，再用归一化高斯矩阵算子范数的固定阶矩和 H\"older 不等式，
不需乘积中各因子独立。列置换不变性使互异坐标的二阶矩相同，故每个至多
$C/(k)_R=O(k^{-R})$。每个坐标是固定次数 $D_0$ 的高斯多项式，
$\|Z\|_u\le(u-1)^{D_0/2}\|Z\|_2$；取 $u=C_M\log d$，在 $e\|Z\|_u$ 处用 Markov
并对至多 $k^R$ 个坐标作并集界，即得第一界。
第二界由同层旋转引理~\ref{fp:core-fresh}及式~\eqref{fp:core-collision-bound}应用于每个有序展开项得到；
固定有限个排列与展开项的和不改变阶数。矩界及 Cauchy--Schwarz 给
$\mathbb E\|\operatorname{Coll}_R C_R\|^2\le L_d/d+
(\mathbb E\|C_R\|^4)^{1/2}O(d^{-M/2})=o(1)$。
这里只控制原始系数，尚未施加依赖两个网络的外部映射。

\refstepcounter{lemma}
\paragraph{引理~\thelemma（两个父索引的收缩与二阶非对角项）。}
\label{fp:core-two-parent}
在一个未加掩码的原始切口中，设不同父模式 $p,q$ 的所选线性槽与教师行
收缩后给出 $\eta_i,\zeta_i$，其 $\ell^2$ 范数有界。模式 $p$ 是本层引入的模式，
且删去所选槽后仍有非空残余；所有完整残余家族均为有界算子。
若总模式数 $R\ge3$，完整剩余源张量 $S_i$ 满足 $\|S_i\|_{\rm HS}\le C\rho_d$。
事实上，令 $B_ie_b=\zeta_{ib}f_b$，其中 $f_b$ 是第二个源索引删除所选槽后的完整
剩余特征；无剩余槽时取 $f_b=1$。逐点有
$\|B_i\|_{\rm HS}^2=\sum_b|\zeta_{ib}|^2\|f_b\|^2\le C$。
将原始切片 $E(e_a,\cdot)$ 看作从 $q$ 索引到其余索引的算子 $T_a$。
式~\eqref{fp:core-fresh-slice}给 $\|T_a\|_{\rm op}\le\rho_d$，故
$\|T_aB_i^*\|_{\rm HS}\le C\rho_d$。
保留第一个索引 $a$，乘 $\eta_{ia}$ 后平方求和，再施加完整剩余特征，即得所需界。
这些估计逐点成立，不要求 $B_i$ 与网络独立。

若 $R=2$、两个索引属于同一原网络层，且删除所选槽后两个模式均有非空残余，式~\eqref{fp:core-coordinate-bounds}给
$\max_{a\ne b}|D_{ab}|\le L_d/d$；保留两标签后施加完整特征映射，非对角部分满足
\[
 \|S_i^{\rm off}\|\le C\Big(\sum_{a\ne b}|D_{ab}\eta_{ia}\zeta_{ib}|^2\Big)^{1/2}
 \le CL_d/d.
\]
两种界与教师切片和 $C\sqrt d\|C\|$ 配对均为 $o(1)\|C\|$。
该二阶结论不控制对角或坐标--向量切片；不同层的两个索引亦不适用，须保留根索引另行处理。
同一父索引的确定性复制不是两个不同父节点，不能使用本估计。

\subsection{保留剩余因子的局部收缩估计}\label{fp:core-residual-section}

\begin{lemma}[残余家族与局部收缩]\label{fp:core-residual-flat}
\label{fp:core-local-contraction}\label{fp:core-split-hit}
\label{fp:core-hybrid-Rge2}\label{fp:core-hybrid-R1}
设 $F_a=\sum_bU_{ab}Q_b$，$U$ 为原高斯层或连续积，$Q_b$ 为正次数原计算树；
停止叶允许有界自适应单槽映射。完整家族 $e_a\mapsto F_a$ 有界，
且当 $F_a$ 至少有两个输出因子时，每个非平凡输出矩阵化均至多 $C\delta_d$。
同指标乘积家族也有界。

固定一个有 $m\ge2$ 个因子的教师节点。设它接触源父索引 $a$ 的一个旧收缩因子 $\ell_a$，
该源列除 $\ell_a$ 外至少还有一个完整因子，且 $G_{ia}=\langle a_i,\ell_a\rangle$ 满足
$\|G\|_{\max}\le\kappa_d$、$\sup_i\|G_{i,:}\|_2\le C$。
所有其他完整残余均保留，其他源父标签的映射不依赖 $a$。
教师、源原系数分别为 $C,D$。则完整配对满足：
\begin{equation}\label{fp:local-unified}
 |\mathrm{pairing}|\le
 \begin{cases}
 C\kappa_d\|C\|\|D\|,&\text{本叶还接触另一源父节点},\\
 C\sqrt d\,\alpha(D)(\delta_d+\kappa_d)\|C\|,&\text{本叶全在父节点 }a,
 \end{cases}
\end{equation}
第二行中，多模核取 $\alpha(D)=\|\operatorname{Flat}_{a|\mathrm{rest}}D\|$；
一模核 $D=d$ 取 $\alpha(D)=\|d\|_\infty$，但要求除 $\ell_a$ 外至少有两个完整因子。
因此 $\alpha(D),\delta_d,\kappa_d\le L_d/\sqrt d$ 时两类配对均为 $o(1)$。
\end{lemma}
\begin{proof}
残余家族是原展开算子与 $U^*$ 的复合。对固定 $a$，以 $U_{a,:}$ 为读出重建核；
式~\eqref{fp:core-product-event} 和引理~\ref{fp:core-closure} 给各非平凡矩阵化界，
外部单槽映射保持它们。乘积家族由对角等距嵌入保持有界。

若还接触另一父节点，先将来自其他父节点的全部 $t\ge1$ 个槽映到正交教师标签 $i$。
该映射的伴随为 $(A^*)^{\otimes t}\Delta_t$，故有界。
保留 $a,i$ 后乘 $G_{ia}$ 的算子范数为 $\|G\|_{\max}$；再在每个 $i$ 块内
提升本列的剩余家族并完成其余配对。至少一个其他完整因子保证该家族有界；
即使它被本叶完全收缩，也留下 $\ell^2$ 有界的标量家族，而非无权删除标签。
全部为完整张量上的有界算子，无部分迹或逐 $i$ 求和损失。

以下本叶全在 $a$。固定 $i$，其余因子收缩后形成家族 $H_j$：完全收缩者是
$\ell^2$ 有界向量；部分收缩者的列范数至多 $C\delta_d$；完全保留者仍为有界家族。
因为本叶还有 $m-1\ge1$ 个槽，或者存在部分收缩者，或者至少一个因子完全收缩。
令 $B_ie_a=G_{ia}\bigotimes_jH_j e_a$。前一情形用 $\|G_{i,:}\|_2$，
后一情形用 $\|G\|_{\max}$ 及完全收缩向量的 $\ell^2$ 范数，均得
$\|B_i\|_{\rm HS}\le C(\delta_d+\kappa_d)$。
多模情形把其余模式展开记为 $T$，于是
$\|(B_i\otimes T)D\|_{\rm HS}\le C\alpha(D)(\delta_d+\kappa_d)$。

一模情形至少有两个 $H_j$。若有部分收缩者 $H_u$，取 $j\ne u$，利用
$H_j^*H_j\preceq CI$ 和 $\operatorname{diag}(H_u^*H_u)\preceq C\delta_d^2I$，
Schur 乘积序给乘积家族算子范数至多 $C\delta_d$；作用于 $(d_aG_{ia})_a$ 即得
$C\|d\|_\infty\delta_d$。否则移入一个完全收缩向量 $\eta$，
$\|(d_aG_{ia}\eta_a)_a\|_2\le C\|d\|_\infty\kappa_d$，其余至少一个家族仍有界。
最后仅用一次 $\sum_i\|C(e_i,\cdot)\|\le C\sqrt d\|C\|$，即得第二行。
\end{proof}
二阶非对角核可写成 $D-\operatorname{diag}D$ 分别应用此界；对角核的算子范数和
Hilbert 范数均不增。此论证不适用于其他掩码，也未声称二阶对角为小量。
一模且仅有一个其他因子的边界使用引理~\ref{fp:permanent-mark} 中的保留索引子程序。

\section{实际激活的输入混沌投影与多项式近似}
\label{fp:actual-section}\label{app:closure}
固定 $N>2$，记 $A_\ell=W^{(\ell)}/\sqrt{k_{\ell-1}}$、$u=c_0=\sqrt{2/\pi}$，并置
\begin{equation}\label{fp:actual-PQ}
 P_N(z)=\sum_{j=1}^N\frac{a_jz^j}{\sqrt{j!}},\qquad
 Q_N(z)=\sum_{j=1}^N\frac{b_jz^j}{\sqrt{j!}},\qquad
 b_j=\E[(\sigma(G)^2-1)\psi_j(G)].
\end{equation}
$Q_N$ 只用于分段线性分支。对使用原矩阵 $A_1,\ldots,A_t$、读出
$k_t^{-1/2}\mathbf1$、输出激活 $A$ 及下层激活 $p_1,\ldots,p_{t-1}$ 的多项式网络，
先取形式 $N$ 次对称系数、再施加 $I_N$，记为 $R_{t,N}[A;p]$；
只保留每个隐藏层全部索引互异的展开项所得函数记为 $J_{t,N}[A;p]$。
输入坐标不受限制，故 $J_{1,N}=R_{1,N}$；$;P_N$ 表示所有下层均用 $P_N$。
这些定义允许固定辅助变量取系数。形式取系数在 $I_N$ 之前，
不包括更高形式次数经高斯收缩得到的第 $N$ 阶项。

\subsection{条件展开及其误差}
\label{fp:actual-finite-support}\label{fp:actual-positive}
本节所有 $L^p$ 均对原高斯先验及输入。引理~\ref{an:moments} 和
第~\ref{an:appendix} 节的矩阵尾界给出
\begin{equation}\label{fp:actual-local-interface}
 \|q_\ell-1\|_p=O(d^{-1/2}),\qquad
 \sup_d\E\left[\|A_\ell\|_{\rm op}^p+|h_i^{(\ell)}|^p\right]<\infty.
\end{equation}
固定乘积也有全部固定阶矩，允许重复索引；$q_\ell\notin[1/2,2]$ 的概率因此
可取任意多项式衰减。首层行写为 $A_{1,i:}=s_iv_i$，$\|v_i\|=1$。
高斯长度和内积尾界及矩阵尾界给出概率至少 $1-d^{-M}$ 的事件
\begin{equation}\label{fp:actual-row-event}
 \max_i|s_i-1|+\max_{i\ne j}|\langle v_i,v_j\rangle|
 \le C_M\sqrt{\log d/d},\qquad \|A_1\|_{\rm op}\le C_M.
\end{equation}

\paragraph{首层有限乘积。}
若固定 $t$ 个单位向量的 Gram 矩阵 $G$ 满足 $\|G-I\|\le\eta$，
且 $\max_i|s_i-1|\le\delta$，则
\begin{equation}\label{fp:actual-ridge-projection}
 \left\|\Pi_N\prod_{i=1}^t\sigma(s_iv_i\cdot X)
 -\sum_{\substack{n_i\ge1\\\sum_i n_i=N}}
 \frac{\prod_i a_{n_i}}{\sqrt{\prod_i n_i!}}
 I_N\!\left(\Sym\bigotimes_i(s_iv_i)^{\otimes n_i}\right)
 \right\|_2\le C_{t,N}(\eta+\delta).
\end{equation}
为证明它，在 $\Sym^N\operatorname{span}\{v_i\}$ 中取
$B_n=I_N(\Sym\bigotimes_i v_i^{\otimes n_i})$，$n_i\ge0$、$\sum n_i=N$。
其 Gram 矩阵是 $G$ 的多项式，在 $G=I$ 附近一致可逆；投影坐标等于
该矩阵的逆乘 $\E[\prod_i\sigma(s_iZ_i)B_n(Z;G)]$，$Z\sim N(0,G)$。
对高斯密度按协方差求导，导数由可积的多项式乘高斯密度控制；
$B_n$ 的系数多项式依赖于 $G$，激活至多线性增长且 Lipschitz，
故上述有限个积分关于 $(G,s)$ 局部 Lipschitz。在 $(I,\mathbf1)$ 处，
独立性给系数 $\prod_i a_{n_i}/\sqrt{\prod_i n_i!}$，$a_0=0$ 删除零次数。
再把 $v_i$ 改为 $s_iv_i$ 即得上式，包含完整非多项式乘积的一切收缩项。
特别地 $t>N$ 时近似和为空。

\paragraph{逐层条件展开。}
令 $\chi=\sigma^2-1$，$f=\sigma$，或在分段线性支取 $f=\chi$。
给定前层特征 $h\in\R^k$，置 $q=\|h\|^2/k$、
$c_r^f(q)=\E f(\sqrt qG)\psi_r(G)$。条件 Hermite 展开使用
\begin{equation}\label{fp:actual-multivariate}
 f(W_i\cdot h/\sqrt k)=\sum_{r\ge0}c_r^f(q)
 \sum_{|\alpha|=r}\sqrt{\frac{r!}{\alpha!}}
 (h/\|h\|)^\alpha\prod_j\psi_{\alpha_j}(W_{ij}).
\end{equation}
$q=0$ 按实际随机变量的连续值解释。由于 $|f(x)|\le C(1+x^2)$，
$q\mapsto f(\sqrt q\,\cdot)$ 为 $[1/2,2]$ 到 $L^2(\gamma)$ 的连续映射，
其 Hermite 尾一致趋零。区间外用完整条件二阶矩 $C(1+q^2)$、
式~\eqref{fp:actual-local-interface} 及任意高阶小概率控制。
对固定多个不同行输出，条件独立性使逐因子差的条件平方期望等于
一个尾的平方期望乘其余因子的二阶矩；故乘积也可截断，无须单行尾的高阶收敛。

固定 $r$ 后，换元 $y=\sqrt qG$ 并对密度及 $\psi_r(y/\sqrt q)$ 求导，
可知 $c_r^f(q)q^{-r/2}$ 在 $[1/2,2]$ 上 Lipschitz，故可替换为 $c_r^f(1)$，
误差趋零。区间外，精确有限分量的固定阶条件矩由 $c_r^f(q)$ 的增长及
高斯 Hermite 矩界控制，替换后的多项式由特征矩控制；H\"older 与小概率界
再次给 $o(1)$。这里不使用负阶 $q$ 矩，也不对 $f$ 求导。

若固定展开含 $T$ 个前层输出因子，重复索引赋值至多 $C_Tk^{T-1}$ 个，
而平方归一化因子为 $k^{-T}$。先合并给出相同权重 Hermite 单项式的排序
（重数只依赖 $T$），由正交性及固定乘积矩，删去这些赋值的平方误差为 $O_T(k^{-1})$。
剩余权重变量均只出现一次，Hermite 单项式成为普通权重乘积。
隐藏层零阶系数为零（奇支）或 $bu(\sqrt q-1)=O_{L^p}(d^{-1/2})$
（分段线性支），也可在固定乘积中删去；输出层零阶项稍后单独保留。

这些局部误差经整个读出求和仍趋零。事实上，在任一阶段先合并相同的上方权重单项式，写
\begin{equation}\label{fp:actual-phase-bound}
 \Delta=\sum_g\beta_g\Psi_g(W_{>\ell})E_g(W_{\le\ell},X),
 \quad \sum_g\beta_g^2\le C_{\rm fixed},
 \quad \E\|\Delta\|_{L^2_X}^2
 \le C_{\rm fixed}\sup_g\E\|E_g\|_{L^2_X}^2.
\end{equation}
这里不同 $\Psi_g$ 正交、每个变量次数为一，且独立于下层；
其系数平方和有界，是因为读出 $k_t^{-1/2}$ 与每条边的输入宽度 $-1/2$ 次幂，
平方后恰好抵消输出及各自由索引的数目，合并重数只依赖固定展开结构。
从输出层向下依次取有限截断，使总 $L^2$ 尾任意小，再用
\eqref{fp:actual-ridge-projection} 处理互异的首层激活因子。
式~\eqref{fp:actual-row-event} 与上述平方和界使首层误差趋零；
坏事件上固定乘积和固定次数多项式矩至多有固定多项式维数增长，可由任意高阶失败概率吸收。
若某节点次数超过 $N$，其不同子树至少给出超过 $N$ 个互异首层因子，投影近似为空。
故各层截断达到 $N$ 后所得表达式稳定；先令 $d\to\infty$、再令截断误差趋零，得到
\begin{equation}\label{fp:actual-positive-local}
 \Pi_NF_{t,+}^f=J_{t,N}[H_f;P_N]+o_{L^2}(1),\qquad
 H_\sigma=P_N,\quad H_\chi=Q_N.
\end{equation}
其中 $F_t^f=k_t^{-1/2}\sum_i f((A_th^{(t-1)})_i)$，$F_{t,+}^f$ 是减去给定前层的条件均值后所得函数。
$t=1$ 时直接展开每行：$\E f(sG)\psi_N(G)$ 关于 $s\in[1/2,2]$ Lipschitz，
故可换成 $s^N\E f(G)\psi_N(G)$。首层向量映射 $e_i\mapsto v_i^{\otimes N}$ 有界，
因为其 Gram 绝对行和至多 $\sum_j|\langle v_i,v_j\rangle|^2\le\|V\|_{\rm op}^2$。
读出范数为一，逐行 $O(\max_i|s_i-1|)$ 的系数误差因此给总 $o(1)$，坏事件同上。

\subsection{二次对角与索引恢复}
\label{fp:actual-quadratic}\label{fp:actual-telescope}
对完整上部多项式，$r\ge3$ 的重复坐标由推论~\ref{fp:core-raw-collision-expectation}
控制；该结论不能直接用于受索引限制的 $J$。二次项满足如下补充公式：
输入宽度为 $k$、深度为 $t$、各层激活 $p_j(z)=\sum_{r\ge1}g_{j,r}z^r$ 时，
\begin{equation}\label{fp:actual-diag}
 \E\|\operatorname{Diag}D_2-\lambda I_k/\sqrt k\|_{\rm HS}^2=O(k^{-1}),
 \qquad \lambda=g_{t,2}\prod_{j<t}g_{j,1}^2\sqrt{k_t/k}.
\end{equation}
正次网络二次项恰有一个二次节点；节点位于第 $j$ 层时，去掉固定标量后为
$D_2^{(j)}=B_j^T\operatorname{diag}(v_j)B_j$，其中
$B_j=A_j\cdots A_1=C_jA_1$，$v_j=A_{j+1}^T\cdots A_t^T\mathbf1/\sqrt{k_t}$。
令 $H_j=C_j^T\operatorname{diag}(v_j)C_j$。条件于 $H_j$，$A_1$ 各列独立，故
\[
 \E((D_2^{(j)})_{ii}\mid H_j)=\tr H_j/k,\qquad
 \E\sum_i|(D_2^{(j)})_{ii}-\tr H_j/k|^2=2\E\|H_j\|_{\rm HS}^2/k=O(k^{-1}).
\]
末步用 $\|H_j\|_{\rm HS}\le\|C_j\|_{\rm op}^2\|v_j\|$。
$j<t$ 时 $v_j$ 独立于 $C_j$，均值零、协方差 $I/k_j$，所以
$\E|\tr H_j|^2=k_j^{-1}\E\sum_a\|C_{j,a:}\|^4=O(1)$。
$j=t$ 时 $\tr H_t=\|C_t\|_F^2/\sqrt{k_t}$，均值为 $\sqrt{k_t}$，且方差为 $O(k_t^{-1})$：
对独立下一层 $A$，
\[
 \E(\|AC\|_F^2\mid C)=\frac{k_{\rm new}}{k_{\rm old}}\|C\|_F^2,
 \qquad \operatorname{Var}(\|AC\|_F^2\mid C)
 =\frac{2k_{\rm new}}{k_{\rm old}^2}\|CC^T\|_{\rm HS}^2=O_{L^1}(1),
\]
由全方差公式沿固定长度乘积归纳即得 $\operatorname{Var}\|C_t\|_F^2=O(1)$。
乘回标量证明\eqref{fp:actual-diag}，亦适用于辅助变量取系数。

\begin{lemma}[去掉索引互异限制]\label{fp:actual-restoration}
对固定零常数多项式 $A,p_1,\ldots,p_{t-1}$，$t\ge2$，以 $[p]_r$ 表示 $z^r$ 系数，有
\begin{equation}\label{fp:actual-restoration-exact}
 J_{t,N}[A;p]=R_{t,N}[A;p]-[A]_2\sqrt{k_t/k_{t-1}}
 R_{t-1,N}[p_{t-1}^2;p_{<t-1}]+o_{L^2}(1).
\end{equation}
\end{lemma}
\begin{proof}
令 $\gamma=[A]_2\sqrt{k_t/k_{t-1}}$。在取形式系数前，从完整网络减去
$\gamma k_{t-1}^{-1/2}\sum_i y_i y_i$，其中 $y$ 为第 $t-1$ 层输出；
两个因子保留各自的父索引，虽数值相等仍视为两个槽。
由下向上依次施加第 $s=1,\ldots,t-1$ 层的互异限制。
此时上方仍是完整多项式。$s<t-1$ 时可将减项的上部识别为根激活
$p_{t-1}^2$ 的浅网络；两项的确定二次对角标量分别为
\[
 [A]_2\prod_{j=s+1}^{t-1}[p_j]_1^2\sqrt{k_t/k_s}
 =\gamma[p_{t-1}^2]_2\prod_{j=s+1}^{t-2}[p_j]_1^2\sqrt{k_{t-1}/k_s},
\]
其中用了 $[p^2]_2=[p]_1^2$。$s=t-1$ 时减项就是原上部系数的确定二次对角。
两项在代入共同下层之前相消；二次余项由\eqref{fp:actual-diag}、更高次重复坐标
由原始碰撞估计给 $o_{L^2}(1)$，一次没有重复坐标。
共同下层映射由固定多个原展开映射、坐标投影和对称化组成，其范数由
有限个 $\|A_j\|_{\rm op}$ 的固定次幂控制；已施加的全局坐标限制是收缩，
无须分解为逐槽映射。固定次数高斯多项式的矩等价将系数误差提升为 $L^4$ 小量，
再用 H\"older 与\eqref{fp:actual-isometry} 得到每步函数误差 $o_{L^2}(1)$，不需独立性。
最后的第 $t-1$ 层限制删掉减项的两个相等父标签，原项成为 $J$，即得结论。
\end{proof}

\subsection{条件均值递推与全部浅层修正}
\label{fp:actual-mean}\label{fp:actual-masked-formula}
分段线性支对 $f=\sigma,\chi$ 有精确公式
\begin{equation}\label{fp:actual-root-mean}
 \E f(\sqrt qG)=\lambda_f(q-1)+c_f(\sqrt q-1)^2,
 \quad (\lambda_\sigma,c_\sigma)=(bu/2,-bu/2),\quad
 (\lambda_\chi,c_\chi)=(1,b^2u^2).
\end{equation}
由 $|\sqrt q-1|\le|q-1|$ 和引理~\ref{an:moments}，乘读出因子 $\sqrt{k_t}$ 后的
二次余项仍为 $O_{L^2}(d^{-1/2})$。$t\ge2$ 时线性项等于
$\lambda_f\sqrt{k_t/k_{t-1}}F_{t-1}^\chi$；$t=1$ 时为
$\lambda_f\sqrt{k_1}(q_0-1)$，被 $N>2$ 投影精确消去。
高斯密度按方差求导还给
$[H_f]_2=\frac12\E[f(G)(G^2-1)]=\lambda_f$。
结合\eqref{fp:actual-positive-local} 与引理~\ref{fp:actual-restoration}，直接得到
\begin{equation}\label{fp:actual-cancel-recursion}
 \Pi_NF_t^f=R_{t,N}[H_f;P_N]+\lambda_f\sqrt{k_t/k_{t-1}}
 \bigl(\Pi_NF_{t-1}^\chi-R_{t-1,N}[P_N^2;P_N]\bigr)+o_{L^2}(1),
 \quad t\ge2,
\end{equation}
初值为 $\Pi_NF_1^f=R_{1,N}[H_f]+o_{L^2}(1)$。
置 $U_N=[Q_N-P_N^2]_{\deg\le N}$、
$E_t=\Pi_NF_t^\chi-R_{t,N}[P_N^2;P_N]$，则
$E_1=R_{1,N}[U_N]+o_{L^2}(1)$ 且
$E_t=R_{t,N}[U_N;P_N]+\sqrt{k_t/k_{t-1}}E_{t-1}+o_{L^2}(1)$。
固定深度归纳及 $f=\sigma$ 的递推遂给
\begin{equation}\label{fp:actual-U}
 \mathcal Q_N=R_{L,N}[P_N;P_N]
 +\mu'\sum_{t=1}^{L-1}\sqrt{k_L/k_t}\,R_{t,N}[U_N;P_N],
 \qquad \mu'=bu/2,
\end{equation}
\begin{equation}\label{fp:actual-final}
 \boxed{\E_W\|\Pi_NF_L-\mathcal Q_N\|_{L^2_X}^2\longrightarrow0,\qquad N>2.}
\end{equation}
奇支各条件均值及二次系数均为零，同样证明给出 $\mu'=0$，无需 $\chi$。
所有多项式零常数，逐层代入不降低形式次数，故超过 $N$ 次的项均可删去。
以下比较深层项和式~\eqref{fp:actual-U} 中全部浅层修正；单网络 $L^2$ 误差将用
Cauchy--Schwarz 转移到任意先验边缘耦合。


\section{计算树展开项的逐层比较}\label{fp:history}\label{app:cuts}
固定 $L$、$N=m^L$；奇激活取奇数 $m\ge5$，分段线性激活取偶数 $m\ge6$。
令 $\delta_d,\kappa_d\le K_d/\sqrt d$，$K_d$ 是可随有限次运算增大的固定对数幂。
每个节点选一个正次单项式并将孩子排序，得到有序展开项；先固定其 $N$ 个输入因子
与齐次教师项的槽配对，最后对有限种选择和至多 $N!$ 个排列求和。
教师、源各自的原高斯事件先固定，跨网络映射随后可任意依赖两网络。

\begin{lemma}[有序树消元]\label{fp:lowest-cut}\label{fp:permanent-mark}\label{fp:core-spectator}
高度 $t<L$、根为任意零常数多项式 $A$、下层为 $P$ 的每个有序 $N$ 次展开项，
与高度 $L$ 的齐次教师项的每种固定槽配对均为 $o(1)$。
高度 $L$ 的普通 $P$ 展开项中，除每个节点次数均为 $m$ 的项外，结论相同。
\end{lemma}
\begin{proof}
对源高度 $t$ 作强归纳；固定 $t$ 时按教师高度处理尚未完成的节点。
已完成高度 $h-1$ 的收缩作为第 $h$ 步的一个因子，刚完成的输出留到下一步才使用。
所有这些跨网络收缩放在源侧；教师始终保留原始齐次系数 $C$。
固定当前教师行 $i$ 后，其余完整教师因子的范数至多 $K_d\|C(e_i,\cdot)\|$，故
\begin{equation}\label{fp:teacher-slices}
 \sum_i\|C(e_i,\cdot)\|\le\sqrt{k}\|C\|\le K_d\sqrt d.
\end{equation}
每次在源原树的停止节点重建系数核，再施加有界完整特征映射；不同高度的输出分别
保留标签。同一父索引的复制也保留，绝不当作独立高斯列索引或可删除的碰撞。
纯一元链先用引理~\ref{fp:core-linear} 精确求和，所有收缩均作用于恢复后的无掩码项。

\paragraph{完整收缩与保留索引子程序。}
一次合法完整收缩使用同一源节点的 $m$ 个完整线性孩子，其矩阵
$M_{ia}=\prod_{j=1}^m\rho^{(j)}_{ia}$ 满足
\begin{equation}\label{fp:metric-schur}
 \sup_i\sum_a|M_{ia}|+\sup_a\sum_i|M_{ia}|\le K_d,
 \qquad \|AM\|_{\max}\le K_d\kappa_d,
\end{equation}
其中各 $\rho^{(j)}$ 的行列 $\ell^2$ 范数有界，$A$ 为下一教师原高斯矩阵。
前式由 H\"older 得到，后式只用 $\|A\|_{\max}\le\kappa_d$，不要求 $M$ 独立于 $A$。
完整孩子条件本身尚不保证高度增长：同一原节点可先产生多个旧输出。
以下每次复用旧输出前，先排除其来源节点仍有其他因子的项；通过此过滤后，
父收缩才严格位于已完成孩子之上。因此高度 $h$ 的教师子树在每条源路径上至少占用 $h$ 个原节点。

二模式的特殊状态写成
\begin{equation}\label{fp:marked-state}
 \sum_{a,b}E_{ab}\ell_a\otimes Q_b,
 \qquad \|E\|_{\rm HS}\le K_d,\quad\|E\|_{\rm op}\le\delta_d.
\end{equation}
$\ell$ 是一个完整当前因子，$Q$ 为原始高度至多 $L-1$ 的完整子树；
$E$ 是原二阶停止核或沿原一元链传播的核，故满足引理~\ref{fp:core-closure}。
所有尚未收缩因子和下一步输出都包含在此式中。
保留 $a$，暂缓从 $\ell$ 到教师根的整条路径，按高度处理路径之外的教师子树。
每次先检查旧输出复用，再推进选中路径；旧 $G$ 只来自已认证完整的较低教师高度步骤。
此时选一个 $Q$ 后代就有至少两个模式，选两个不同后代就有至少三个模式。
因此旧收缩因子仍带残余时，式~\eqref{fp:local-unified} 的混合父界或多模界适用；
两个不同源父节点相遇时，引理~\ref{fp:core-two-parent} 的 $R\ge3$ 界适用。
混合父旧输出的完整配对直接为 $O(K_d\kappa_d)=o(1)$；
同父旧输出与双父切片情形，固定教师行后的完整余张量分别至多
$K_d\delta_d(\delta_d+\kappa_d)$ 和 $K_d/d$，由\eqref{fp:teacher-slices} 均为 $o(1)$。
在余下项中，将选中路径推进到最低的非单因子节点；若它们来自同一非线性孩子，
就应更早停止，故每步只能完整使用该节点的 $m$ 个线性孩子。
每完成一次收缩，已完成教师节点数增加；其间线性路径游标只沿有限原树向上移动。
旧输出的上述过滤保证每次父收缩严格高于孩子，故高度计数适用于 $Q$。
此子程序既不调用本引理的结论，也不再次调用二模式子程序。

根部其余 $m-1$ 个教师孩子高度均为 $L-1$。若 $Q$ 更浅，消元不能完成；
若其高度恰为 $L-1$，这 $m-1$ 次收缩只能在 $Q$ 原根 $b$ 处完成。
记矩阵为 $M^{(j)}$，$G^{(j)}=AM^{(j)}$；每个 $G^{(j)}$ 的列 $\ell^2$ 范数有界，
最大条目至多 $K_d\kappa_d$。保留整条暂缓路径及所有残余，去掉这 $m-1$ 个已知因子后有
\[
 R_b=\sum_aE_{ab}\ell_a\otimes Q_b^{\rm res},\qquad
 \sum_b\|R_b\|\le K_d\sqrt d\|E\|_{\rm HS}.
\]
固定最终教师行时其完整路径范数有界。结合归一化读出的 $O(d^{-1/2})$ 最大值，
对两个 $G$ 用列 $\ell^2$ 界、其余 $m-3$ 个用条目界，整个配对至多
\begin{equation}\label{fp:marked-terminal}
 K_d\|E\|_{\rm HS}\kappa_d^{m-3}=o(1).
\end{equation}
这里同时使用全部 $m-1$ 个因子，未将一个小标量核单独乘任意余张量。

\paragraph{消元顺序与全部边界。}
首先排除旧收缩因子再次使用时仍带其他因子的项。
当前教师节点若还接触另一源父节点，使用\eqref{fp:local-unified} 第一行；
否则原核为多模，或一模但另有至少两个因子时，使用第二行。
一模且没有其他因子时允许继续。仅有一个其他因子时，若它也是旧收缩因子，
本列只有两个当前位置，不能容纳 $m\ge5$ 个槽，已属第一种情况。
若它是原始完整孩子，正次数及其他输出标签的保留迫使上方只有原一元链；
故系数 $d$ 满足 $\|d\|_2\le K_d$、$\|d\|_\infty\le\kappa_d$。
展开该孩子第一条边，得到\eqref{fp:marked-state}，其中
$E_{ab}=d_aU_{ab}$、$Q$ 高度至多 $t-1\le L-1$，由保留索引子程序排除。
因此以后复用的收缩因子均已包含其来源节点的全部因子。

然后沿当前教师节点所选的 $m$ 条源路径向上推进；一元节点精确求和，
在最低的非单因子节点停止，较高路径暂缓，较低路径推进至同高或其更低分支。
选中最低父节点的当前层列索引随该层右旋转，且删除选中槽后仍有残余。
若选中因子涉及不同父节点，原核有至少三个模式时用双父引理的 $R\ge3$ 情形。
二模式中若一父为单因子、另一父有残余，使用保留索引子程序。
若两父都有残余，则两者同属当前原高斯层：较高层的旧收缩因子已不带其他因子，
而暂不处理的较高输出另保留标签，会使模式数至少三。
因此二阶非对角项适用双父引理，其两边非空残余条件同时满足。

二阶对角由式~\eqref{fp:actual-diag} 化为
$\lambda_sI/\sqrt{k_s}$ 加 $o_{L^2}(1)$ 的 Hilbert--Schmidt 误差；
误差通过完整有界特征映射传递。主项将两个原父标签识别，将根单项式相乘，
保持全部有序孩子与原教师槽分配，成为高度 $s<t$、读出 $1/\sqrt{k_s}$ 的前缀项。
根次数可有任意奇偶性，故由加强的源高度归纳控制；已有收缩展开回原始等式后也无残缺子树。
$s=0$ 时只有输入次数二，不贡献 $N>2$。
不同层的二模式不使用此对角公式：较高的旧因子若参与，就用保留索引子程序；
若不参与，本次仅涉及一个父节点。两个模式若都为较高旧输出，每个至多含
$m^{L-1}$ 个原输入因子，合计小于 $N$；一个若已达到高度 $L$，它已独自耗尽 $N$ 个因子。
确定性复制只按原输入因子数计数，不增加原父节点数。

所有坏情形已被排除，余下 $m$ 条路径必在同一最低源父节点相遇，且来自
$m$ 个不同的完整线性孩子，否则应在更低非线性孩子处停止。
因此每步都是前述合法完整收缩；有限原树的边只被推进有限次，过程终止。
高度计数排除 $t<L$。$t=L$ 时每条路径必须取等号，故无额外一元节点且每节点次数恰为 $m$；
除预先分出的全 $m$ 项外均矛盾。二阶融合已严格降低源高度，归纳从 $t=1$ 起闭合。
全 $m$ 项的系数、权重与归一化保持原值，两类激活的结论同时成立。
\end{proof}


\subsection{从确定性事件到任意先验耦合}\label{fp:coupling-uniform}
\newthought*{先控制各自先验下的误差，再对耦合取上确界。}
固定次数和深度时，有序展开项、因子配对、停止展开的位置选择、合并高度及收缩步骤的种类数均与 $d$ 无关。分别在两个边缘上同时施加所有前述原始高斯事件，然后才允许跨网络收缩矩阵依赖两个网络。在共同好事件上，各项估计均为确定性 $o(1)$。在坏事件上，用固定次数系数的有界边缘高阶矩和 H\"older 不等式；坏事件概率可取任意所需多项式阶。该控制不依赖耦合，因此对所有先验边缘耦合一致。

设 $T_W$ 为与激活对应、带完整标量系数的全 $m$ 平衡齐次输入混沌项，
$\mathcal Q_N(V)$ 为式~\eqref{fp:actual-U} 的多项式近似。
上述逐展开项估计给出 $\mathbb E_\pi|\langle T_W,\mathcal Q_N(V)-T_V\rangle|=o(1)$，对耦合一致。
又 $T_W$ 位于第 $N$ 输入混沌，故 $\langle T_W,F_V\rangle=\langle T_W,\Pi_NF_V\rangle$。
剩余配对期望由 Cauchy--Schwarz 不等式控制为
$\|T_W\|_{L^2(W,X)}\|\Pi_NF_V-\mathcal Q_N(V)\|_{L^2(V,X)}=o(1)$；
这里使用有界边缘二阶矩与式~\eqref{fp:actual-final}，同样对耦合一致。因此
\begin{equation}\label{fp:global-function-pairing}
 \boxed{\displaystyle
 \sup_{\pi\in\Gamma(\mathsf P_W,\mathsf P_V)}
 \mathbb E_\pi|\langle T_W,F_V-T_V\rangle|\longrightarrow0.}
\end{equation}

\section{齐次比较网络的全局匹配}\label{hm:section}\label{app:matching}
固定 $L$、整数 $m\ge5$、$k_0=d$ 及 $k_s\asymp d$。本节的齐次网络为
$h^{(s)}=c_m(W^{(s)}h^{(s-1)}/\sqrt{k_{s-1}})^{\odot m}$，
$c_m=(m!)^{-1/2}$，$h^{(0)}=x$，输出为
$\mathcal H_W=k_L^{-1/2}\sum_i h_i^{(L)}$，参数数目为 $p=\sum_s k_{s-1}k_s$。
将 $h^{(s)}$ 换成形式变量 $Y$ 后的输出记为 $P_s(Y)$，其次数为 $q_s=m^{L-s}$。
写 $P_s(Y)=\langle C_s,Y^{\otimes q_s}\rangle$，使用继承的 Hilbert 范数，即
\begin{equation}\label{hm:coeff-norm}
 \|C_r(P)\|^2=\sum_{|\nu|=r}\frac{\nu!}{r!}|p_\nu|^2
 \quad\text{若 }P(Y)=\sum_\nu p_\nu Y^\nu;
 \qquad C_L=k_L^{-1/2}{\bf1}.
\end{equation}
\begin{theorem}[齐次网络的有限标签全局匹配]\label{hm:global-matching}
存在确定性的参数对齐映射族 $\mathcal G_d$，使任意两个具有原独立标准高斯先验边缘的网络满足
\begin{equation}\label{hm:main}
 \log|\mathcal G_d|=O(d\log d)=o(p),\qquad
 p^{-1}\mathbb E\min_{g\in\mathcal G_d}\|W-gV\|_F^2
 \le K\mathbb E\|C_0(W)-C_0(V)\|^2+o_d(1).
\end{equation}
$K$ 只依赖固定深度、次数和宽度比例，余项对耦合一致。
右侧亦可换为 $K\mathbb E\|\mathcal H_W-\mathcal H_V\|_{L^2(N(0,I_d))}^2+o_d(1)$。
\end{theorem}

\paragraph{共同典型事件。}
归一化权重行的长度一致趋于一，单位行自身相干度
$\eta_d=O(\sqrt{\log d/d})$，行阵列算子范数有界，列平方和有正的常数上下界。
在此高斯集中事件上，第~\ref{tg:section} 节的两个 Gram 界适用。
令 $B_{s,i}=\partial_iP_s$、$w_j=W^{(s+1)}_{j:}/\sqrt{k_s}$；链式法则给出
\[
 B_{s,i}(Y)=mc_m\sum_jw_{ji}(w_j\cdot Y)^{m-1}
 B_{s+1,j}\big(c_m(w\cdot Y)^{\odot m}\big).
\]
展开后的项正是 \eqref{tg:frame} 中带标签 $(\nu,j)$ 的删除因子张量。
固定次数下单项式系数范数与 \eqref{hm:coeff-norm} 等价，故从
$B_{L,i}=k_L^{-1/2}$ 反向归纳，得到
$\|C(B_{s,i})\|^2\asymp\sum_j w_{ji}^2\|C(B_{s+1,j})\|^2\asymp k_s^{-1}$。
因此完整切片满足
\begin{equation}\label{hm:mode}
 a/k_s\le\|C_s(e_i,\cdot)\|^2=q_s^{-2}\|C(\partial_iP_s)\|^2\le b/k_s.
\end{equation}
特别地 $\|C_s\|\le C$。令 $\mathrm{SF}$ 投影到平方自由坐标，
其正交基中的系数为 $c_I=p_I/\sqrt{q_s!}$（$|I|=q_s$）。
式~\eqref{fp:core-coordinate-bounds} 及原始系数界~\ref{fp:core-raw-collision-expectation}
给出同一边缘高概率事件上的
\begin{align}
 \max_I|c_I|^2&\le C(\log d)^{D_s}k_s^{-q_s},\qquad D_s=m+\cdots+m^{L-s},\label{hm:dense}\\
 \|(I-\mathrm{SF})C_s\|^2&\le L_d/d=o(1).\label{hm:collision}
\end{align}
终层取 $D_L=0$，第二项为零。这里 $L_d$ 可取适当的对数幂。
这些事件在后文内部层的协调行列变换下保持：权重的集中量不变，系数仅重排、变号。
故后面的数据依赖对齐仍可逐点使用全部估计。

\paragraph{全局对称与分组提升。}
在一层写归一化行 $w_i=r_i a_i$、$\|a_i\|=1$，置
$U_Ae_i=a_i^{\otimes m}$、$\overline C_s=\operatorname{diag}(r_i^m)^{\otimes q_s}C_s$。
有 $\|U_A\|\le\|A\|^m\le C$ 及精确代入式
\begin{equation}\label{hm:substitution}
 C_{s-1}=c_m^{q_s}\operatorname{Sym}_{mq_s}U_A^{\otimes q_s}\overline C_s.
\end{equation}
半径一致趋于一，故 $\overline C_s-C_s=o(1)$，且逐坐标的半径乘子一致为 $1+o(1)$，故 \eqref{hm:mode}--\eqref{hm:collision} 仍成立。
下证对任意两个典型网络、$q=q_s$，一致有
\begin{equation}\label{hm:sym-group}
 \|C_{s-1}^A-C_{s-1}^B\|^2
 =\lambda_{m,q}\|G_A-G_B\|^2+o(1),\qquad
 \lambda_{m,q}=\frac{q!}{(mq)!}.
\end{equation}
这里 $G_A=U_A^{\otimes q}\overline C_s^A$、$G_B=U_B^{\otimes q}\overline C_s^B$。
定义平方自由基的分组与全对称提升
$Q_I=\sqrt{q!}\operatorname{Sym}_q\bigotimes_{i\in I}a_i^{\otimes m}$、
$E_I=\sqrt{(mq)!/(m!)^q}\operatorname{Sym}_{mq}\bigotimes_{i\in I}a_i^{\otimes m}$。
式~\eqref{hm:substitution} 将该基送到 $\sqrt{\lambda_{m,q}}E_I$。
置 $\rho_{ij}=\langle a_i,b_j\rangle$、$\tau=d^{-1/8}$。
二部图 $|\rho_{ij}|>\tau$ 的度数至多 $D=\lceil C/\tau^2\rceil$；
在各侧连接共享邻居的顶点，度数至多 $D^2$。不含这种边的 $q$ 元集称为好集。
对好 $I=\{i_1,\ldots,i_q\}$ 和 $\sum_av_a=m$，
\begin{equation}\label{hm:mixed-column}
 \sum_j\prod_a|\rho_{i_aj}|^{v_a}\le
 \begin{cases}C,&|\operatorname{supp}v|=1,\\ C\tau,&|\operatorname{supp}v|\ge2.\end{cases}
\end{equation}
确实，每列至多一个因子超过 $\tau$；按最大因子的位置分区，
从乘积保留该因子的平方，或保留它与另一因子各一幂。
余下至少一幂不超过 $\tau$，所留两幂的和由平方和界或 Cauchy--Schwarz 控制。
配对计数给出
\[
 \langle E_I^A,E_J^B\rangle
 =\sum_{\substack{e_{ab}\ge0\\\sum_be_{ab}=\sum_ae_{ab}=m}}
 \frac{(m!)^q}{\prod_{a,b}e_{ab}!}\prod_{a,b}\rho_{i_aj_b}^{e_{ab}}.
\]
$e=mP$（$P$ 为排列矩阵）的项之和正是 $\langle Q_I^A,Q_J^B\rangle$。
其余项至少一列混合两个标签；把对 $J$ 的求和扩大到所有有序指标并分离各列，
式~\eqref{hm:mixed-column} 给出绝对行和 $O(\tau)$。交换两侧给出列和。
令 $K_{\rm act}=(\langle E_I^A,E_J^B\rangle)_{I,J}$、
$K_{\rm grp}=(\langle Q_I^A,Q_J^B\rangle)_{I,J}$ 为好集上的交叉 Gram，则
\begin{equation}\label{hm:cross-schur}
 \|K_{\rm act}-K_{\rm grp}\|_{\rm op}\le C\tau.
\end{equation}
自身 Gram 是同一论证的特例，因 $\eta_d<\tau$ 而所有集合均好。
坏集至多 $C_qD^2k^{q-1}$ 个，其总系数能量由 \eqref{hm:dense} 至多为
$Cd^{-1/2}(\log d)^{D_s}=o(1)$，这一计数对依赖两个网络和系数的图也成立。
两个提升有界，故删除坏集及重复指标的误差均为 $o(1)$；对三个 Gram 块应用
\eqref{hm:cross-schur} 并恢复完整系数，即得 \eqref{hm:sym-group}。
$q=1$ 时无拆槽，等式精确成立。

\paragraph{单层匹配与系数恢复。}
令 $M=U_A^*U_B=(\rho_{ij}^m)$、$\rho_i=\max_j|\rho_{ij}|$。
一行中超过 $\tau$ 的至多 $D$ 个学生行，其自身 Gram 范数至多 $1+D\eta_d$；
其余项保留平方并提出 $\tau^{m-2}$，得到
\begin{equation}\label{hm:cross-power}
 \sum_j|M_{ij}|\le\rho_i^{m-2}+e_d,\qquad
 \sum_i|M_{ij}|\le1+e_d,\qquad
 e_d=C(D\eta_d+\tau^{m-2})=o(1).
\end{equation}
写完整有序系数数组 $C=\overline C_s^A=(c_I)$、$D'=\overline C_s^B=(d_J)$，
并令 $\mathsf K=(M/(1+e_d))^{\otimes q}$。其绝对行列和 $r_I,s_J\le1$，且
\begin{equation}\label{hm:transport-identity}
 \begin{aligned}
 \mathcal E&:=\|C\|^2+\|D'\|^2-2\langle C,\mathsf K D'\rangle\\
 &=\sum_I(1-r_I)c_I^2+\sum_J(1-s_J)d_J^2
 +\sum_{I,J}|\mathsf K_{IJ}|(c_I-\operatorname{sgn}(\mathsf K_{IJ})d_J)^2.
 \end{aligned}
\end{equation}
因 $U_A^*U_A,U_B^*U_B=I+o_{\rm op}(1)$（非对角绝对行和至多 $k\eta_d^m=o(1)$），
有 $\mathcal E=\|G_A-G_B\|^2+o(1)$。
$r_I$ 为各槽行和的乘积，故用第一槽的缺额及完整切片下界 \eqref{hm:mode} 得
$k_s^{-1}\sum_i(1-\rho_i^{m-2})\le C\mathcal E+o(1)$。
固定 $\rho_0\in(1/\sqrt2,1)$。因 $2\rho_0^2>1+\eta_d$，绝对相关超过 $\rho_0$ 的边构成部分双射。
补为排列并逐行定向后，对角相关非负；未匹配比例由上述缺额控制，
已匹配行用 $1-x^m\le m(1-x^{m-2})/(m-2)$，便得
\begin{equation}\label{hm:row-loss}
 k_s^{-1}\sum_i\big(\ell_i+\|a_i-b_i\|^2\big)\le C\mathcal E+o(1),
 \qquad \ell_i=1-\langle a_i,b_i\rangle^m.
\end{equation}
同步重排系数并乘相应 $m$ 次符号以保持 $G_B$ 不变。
此时 $\mathsf K_{II}\ge0$，$1-\mathsf K_{II}\le\sum_{a=1}^q(\ell_{i_a}+e_d)$。
保留 \eqref{hm:transport-identity} 的对角平方项，并用完整切片上界，得到
\[
 \|C-D'\|^2\le\mathcal E+\sum_I(1-\mathsf K_{II})(c_I-d_I)^2
 \le\mathcal E+\frac{Cq}{k_s}\sum_i(\ell_i+e_d)\le C\mathcal E+o(1).
\]
去掉半径因子，并结合 \eqref{hm:sym-group}，得到
\begin{equation}\label{hm:recover}
 \|C_s^A-C_s^B\|^2+k_s^{-1}\sum_i\|w_i-v_i\|^2
 \le C\|C_{s-1}^A-C_{s-1}^B\|^2+o(1).
\end{equation}
此处行已对齐；半径误差由
$\|r_ia_i-t_ib_i\|^2\le C\|a_i-b_i\|^2+C\max_j(|r_j-1|+|t_j-1|)^2$ 吸收。

\paragraph{深度归纳及任意耦合。}
内部层的有符号行排列写作 $S_s=D_sP_s$，令
$R_s=D_s^{m\bmod2}P_s$。将下一层的列同时变为 $V^{(s+1)}R_s^{-1}$，
保持整个齐次函数及当前 $C_{s-1}$ 不变。
故自下而上在各内部层应用 \eqref{hm:recover}，以后对齐不影响已完成的层，
固定深度下每层平均归一化行距离均至多 $C\|C_0(W)-C_0(V)\|^2+o(1)$。
终层 $q_L=1$ 只需 \eqref{hm:row-loss}：若定向符号为 $\varsigma_i$，
在分析表示中把学生系数改为 $\varsigma_i^m t_i^m/\sqrt{k_L}$，原张量仍不变，
切片界仍成立。无需恢复终层系数；终层符号只是参数标签，不要求保持函数。
取所有映射
$(gV)^{(1)}=S_1V^{(1)}$、$(gV)^{(s)}=S_sV^{(s)}R_{s-1}^{-1}$，则
$\log|\mathcal G_d|\le\sum_s(\log(k_s!)+k_s\log2)=O(d\log d)$。
按固定标签次序打破平局可使选择可测。将各层界以 $k_{s-1}k_s/p$ 加权相加，
即得典型事件上的 \eqref{hm:main}。
任意先验耦合的联合失败概率由两个边缘失败概率之和控制；坏事件上用恒等对齐及
$p^{-1}\|W-V\|^2\le2p^{-1}(\|W\|^2+\|V\|^2)$，
再用归一化高斯平方范数的一致二阶矩和 Cauchy--Schwarz，期望贡献为一致的 $o(1)$。
最后 $N=m^L$ 阶高斯混沌的等距性给出
$N!\|C_0(W)-C_0(V)\|^2=\|\Pi_N(\mathcal H_W-\mathcal H_V)\|_{L^2}^2
\le\|\mathcal H_W-\mathcal H_V\|_{L^2}^2$，证明函数距离版本。\hfill$\square$

\paragraph{实际函数的参数比较。}
式~\eqref{fp:global-function-pairing} 和正文极化恒等式将实际函数距离控制转为齐次系数距离控制，
再应用定理~\ref{hm:global-matching}。行符号仅用于有限标签对齐，不要求保持实际激活的函数值。

\section{条件 I--MMSE 与统一平滑测试场}\label{app:information}
记 $f=F_{\theta(W)}^{(L)}(X)$、$a=\nabla_Wf$、
$\mathcal I_W=\E_Xaa^T/\Delta$、$Z=(\mathcal D_n,X)$。
另取独立标准高斯 $N$，令 $U_s=\sqrt sf+N$、
$m(s)=\E[(f-\E[f\mid Z,U_s])^2]$。真实高斯输出通道及条件 I--MMSE 给
\begin{equation}\label{eq:conditional-immse}
 \epsilon_{\log,d,n}=I(f;U_{1/\Delta}\mid Z)
 =\frac12\int_0^{1/\Delta}m(s)\,ds.
\end{equation}
为验证第二等号，条件于 $Z=z$，令 $q_v$ 为 $f+\sqrt vN$ 的密度。
高斯热方程、分部积分及条件期望公式给
\[
 \frac d{dv}h(q_v)=\frac12\mathcal J(q_v),\qquad
 \partial_y\log q_v(y)=\frac{\E[f\mid z,f+\sqrt vN=y]-y}{v}.
\]
正交分解得到 $\operatorname{mmse}(f\mid z,f+\sqrt vN)=v-v^2\mathcal J(q_v)$。
从 $h(q_v)$ 减去噪声熵后微分，再令 $v=1/s$，互信息导数即为 MMSE 的一半。
上述计算先对截断信号及有界空间成立；由 $\mathcal J(q_v)\le1/v$
和信号二阶矩的一致可积性，可在任意 $0<s_0\le s\le s_1$ 上去掉截断。
最后用 $I(f;U_s\mid Z)\le s\E f^2/2$、$m(s)\le\E f^2$，
平均 $z$ 并令 $s_0\downarrow0$，即得 \eqref{eq:conditional-immse}。

\subsection{真实通道的分部积分界}
对只依赖 $(W,X)$ 的向量场 $b$，令
\[
 G_s=\delta b+b^TS_D+\sqrt sN b^Ta,\qquad
 \delta b=\operatorname{div}_Wb-W\cdot b,\quad
 S_D=\nabla_W\log p(\mathcal D_n\mid W).
\]
固定观察并对参数分部积分得
$\E[(\E[f\mid Z,U_s]-f)G_s]=\E b^Ta$。
给定 $(W,X)$，训练得分与辅助噪声独立且中心化，训练得分协方差为
$n\mathcal I_W$。因此 Cauchy--Schwarz 给
\begin{equation}\label{eq:vt-mmse}
 m(s)\ge\frac{(\E b^Ta)^2}
 {n\E b^T\mathcal I_Wb+\E(\delta b)^2+s\E(b^Ta)^2}.
\end{equation}
先对紧支撑光滑场及截断的后验均值证明，再用引理 \ref{an:sobolev}
的 Sobolev 逼近和 $L^2$ 收敛取极限；以下估计保证所需可积性。

\subsection{正则化测试场的 OU 平滑}
固定 $0<\tau\le r\le1$，令 $\lambda=r/p$、
$M_W=(\mathcal I_W+\lambda I_p)^{-1}$、$b_0(W,X)=M_Wa(W,X)$。
取 \eqref{eq:gradient-stability} 中的 $W_\tau$，仅对参数作条件平均：
\[
 b_\tau(W,X)=\E[b_0(W_\tau,X)\mid W,X].
\]
真实网络、训练似然及辅助观测 $U_s$ 均不变。首先
\[
 \E\|b_0\|^2=\Delta\E\tr(M_W^2\mathcal I_W)
 \le\frac{\Delta p}{4\lambda},\qquad
 \E\|b_\tau\|^4\le\E\|b_0\|^4\le C\lambda^{-4}.
\]
将 $b_0$ 视为 $L^2(P_X;\mathbb R^p)$ 值函数，记第 $k$ 阶参数 Hermite
能量为 $c_k$。OU 半群逐阶乘 $e^{-\tau k}$，所以
\[
 \E\|Db_\tau\|_{\rm HS}^2
 =\sum_{k\ge1}ke^{-2\tau k}c_k
 \le C\tau^{-1}\E\|b_0\|^2\le Cp^2/(r\tau).
\]
这直接使 $b_\tau$ 属于高斯 $W^{1,2}$，不需要对 $M_W$ 求导。
由引理 \ref{an:sobolev}、梯度四阶矩及 $\tau\le r$，
\begin{equation}\label{unified:field-cost}
 \E(\delta b_\tau)^2\le Cp^2/(r\tau),\qquad
 \E(b_\tau^Ta)^2\le
 (\E\|b_\tau\|^4\E\|a\|^4)^{1/2}
 \le Cp^2/r^2\le Cp^2/(r\tau).
\end{equation}

写 $b'(x)=b_0(W_\tau,x)$、$a'(x)=a(W_\tau,x)$、
$e(x)=a(W,x)-a'(x)$，并另取独立输入 $\overline X$。
条件于 $(W,W_\tau,\overline X)$，$X$ 仍独立，因此
\[
 \E_Xb'(X)b'(X)^T=\Delta M_{W_\tau}\mathcal I_{W_\tau}M_{W_\tau}
 \preceq\frac{\Delta}{4\lambda}I_p.
\]
由 \eqref{eq:gradient-stability}，$\E\|e(X)\|^2\le C\tau^\delta$。
分别用 Cauchy--Schwarz 和上式条件协方差，得到
\[
 |\E b'(X)^Te(X)|\le Cp\tau^{\delta/2}/\sqrt r,
 \qquad
 \Delta^{-1}\E[(b'(X)^Te(\overline X))^2]
 \le\frac{\E\|e(\overline X)\|^2}{4\lambda}\le Cp\tau^\delta/r.
\]
高斯先验不变性给 $\E b'(X)^Ta'(X)=\Delta p[1-D_d(r)]$；
独立输入给
$\Delta^{-1}\E[(b'(X)^Ta'(\overline X))^2]
 =\Delta\E\tr[(M_W\mathcal I_W)^2]\le\Delta p$。
故条件 Jensen 与 $L^2$ 三角不等式给
\begin{equation}\label{field:target}\begin{aligned}
 \E b_\tau^Ta&=\E b'(X)^T(a'(X)+e(X))
 \ge\Delta p[1-D_d(r)-C\tau^{\delta/2}/\sqrt r],\\
 \E b_\tau^T\mathcal I_Wb_\tau
 &\le\Delta^{-1}\E[(b'(X)^T(a'(\overline X)+e(\overline X)))^2]
 \le\Delta p[1+C\tau^{\delta/2}/\sqrt r]^2.
\end{aligned}\end{equation}
将目标量下界取正部，与 \eqref{unified:field-cost} 一同代入
\eqref{eq:vt-mmse} 并对 \eqref{eq:conditional-immse} 积分。
最后取 $\tau=r^{2/\delta}\le r$，使 $\tau^{\delta/2}/\sqrt r=\sqrt r$，
即得 \eqref{eq:lower-unified}。
